126UART_Status_t UART1_PM25_Recover(PMS25_App_t *app)
127{
128 UART_Status_t uart_st;
129 uint32_t frame_poll_timeout_ms;
130 uint32_t no_frame_fault_ms;
131 uint32_t sample_period_ms;
132 uint32_t recovery_delay_ms;
133 uint8_t max_recovery_attempts;
134 uint8_t recovery_attempt;
135 if (app == NULL)
136 {
137 return UART_STATUS_INVALID_PARAM;
138 }
139 /* Preserve runtime settings across UART reinitialization. */
140 frame_poll_timeout_ms = app->FramePollTimeoutMs;
141 no_frame_fault_ms = app->NoFrameFaultMs;
142 sample_period_ms = app->SamplePeriodMs;
143 recovery_delay_ms = app->RecoveryDelayMs;
144 max_recovery_attempts = app->MaxRecoveryAttempts;
145 recovery_attempt = app->RecoveryAttempt;
146 /* Stop IRQ and DMA activity before tearing the peripheral down. */
147 HAL_NVIC_DisableIRQ(DMA2_Stream5_IRQn);
148 NVIC_ClearPendingIRQ(DMA2_Stream5_IRQn);
149 /* Stop any receive still owned by HAL. */
150 (void)HAL_UART_DMAStop(&huart1);
151 if (huart1.hdmarx != NULL)
152 {
153 (void)HAL_DMA_Abort(huart1.hdmarx);
154 (void)HAL_DMA_DeInit(huart1.hdmarx);
155 }
156 (void)HAL_UART_DeInit(&huart1);
157 UART1_PM25_ResetHandles();
158 /* Rebuild UART and DMA from a clean handle state. */
159 uart_st = UART1_PM25_Init();
160 if (uart_st != UART_STATUS_OK)
161 {
162 return uart_st;
163 }
164 if (PMS25_AppInit(app,
165 frame_poll_timeout_ms,
166 no_frame_fault_ms,
167 sample_period_ms,
168 recovery_delay_ms,
169 max_recovery_attempts) != PMS25_APP_RUNTIME_OK)
170 {
171 return UART_STATUS_ERROR;
172 }
173 /* Keep the caller recovery budget after AppInit resets the context. */
174 app->RecoveryAttempt = recovery_attempt;
175 uart1_pm25_drain_rx_sem();
176 return UART_STATUS_OK;
177}

185UART_Status_t UART1_RX_DMA_Receive(uint8_t *p_data, uint16_t length, uint32_t timeout_ms)
186{
187 HAL_StatusTypeDef hal_st;
188 uint32_t start_tick;
189 if ((p_data == NULL) || (length == 0U) || (timeout_ms == 0U))
190 {
191 return UART_STATUS_INVALID_PARAM;
192 }
193 g_uart1_rx_dma_event = UART1_RX_DMA_EVENT_IDLE;
194 uart1_pm25_drain_rx_sem();
195 /* Arm one bounded DMA receive. */
196 hal_st = HAL_UART_Receive_DMA(&huart1, p_data, length);
197 if (hal_st == HAL_BUSY)
198 {
199 return UART_STATUS_BUSY;
200 }
201 if (hal_st != HAL_OK)
202 {
203 return UART_STATUS_DMA_ERROR;
204 }
205 /* Runtime path blocks until the DMA callback gives the semaphore. */
206 if ((pm25_uart_rx_sem != NULL) &&
207 (xTaskGetSchedulerState() == taskSCHEDULER_RUNNING))
208 {
209 if (xSemaphoreTake(pm25_uart_rx_sem, pdMS_TO_TICKS(timeout_ms)) != pdTRUE)
210 {
211 (void)HAL_UART_DMAStop(&huart1);
212 g_uart1_rx_dma_event = UART1_RX_DMA_EVENT_IDLE;
213 return UART_STATUS_TIMEOUT;
214 }
215 if (g_uart1_rx_dma_event == UART1_RX_DMA_EVENT_DONE)
216 {
217 g_uart1_rx_dma_event = UART1_RX_DMA_EVENT_IDLE;
218 return UART_STATUS_OK;
219 }
220 g_uart1_rx_dma_event = UART1_RX_DMA_EVENT_IDLE;
221 return UART_STATUS_DMA_ERROR;
222 }
223 /* Before the scheduler starts, wait by polling the DMA event. */
224 start_tick = HAL_GetTick();
225 while (g_uart1_rx_dma_event == UART1_RX_DMA_EVENT_IDLE)
226 {
227 if ((HAL_GetTick() - start_tick) >= timeout_ms)
228 {
229 (void)HAL_UART_DMAStop(&huart1);
230 g_uart1_rx_dma_event = UART1_RX_DMA_EVENT_IDLE;
231 return UART_STATUS_TIMEOUT;
232 }
233 cpu_sleep_until_irq();
234 }
235 if (g_uart1_rx_dma_event == UART1_RX_DMA_EVENT_ERROR)
236 {
237 g_uart1_rx_dma_event = UART1_RX_DMA_EVENT_IDLE;
238 return UART_STATUS_DMA_ERROR;
239 }
240 g_uart1_rx_dma_event = UART1_RX_DMA_EVENT_IDLE;
241 return UART_STATUS_OK;
242}
