144jitter_status_t jitter_collect_samples(void)
145{
146 uint32_t i;
147 uint32_t timeout;
148 idx_jitter = 0;
149 memset((void*)jitter_differences, 0, sizeof(jitter_differences));
150 for (i = 0; i < JITTER_CAPTURE_SIZE; i++)
151 {
152 g_jitter_capture_complete = 0;
153 jitter_status_t st = jitter_prepare_dma_sample(i);
154 if (st != JITTER_OK) return st;
155 /* Vary the software delay before each capture to perturb sampling time. */
156 uint32_t seed = TIM1->CNT ^ (i * 0x45D9F3BU);
157 seed ^= (seed << 13);
158 seed ^= (seed >> 17);
159 seed ^= (seed << 5);
160 uint32_t delay = (seed & 0xFFU) + 1U;
161 for (volatile uint32_t d = 0; d < delay * 50U; d++) { __NOP(); }
162 jitter_trigger_capture_event();
163 timeout = JITTER_SAMPLE_TIMEOUT;
164 /* Wait for the DMA transfer-complete interrupt with a bounded timeout. */
165 while (!g_jitter_capture_complete && timeout--)
166 {
167 __NOP();
168 }
169 if (timeout == 0U)
170 {
171 jitter_wait_dma_disable();
172 return JITTER_ERR_DMA_TRANSFER;
173 }
174 jitter_wait_dma_disable();
175 }
176 return jitter_compute_differences();

182jitter_status_t jitter_compute_differences(void)
183{
184 if (JITTER_CAPTURE_SIZE < 2) return JITTER_ERR_NO_SAMPLES;
185 for (uint32_t i = 0; i < (JITTER_CAPTURE_SIZE - 1); i++)
186 {
187 uint32_t t1 = capture_buffer[i] & 0xFFFFU;
188 uint32_t t2 = capture_buffer[i + 1] & 0xFFFFU;
189 uint32_t diff;
190 /* Calculate the delta while accounting for the 16-bit timer wrap. */
191 if (t2 >= t1) diff = t2 - t1;
192 else diff = (0xFFFFU - t1) + t2 + 1U;
193 jitter_differences[i] = diff;
194 }
195 jitter_differences[JITTER_CAPTURE_SIZE - 1] = 0;
196 idx_jitter = JITTER_CAPTURE_SIZE - 1;
197 return JITTER_OK;
198}

218uint32_t jitter_assess_entropy(void)
219{
220 uint32_t zero = 0, one = 0, small = 0, large = 0;
221 for (uint32_t i = 0; i < JITTER_CAPTURE_SIZE - 1; i++)
222 {
223 uint32_t d = jitter_differences[i];
224 if (d == 0) continue;
225 if (d & 1U) one++; else zero++;
226 if (d < 10U) small++;
227 if (d > 1000U) large++;
228 }
229 uint32_t samples = zero + one;
230 if (samples == 0) return 0;
231 int32_t diff = (int32_t)zero - (int32_t)one;
232 if (diff < 0) diff = -diff;
233 /* Use bit balance as the base heuristic, then penalize extreme distributions. */
234 uint32_t score = 100U - (100U * (uint32_t)diff) / samples;
235 if (small > (samples / 2U)) score = (score > 20U) ? score - 20U : 0U;
236 if (large > (samples / 10U)) score = (score > 10U) ? score - 10U : 0U;
237 return score;
238}
