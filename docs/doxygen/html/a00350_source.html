371 const uint8_t *data,
372 size_t len,
373 uint32_t timeout_ms)
374{
375 int written;
376
377 if ((data == NULL) ||
378 (len == 0U))
379 {
380 return STM32_UART_STATUS_INVALID_PARAM;
381 }
382
383 if (!g_stm32_uart_initialized)
384 {
385 return STM32_UART_STATUS_NOT_INITIALIZED;
386 }
387
388 written =
389 uart_write_bytes(
390 STM32_UART_PORT,
391 (const char *)data,
392 len);
393
394 if (written < 0)
395 {
396 return STM32_UART_STATUS_ERROR;
397 }
398
399 if ((size_t)written != len)
400 {
401 return STM32_UART_STATUS_TIMEOUT;
402 }
403
404 if (uart_wait_tx_done(
405 STM32_UART_PORT,
406 pdMS_TO_TICKS(timeout_ms)) != ESP_OK)
407 {
408 return STM32_UART_STATUS_TIMEOUT;
409 }
410
411 g_stats.tx_bytes +=
412 (uint32_t)written;
413
414 g_stats.tx_frames++;
415
416 return STM32_UART_STATUS_OK;
417}

717

727 (queued_packet.len > packet_capacity))
728 {
729 return false;
730 }
731
732 memcpy(
733 packet,
734 queued_packet.bytes,
735 queued_packet.len);
736
737 *packet_len =
738 queued_packet.len;
739
740 return true;

830static void STM32_UART_EventTask(void *arg)
831{
832 (void)arg;
833
834 uart_event_t event;
835 uint8_t rx_data[STM32_UART_TEST_MAX_FRAME_SIZE];
836 int64_t last_health_us = esp_timer_get_time();
837
838 for (;;)
839 {
840 if (xQueueReceive(
841 g_stm32_uart_event_queue,
842 &event,
843 pdMS_TO_TICKS(STM32_UART_EVENT_POLL_MS)) == pdTRUE)
844 {
845 switch (event.type)
846 {
847 case UART_DATA:
848 {
849 size_t remaining =
850 event.size;
851
852 while (remaining > 0U)
853 {
854 size_t chunk_size =
855 remaining;
856
857 size_t bytes_read = 0U;
858 STM32_UART_Status_t status;
859
860 if (chunk_size > sizeof(rx_data))
861 {
862 chunk_size =
863 sizeof(rx_data);
864 }
865
866 memset(
867 rx_data,
868 0,
869 sizeof(rx_data));
870
871 status =
872 STM32_UART_Read(
873 rx_data,
874 chunk_size,
875 STM32_UART_READ_TIMEOUT_MS,
876 &bytes_read);
877
878 if ((status != STM32_UART_STATUS_OK) ||
879 (bytes_read == 0U))
880 {
881 break;
882 }
883
884 /* Subtract the bytes consumed from this event. */
885 if (bytes_read >= remaining)
886 {
887 remaining = 0U;
888 }
889 else
890 {
891 remaining -= bytes_read;
892 }
893
894 /* Discard the entire chunk while the quiet window is active. */
895 if (STM32_UART_ResynchronizationActive())
896 {
897 ESP_LOGW(
898 TAG,
899 "Discarding %u UART bytes during "
900 "resynchronization",
901 (unsigned int)bytes_read);
902
903 continue;
904 }
905
906 STM32_UART_ConsumeEncryptedBytes(
907 rx_data,
908 bytes_read);
909 }
910
911 break;
912 }
913
914 case UART_FIFO_OVF:
915 g_stats.fifo_overflows++;
916
917 ESP_LOGW(
918 TAG,
919 "UART FIFO overflow; discarding partial packet");
920
921 STM32_UART_EnterResynchronization();
922 break;
923
924 case UART_BUFFER_FULL:
925 g_stats.buffer_full++;
926
927 ESP_LOGW(
928 TAG,
929 "UART buffer full; discarding partial packet");
930
931 STM32_UART_EnterResynchronization();
932 break;
933
934 case UART_PARITY_ERR:
935 g_stats.parity_errors++;
936
937 ESP_LOGW(
938 TAG,
939 "UART parity error; discarding partial packet");
940
941 STM32_UART_EnterResynchronization();
942 break;
943
944 case UART_FRAME_ERR:
945 g_stats.frame_errors++;
946
947 ESP_LOGW(
948 TAG,
949 "UART frame error; discarding partial packet");
950
951 STM32_UART_EnterResynchronization();
952 break;
953
954 case UART_BREAK:
955 ESP_LOGW(
956 TAG,
957 "UART break detected; discarding partial packet");
958
959 STM32_UART_EnterResynchronization();
960 break;
961
962 case UART_PATTERN_DET:
963 g_stats.pattern_detected++;
964 break;
965
966 default:
967 g_stats.unknown_events++;
968 break;
969 }
970 }
971
972 /* Discard stalled partial packets after an interrupted STM32 transfer. */
973 STM32_UART_CheckPartialAssemblyTimeout();
974
975 {
976 const int64_t now_us = esp_timer_get_time();
977
978 if ((now_us - last_health_us) >= STM32_UART_HEALTH_PERIOD_US)
979 {
980 STM32_UART_LinkTest_PrintHealth();
981 last_health_us = now_us;
982 }
983 }
984 }
985}

993static void STM32_UART_ConsumeEncryptedBytes(
994 const uint8_t *data,
995 size_t len)
996{
997 if ((data == NULL) ||
998 (len == 0U))
999 {
1000 return;
1001 }
1002
1003 /* Track the last accepted RX burst for the partial-packet timeout. */
