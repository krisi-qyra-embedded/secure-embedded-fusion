68void clock_96_mhz_init(void)
69{
70 /* 1) Enable HSE */
71 RCC->CR |= RCC_CR_HSEON;
72 while ((RCC->CR & RCC_CR_HSERDY) == 0U) {
73 /* wait */
74 }
75
76 /* 2) Enable power interface clock */
77 RCC->APB1ENR |= RCC_APB1ENR_PWREN;
78 (void)RCC->APB1ENR;
79
80 /* 3) Configure Flash for high-speed operation before switching clock */
81 clock_set_flash_latency_96mhz();
82
83 /*
84 * 4) Configure bus prescalers
85 * AHB = SYSCLK / 1 = 96 MHz
86 * APB1 = HCLK / 2 = 48 MHz
87 * APB2 = HCLK / 1 = 96 MHz
88 */
89 RCC->CFGR &= ~(RCC_CFGR_HPRE | RCC_CFGR_PPRE1 | RCC_CFGR_PPRE2);
90 RCC->CFGR |= RCC_CFGR_HPRE_DIV1
91 | RCC_CFGR_PPRE1_DIV2
92 | RCC_CFGR_PPRE2_DIV1;
93
94 /* 5) Disable PLL before reconfiguring it (safe if previously enabled) */
95 if ((RCC->CR & RCC_CR_PLLON) != 0U) {
96 RCC->CR &= ~RCC_CR_PLLON;
97
98 while ((RCC->CR & RCC_CR_PLLRDY) != 0U) {
99 /* wait */
100 }
101 }
102
103 /* 6) Configure PLL for 96 MHz SYSCLK from 8 MHz HSE */
104 RCC->PLLCFGR =
105 ((uint32_t)PLL_M << RCC_PLLCFGR_PLLM_Pos)
106 | ((uint32_t)PLL_N << RCC_PLLCFGR_PLLN_Pos)
107 | ((((uint32_t)PLL_P >> 1U) - 1U) << RCC_PLLCFGR_PLLP_Pos)
108 | RCC_PLLCFGR_PLLSRC_HSE
109 | ((uint32_t)PLL_Q << RCC_PLLCFGR_PLLQ_Pos);
110
111 /* 7) Enable PLL */
112 RCC->CR |= RCC_CR_PLLON;
113
114 while ((RCC->CR & RCC_CR_PLLRDY) == 0U) {
115 /* wait */
116 }
117
118 /* 8) Switch SYSCLK to PLL */
119 RCC->CFGR &= ~RCC_CFGR_SW;
120 RCC->CFGR |= RCC_CFGR_SW_PLL;
121
122 while ((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_PLL) {
123 /* wait */
124 }
125
126 /* 9) Manual core clock update: no SystemCoreClockUpdate() dependency */
127 SystemCoreClock = SYSCLK_96MHZ;
128
129 __DSB();
130 __ISB();
131}
