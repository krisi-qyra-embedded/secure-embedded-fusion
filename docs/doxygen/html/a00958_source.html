49void dma2_ADC_pc45(uint16_t *mem_buf, uint32_t sample_count)
50{
51 /* Basic argument check */
52 if (mem_buf == NULL || sample_count == 0) {
53 return;
54 }
55
56 /* 1) Enable GPIOC and GPIOB clocks */
57 RCC->AHB1ENR |= RCC_AHB1ENR_GPIOCEN;
58 RCC->AHB1ENR |= RCC_AHB1ENR_GPIOBEN;
59
60 __DSB();
61 __ISB();
62
63
64 /* Configure PC4 and PC5 as analog, no pull/AF */
65 GPIOC->MODER &= ~((3U << (4 * 2)) | (3U << (5 * 2)));
66 GPIOC->MODER |= ((3U << (4 * 2)) | (3U << (5 * 2)));
67
68 GPIOC->PUPDR &= ~((3U << (4 * 2)) | (3U << (5 * 2)));
69 GPIOC->OSPEEDR &= ~((3U << (4 * 2)) | (3U << (5 * 2)));
70 GPIOC->OTYPER &= ~((1U << 4) | (1U << 5));
71 GPIOC->AFR[0] &= ~((0xFU << (4 * 4)) | (0xFU << (5 * 4)));
72
73
74 /* Configure PB0 as analog, no pull/AF */
75 GPIOB->MODER &= ~(3U << (0 * 2));
76 GPIOB->MODER |= (3U << (0 * 2));
77
78 GPIOB->PUPDR &= ~(3U << (0 * 2));
79 GPIOB->OSPEEDR &= ~(3U << (0 * 2));
80 GPIOB->OTYPER &= ~(1U << 0);
81 GPIOB->AFR[0] &= ~(0xFU << (0 * 4));
82
83 __DSB();
84 __ISB();
85
86
87 /* 2) Enable ADC1 clock */
88 RCC->APB2ENR |= RCC_APB2ENR_ADC1EN;
89
90 __DSB();
91 __ISB();
92
93
94 /* Settle after clock enable */
95 for (volatile int i = 0; i < 200; ++i) {
96 __NOP();
97 }
98
99
100 /* 3) ADC sample time: 480 cycles for all channels */
101
102 /* CH14 and CH15 are in SMPR1 */
103 ADC1->SMPR1 &=
104 ~((7U << ((14 - 10) * 3)) |
105 (7U << ((15 - 10) * 3)));
106
107 ADC1->SMPR1 |=
108 ((7U << ((14 - 10) * 3)) |
109 (7U << ((15 - 10) * 3)));
110
111
112 /* CH8 is in SMPR2 */
113 ADC1->SMPR2 &= ~(7U << (8 * 3));
114 ADC1->SMPR2 |= (7U << (8 * 3));
115
116
117 /*
118 * 4) Regular sequence:
119 * CH14 -> CH15 -> CH8
120 *
121 * L = 2 means 3 conversions.
122 */
123 ADC1->SQR1 &= ~(0xFUL << 20);
124 ADC1->SQR1 |= (2U << 20);
125
126 ADC1->SQR3 = 0;
127 ADC1->SQR3 |= (14U << 0); /* SQ1 = CH14 (PC4) */
128 ADC1->SQR3 |= (15U << 5); /* SQ2 = CH15 (PC5) */
129 ADC1->SQR3 |= (8U << 10); /* SQ3 = CH8 (PB0) */
130
131
132 /* 5) Enable ADC scan mode and DMA */
133 ADC1->CR1 |= ADC_CR1_SCAN;
134
135 ADC1->CR2 |= ADC_CR2_CONT;
136 ADC1->CR2 |= ADC_CR2_DMA;
137 ADC1->CR2 |= ADC_CR2_DDS;
138
139 __DSB();
140 __ISB();
141
142
143 /* 6) Configure DMA2 Stream4 Channel 0 */
144 RCC->AHB1ENR |= RCC_AHB1ENR_DMA2EN;
145
146 __DSB();
147 __ISB();
148
149
150 /* Disable DMA stream before configuration */
151 DMA2_Stream4->CR &= ~DMA_SxCR_EN;
152
153 while (DMA2_Stream4->CR & DMA_SxCR_EN) {
154 __NOP();
155 }
156
157
158 /* Clear DMA Stream4 flags */
159 DMA2->HIFCR =
160 DMA_HIFCR_CTCIF4
161 | DMA_HIFCR_CHTIF4
162 | DMA_HIFCR_CTEIF4
163 | DMA_HIFCR_CDMEIF4
164 | DMA_HIFCR_CFEIF4;
165
166
167 /*
168 * DMA configuration:
169 * - Channel 0
170 * - Peripheral to memory
171 * - 16-bit transfers
172 * - Memory increment
173 * - Circular mode
174 * - High priority
175 * - Transfer complete interrupt
176 */
177 uint32_t cr = 0;
178
179 cr |= (0U << DMA_SxCR_CHSEL_Pos);
180 cr &= ~(3U << DMA_SxCR_DIR_Pos);
181
182 cr |= DMA_SxCR_MINC;
183 cr &= ~DMA_SxCR_PINC;
184
185
186 /* PSIZE = 16-bit, MSIZE = 16-bit */
187 cr &= ~(3U << DMA_SxCR_PSIZE_Pos);
188 cr |= (1U << DMA_SxCR_PSIZE_Pos);
189
190 cr &= ~(3U << DMA_SxCR_MSIZE_Pos);
191 cr |= (1U << DMA_SxCR_MSIZE_Pos);
192
193 cr |= DMA_SxCR_CIRC;
194 cr |= (2U << DMA_SxCR_PL_Pos);
195 cr |= DMA_SxCR_TCIE;
196
197 DMA2_Stream4->CR = cr;
198
199
200 /* Set peripheral and memory addresses */
201 DMA2_Stream4->PAR = (uint32_t)(&ADC1->DR);
202 DMA2_Stream4->M0AR = (uint32_t)mem_buf;
203
204
205 /*
206 * Total transfers = sample_count * 3
207 * PC4, PC5 and PB0 per sequence.
208 */
209 DMA2_Stream4->NDTR = sample_count * 3U;
210
211
212 /* Enable DMA2 Stream4 interrupt */
213 NVIC_SetPriority(DMA2_Stream4_IRQn, 3);
214 NVIC_ClearPendingIRQ(DMA2_Stream4_IRQn);
215 NVIC_EnableIRQ(DMA2_Stream4_IRQn);
216
217
218 /* 7) Enable DMA stream */
219 __DSB();
220 __ISB();
221
222 DMA2_Stream4->CR |= DMA_SxCR_EN;
223
224
225 /* 8) Enable ADC */
226 ADC1->CR2 |= ADC_CR2_ADON;
227
228
229 /* Wait for ADC stabilization */
230 for (volatile int i = 0; i < 5000; ++i) {
231 __NOP();
232 }
233
234
235 /* Clear pending ADC status flags */
236 ADC1->SR = 0;
237
238
239 /* 9) Start continuous conversions */
240 ADC1->CR2 |= ADC_CR2_SWSTART;
241
242 //printf("ADC+DMA continuous mode started (PC4=CH14, PC5=CH15, PB0=CH8)\r\n");
243}

274bool store_adc_byte_values(uint16_t pc4_val,
275 uint16_t pc5_val,
276 uint16_t pb0_val)
277{
278 if (idx_pc4 < BUF_SZ) {
279 buf_pc4[idx_pc4++] = (uint8_t)(pc4_val & 0xFF);
280 }
281
282 if (idx_pc5 < BUF_SZ) {
283 buf_pc5[idx_pc5++] = (uint8_t)(pc5_val & 0xFF);
284 }
285
286 if (idx_pb0 < BUF_SZ) {
287 buf_pb0[idx_pb0++] = (uint8_t)(pb0_val & 0xFF);
288 }
289
290 return (idx_pc4 >= BUF_SZ) &&
291 (idx_pc5 >= BUF_SZ) &&
292 (idx_pb0 >= BUF_SZ);

331int collect_adc_samples_until_full(void)
332{
333 for (;;) {
334
335 /* Wait for DMA transfer complete flag */
336 if (!g_transfer_complete) {
337 continue;
338 }
339
340
341 /* Atomically consume flag to avoid races with IRQ */
342 __disable_irq();
343
344 g_transfer_complete = 0;
345
346 __enable_irq();
347
348
349 /* Extract latest ADC triplet */
350 adc_get_latest_triplet(
351 (uint16_t *)adc_buffer,
352 SAMPLE_COUNT,
353 &pc4_val,
354 &pc5_val,
355 &pb0_val);
356
357
358 /* Store values until all buffers are full */
359 all_full =
360 store_adc_byte_values(
361 pc4_val,
362 pc5_val,
363 pb0_val);
364
365
366 if (all_full) {
367 return 0;
368 }
369 }
370
371
372 /* Unreachable */
373 return -1;
374}

386int ADC_DMA_INIT(void)
387{
388 int rv = -1;
389 int attempt;
390
391
392 for (attempt = 0;
393 attempt < ADC_DMA_INIT_RETRY_COUNT;
394 ++attempt) {
395
396 /* Start DMA sampling */
397 dma2_ADC_pc45(
398 (uint16_t *)adc_buffer,
399 SAMPLE_COUNT);
400
401
402 /* Small delay to allow initial conversions */
403 for (volatile int i = 0;
404 i < 1000000;
405 ++i) {
406
407 __NOP();
408 }
409
410
411 /* Collect until buffers are full */
412 rv = collect_adc_samples_until_full();
413
414 if (rv == 0) {
415 break;
416 }
417
418
419 log_printf(
420 "ADC_DMA_INIT: attempt %d - "
421 "collect_adc_samples_until_full failed (rv=%d)\n",
422 attempt + 1,
423 rv);
424 }
425
426
427 if (rv != 0) {
428
429 log_printf(
430 "ADC_DMA_INIT: ADC collection ultimately failed.\n");
431
432 return ADC_DMA_INIT_COLLECTION_FAILED;
433 }
434
435
436 return ADC_DMA_INIT_SUCCESS;
437}
