136static bool ccs811_init_desc_with_retries(ccs811_dev_t *dev)
137{
138 esp_err_t err = ESP_FAIL;
139
140 for (uint32_t attempt = 1U; attempt <= CCS811_MAX_INIT_RETRIES; attempt++)
141 {
142 ESP_LOGI(TAG,
143 "CCS811_INIT | step=descriptor | attempt=%lu/%u",
144 (unsigned long)attempt,
145 CCS811_MAX_INIT_RETRIES);
146
147 err = ccs811_init_desc(dev,
148 CCS811_I2C_ADDR,
149 CCS811_I2C_PORT,
150 CCS811_I2C_SDA,
151 CCS811_I2C_SCL);
152
153 if (err == ESP_OK)
154 {
155 ESP_LOGI(TAG, "CCS811_INIT | step=descriptor | status=ok");
156 return true;
157 }
158
159 ESP_LOGW(TAG,
160 "CCS811_INIT_RETRY | step=descriptor | err=%d - %s | attempt=%lu/%u",
161 err,
162 esp_err_to_name(err),
163 (unsigned long)attempt,
164 CCS811_MAX_INIT_RETRIES);
165
166 /* Keep the heartbeat active during descriptor retries. */
167 app_watchdog_kick(APP_WD_CCS811);
168 vTaskDelay(pdMS_TO_TICKS(CCS811_RECOVERY_DELAY_MS));
169 }
170
171 ESP_LOGE(TAG,
172 "CCS811_INIT_FAIL | step=descriptor | final_err=%d - %s",
173 err,
174 esp_err_to_name(err));
175
176 return false;
177}

187{
188 esp_err_t err = ESP_FAIL;
189
190 for (uint32_t attempt = 1U; attempt <= CCS811_MAX_INIT_RETRIES; attempt++)
191 {

194 phase,
195 (unsigned long)attempt,
196 CCS811_MAX_INIT_RETRIES);
197
198 err = ccs811_init(dev);
199
200 if (err == ESP_OK)
201 {
202 ESP_LOGI(TAG,
203 "CCS811_%s | step=sensor_init | status=ok",
204 phase);
205 break;
206 }
207
208 ESP_LOGW(TAG,
209 "CCS811_%s_RETRY | step=sensor_init | err=%d - %s | attempt=%lu/%u",
210 phase,
211 err,
212 esp_err_to_name(err),
213 (unsigned long)attempt,
214 CCS811_MAX_INIT_RETRIES);
215
216 /* Keep the heartbeat active while initialization retries remain. */
217 app_watchdog_kick(APP_WD_CCS811);
218 vTaskDelay(pdMS_TO_TICKS(CCS811_RECOVERY_DELAY_MS));
219 }
220
221 if (err != ESP_OK)
222 {
223 ESP_LOGE(TAG,
224 "CCS811_%s_FAIL | step=sensor_init | final_err=%d - %s",
225 phase,
226 err,
227 esp_err_to_name(err));
228 return false;
229 }
230
231 err = ESP_FAIL;
232

234 {
235 ESP_LOGI(TAG,
236 "CCS811_%s | step=set_mode | attempt=%lu/%u",
237 phase,
238 (unsigned long)attempt,
239 CCS811_MAX_INIT_RETRIES);
240
241 err = ccs811_set_mode(dev, CCS811_MODE_1S);
242
243 if (err == ESP_OK)
244 {
245 ESP_LOGI(TAG,
246 "CCS811_%s | step=set_mode | status=ok",
247 phase);
248 return true;
249 }
250
251 ESP_LOGW(TAG,
252 "CCS811_%s_RETRY | step=set_mode | err=%d - %s | attempt=%lu/%u",
253 phase,
254 err,
255 esp_err_to_name(err),
256 (unsigned long)attempt,
257 CCS811_MAX_INIT_RETRIES);
258
259 app_watchdog_kick(APP_WD_CCS811);
260 vTaskDelay(pdMS_TO_TICKS(CCS811_RECOVERY_DELAY_MS));
261 }
262
263 ESP_LOGE(TAG,
264 "CCS811_%s_FAIL | step=set_mode | final_err=%d - %s",
265 phase,
266 err,
267 esp_err_to_name(err));

277static bool ccs811_try_recover(ccs811_dev_t *dev)
278{
279 ESP_LOGW(TAG, "CCS811_RECOVERY | trying to reinitialize sensor");
280
281 ccs811_mark_latest_invalid();
282
283 /* Retry initialization and measurement mode before escalating. */
284 bool recovered = ccs811_sensor_init_and_mode_with_retries(dev, "RECOVERY");
285
286 if (!recovered)
287 {
288 ESP_LOGE(TAG, "CCS811_RECOVERY | recovery failed after retries");
289 return false;
290 }
291
292 s_consecutive_severe_failures = 0;
293 s_soft_fault_count = 0;
294 s_no_new_data_count = 0;
295
296 ESP_LOGW(TAG, "CCS811_RECOVERY | sensor recovered successfully");
297
298 return true;
299}

305static void ccs811_test(void *pvParameters)
306{
307 (void)pvParameters;
308
309 ccs811_dev_t dev;
310 memset(&dev, 0, sizeof(ccs811_dev_t));
311
312 ESP_LOGI(TAG, "CCS811 descriptor initialization");
313
314 if (!ccs811_init_desc_with_retries(&dev))
315 {
316 ccs811_fatal_stop_kicking_watchdog("descriptor initialization failed after retries");
317 }
318
319 ESP_LOGI(TAG, "CCS811 sensor initialization and measurement mode setup");
320
321 if (!ccs811_sensor_init_and_mode_with_retries(&dev, "INIT"))
322 {
323 ccs811_fatal_stop_kicking_watchdog("sensor init/set_mode failed after retries");
324 }
325
326 ESP_LOGI(TAG, "CCS811 started successfully");
327 ESP_LOGI(TAG, "Waiting for gas measurements...");
328
329 /* Refresh the heartbeat after sensor setup. */
330 app_watchdog_kick(APP_WD_CCS811);
331
332 while (1)
333 {
334 uint16_t tvoc = 0;
335 uint16_t eco2 = 0;
336 uint8_t raw_current = 0;
337 uint16_t raw_voltage_adc = 0;
338
339 esp_err_t res = ccs811_get_results(&dev,
340 &tvoc,
341 &eco2,
342 &raw_current,
343 &raw_voltage_adc);
344
345 if (res == ESP_OK)
346 {
347 s_consecutive_severe_failures = 0;
348 s_soft_fault_count = 0;
349 s_no_new_data_count = 0;
350
351 float raw_voltage = ((float)raw_voltage_adc * 1.65f) / 1023.0f;
352
353 ccs811_sensor_data_t latest = {
354 .valid = true,
355 .eco2_ppm = eco2,
356 .tvoc_ppb = tvoc,
357 .raw_current_ua = raw_current,
358 .raw_voltage_adc = raw_voltage_adc,
359 .raw_voltage = raw_voltage
360 };
361
362 portENTER_CRITICAL(&g_ccs811_mux);
363 g_ccs811_latest = latest;
364 portEXIT_CRITICAL(&g_ccs811_mux);
365
366 ESP_LOGI(TAG,
367 "eCO2=%u ppm | TVOC=%u ppb | I=%u uA | ADC=%u | V=%.3f",
368 eco2,
369 tvoc,
370 raw_current,
371 raw_voltage_adc,
372 raw_voltage);
373
374 /* Refresh the heartbeat after a valid sample. */
375 app_watchdog_kick(APP_WD_CCS811);
376 }
377 else if (res == CCS811_ERR_NO_NEW_DATA)
378 {
379 /* No sample ready yet; keep the task healthy. */
380 s_consecutive_severe_failures = 0;
381 s_no_new_data_count++;
382
383 ESP_LOGI(TAG,
384 "CCS811: no new data yet | count=%lu",
385 (unsigned long)s_no_new_data_count);
386
387 app_watchdog_kick(APP_WD_CCS811);
388 }
389 else if (ccs811_is_soft_fault(res))
390 {
391 /* Invalidate the reading on a range fault without restarting. */
392 s_consecutive_severe_failures = 0;
393 s_soft_fault_count++;
394
395 ccs811_mark_latest_invalid();
396
397 ESP_LOGW(TAG,
398 "CCS811_SOFT_FAULT | sensor alive but gas reading invalid | err=%d - %s | soft_fault_count=%lu | action=no_reset",
399 res,
400 esp_err_to_name(res),
401 (unsigned long)s_soft_fault_count);
402
403 app_watchdog_kick(APP_WD_CCS811);
404 }
405 else
406 {
407 /* Count communication failures and recover at the retry limit. */
408 s_consecutive_severe_failures++;
409
410 ESP_LOGE(TAG,
411 "CCS811_SEVERE_READ_FAIL | err=%d - %s | consecutive_failures=%lu/%u",
412 res,
413 esp_err_to_name(res),
414 (unsigned long)s_consecutive_severe_failures,
415 CCS811_MAX_CONSECUTIVE_FAILURES);
416
417 if (s_consecutive_severe_failures >= CCS811_MAX_CONSECUTIVE_FAILURES)
418 {
419 ESP_LOGE(TAG,
420 "CCS811_FATAL_CANDIDATE | too many consecutive severe failures | attempting recovery");
421
422 bool recovered = ccs811_try_recover(&dev);
423
424 if (recovered)
425 {
426 /* Refresh the heartbeat after recovery. */
427 app_watchdog_kick(APP_WD_CCS811);
428 }
429 else
430 {
431 /* Escalate failed recovery through the watchdog. */
432 ccs811_fatal_stop_kicking_watchdog("recovery failed after consecutive severe failures");
433 }
434 }
435 else
436 {
437 /* Keep the heartbeat active below the recovery threshold. */
438 app_watchdog_kick(APP_WD_CCS811);
439 }
440 }
441
442 /* Wait slightly longer than the one-second measurement period. */
443 vTaskDelay(pdMS_TO_TICKS(CCS811_SAMPLE_DELAY_MS));
444 }
