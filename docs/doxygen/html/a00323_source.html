110static esp_err_t http_server_send_response_checked(httpd_req_t *req,
111 const char *data,
112 size_t data_len)
113{
114esp_err_t err = httpd_resp_send(req, data, data_len);
115if (err == ESP_OK)
116{
117 s_https_consecutive_failures = 0;
118 http_server_watchdog_kick();
119 return ESP_OK;
120}
121s_https_consecutive_failures++;
122ESP_LOGE(TAG,
123 "HTTPS_RESP_FAIL | err=%d - %s | consecutive_failures=%lu/%u",
124 err,
125 esp_err_to_name(err),
126 (unsigned long)s_https_consecutive_failures,
127 HTTPS_MAX_CONSECUTIVE_FAILURES);
128if (s_https_consecutive_failures >= HTTPS_MAX_CONSECUTIVE_FAILURES)
129{
130 http_server_fatal_stop_kicking_watchdog("too many HTTPS response failures");
131}
132return err;
133}

446esp_err_t http_server_OTA_update_handler(httpd_req_t *req)
447{
448esp_ota_handle_t ota_handle;
449char ota_buff[1024];
450int content_length = req->content_len;
451int content_received = 0;
452int recv_len;
453bool is_req_body_started = false;
454bool flash_successful = false;
455const esp_partition_t *update_partition = esp_ota_get_next_update_partition(NULL);
456ESP_LOGI(TAG, "OTA_UPLOAD | event=start | content_length=%d | target_partition_ptr=%p",
457 content_length, (void *)update_partition);
458http_server_watchdog_kick();
459do
460{
461 /* Receive the next upload chunk. */
462 if ((recv_len = httpd_req_recv(req, ota_buff, MIN(content_length, sizeof(ota_buff)))) < 0)
463 {
464 /* Retry socket timeouts. */
465 if (recv_len == HTTPD_SOCK_ERR_TIMEOUT)
466 {
467 ESP_LOGW(TAG, "OTA_UPLOAD | event=recv_timeout | action=retry | received_bytes=%d | expected_bytes=%d", content_received, content_length);
468 continue; /* Retry this receive after the timeout. */
469 }
470 ESP_LOGE(TAG, "OTA_UPLOAD | event=recv_failed | error_code=%d | received_bytes=%d | expected_bytes=%d", recv_len, content_received, content_length);
471 return ESP_FAIL;
472 }
473 ESP_LOGD(TAG, "OTA_UPLOAD | event=chunk_received | received_bytes=%d | expected_bytes=%d", content_received, content_length);
474 /* The first chunk includes the multipart header. */
475 /* Locate the firmware body before writing to flash. */
476 if (!is_req_body_started)
477 {
478 is_req_body_started = true;
479 /* Skip the multipart header. */
480 char *body_start_p = strstr(ota_buff, "\r\n\r\n");
481 if (body_start_p == NULL)
482 {
483 ESP_LOGE(TAG, "OTA_UPLOAD | event=parse_failed | reason=missing_multipart_body_separator");
484 return ESP_FAIL;
485 }
486 body_start_p += 4;
487 int body_part_len = recv_len - (body_start_p - ota_buff);
488 ESP_LOGI(TAG, "OTA_UPLOAD | event=body_detected | total_request_bytes=%d | first_body_chunk_bytes=%d", content_length, body_part_len);
489 esp_err_t err = esp_ota_begin(update_partition, OTA_SIZE_UNKNOWN, &ota_handle);
490 if (err != ESP_OK)
491 {
492 ESP_LOGE(TAG, "OTA_FLASH | event=begin_failed | error=%s | partition_ptr=%p", esp_err_to_name(err), (void *)update_partition);
493 return ESP_FAIL;
494 }
495 else
496 {
497 ESP_LOGI(TAG, "OTA_FLASH | event=begin_success | partition_subtype=%d | partition_addr=0x%08lx", update_partition->subtype, update_partition->address);
498 }
499 /* Write the first firmware bytes. */
500 esp_ota_write(ota_handle, body_start_p, body_part_len);
501 content_received += body_part_len;
502 http_server_watchdog_kick();
503 }
504 else
505 {
506 /* Write the remaining firmware chunks. */
507 esp_ota_write(ota_handle, ota_buff, recv_len);
508 content_received += recv_len;
509 http_server_watchdog_kick();
510 }
511} while (recv_len > 0 && content_received < content_length);
512if (esp_ota_end(ota_handle) == ESP_OK)
513{
514 /* Select the updated image for the next boot. */
515 if (esp_ota_set_boot_partition(update_partition) == ESP_OK)
516 {
517 const esp_partition_t *boot_partition = esp_ota_get_boot_partition();
518 ESP_LOGI(TAG, "OTA_FLASH | event=boot_partition_updated | next_partition_subtype=%d | next_partition_addr=0x%08lx", boot_partition->subtype, boot_partition->address);
519 flash_successful = true;
520 }
521 else
522 {
523 ESP_LOGE(TAG, "OTA_FLASH | event=set_boot_partition_failed | partition_subtype=%d | partition_addr=0x%08lx", update_partition->subtype, update_partition->address);
524 }
525}
526else
527{
528 ESP_LOGE(TAG, "OTA_FLASH | event=end_failed | received_bytes=%d | expected_bytes=%d", content_received, content_length);
529}
530/* Send the OTA result through the monitor queue. */
531if (flash_successful)
532{
533 http_server_monitor_send_message(HTTP_MSG_OTA_UPDATE_SUCCESSFUL);
534}
535else
536{
537 http_server_monitor_send_message(HTTP_MSG_OTA_UPDATE_FAILED);
538}
539return ESP_OK;
540}

899static void http_server_json_sanitize_ascii(const char *src,

901 size_t dst_size)
902{

918 }
