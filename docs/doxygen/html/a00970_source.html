56 out_len_local = out32_len;

58 device_uid_buffer, sizeof(device_uid_buffer),
59 out32, out32_len,
60 &out_len_local);
61 if (retval != CMOX_HASH_SUCCESS || out_len_local != out32_len) {
62 /* Clear partial hash on failure. */
63 memset(out32, 0, out32_len);
64 return DEVICE_UID_HASH_ERR_HASH_FAILED;
65 }
66 return DEVICE_UID_HASH_SUCCESS;

336/* Entropy pipeline */
337
343int entropy_personalisation_hash(void)
344{
345 int rv;
346 int attempt;
347 /* Clear old entropy first. */
348 secure_zero(entropy_hashed, ENTROPY_HASH_LEN);
349 /* ADC sources. */
350 log_printf("entropy_personalisation_hash: starting ADC hashes\r\n");
351 rv = compute_all_adc_hashes();
352 if (rv != ADC_HASH_ALL_SUCCESS) {
353 log_printf("entropy_personalisation_hash: compute_all_adc_hashes failed (rv=%d)\r\n", rv);
354 return rv; /* Keep the source-specific status so the log shows which ADC stage failed. */
355 }
356 log_printf("entropy_personalisation_hash: ADC hashes succeeded\r\n");
357 /* Timer jitter. */
358 log_printf("entropy_personalisation_hash: starting jitter accumulation hash\r\n");
359 rv = ENTROPY_HASH_ERR_HASH_FAILED;
360 for (attempt = 0; attempt < ENTROPY_CREATION_RETRY_COUNT; ++attempt) {
361 rv = compute_jitter_accum_hashed_global();
362 if (rv == ENTROPY_HASH_SUCCESS) break;
363 log_printf("entropy_personalisation_hash: attempt %d jitter hashing failed (rv=%d)\r\n",
364 attempt + 1, rv);
365 }
366 if (rv != ENTROPY_HASH_SUCCESS) {
367 log_printf("entropy_personalisation_hash: jitter hashing ultimately failed.\r\n");
368 secure_zero(entropy_hashed, ENTROPY_HASH_LEN);
369 return JITTER_HASH_FAILED;
370 }
371 log_printf("entropy_personalisation_hash: jitter hashing succeeded\r\n");
372 /* SRAM source. */
373 log_printf("entropy_personalisation_hash: starting SRAM entropy hashing\r\n");
374 rv = ENTROPY_HASH_ERR_HASH_FAILED;
375 for (attempt = 0; attempt < ENTROPY_CREATION_RETRY_COUNT; ++attempt) {
376 rv = compute_sram_entropy_hashed_global();
377 if (rv == ENTROPY_HASH_SUCCESS) break;
378 log_printf("entropy_personalisation_hash: attempt %d sram entropy hashing failed (rv=%d)\r\n",
379 attempt + 1, rv);
380 }
381 if (rv != ENTROPY_HASH_SUCCESS) {
382 log_printf("entropy_personalisation_hash: sram entropy hashing ultimately failed.\r\n");
383 secure_zero(entropy_hashed, ENTROPY_HASH_LEN);
384 return SRAM_HASH_FAILED;
385 }
386 log_printf("entropy_personalisation_hash: SRAM entropy hashing succeeded\r\n");
387 /* Combine source hashes. */
388 log_printf("entropy_personalisation_hash: combining hashed components\r\n");
389 rv = ENTROPY_HASH_ERR_HASH_FAILED;
390 for (attempt = 0; attempt < ENTROPY_CREATION_RETRY_COUNT; ++attempt) {
391 rv = compute_combined_entropy_hashed_global(); /* writes entropy_hashed */
392 if (rv == ENTROPY_HASH_SUCCESS) break;
393 log_printf("entropy_personalisation_hash: attempt %d combined hashing failed (rv=%d)\r\n",
394 attempt + 1, rv);
395 }
396 if (rv != ENTROPY_HASH_SUCCESS) {
397 log_printf("entropy_personalisation_hash: combined hashing ultimately failed.\r\n");
398 secure_zero(entropy_hashed, ENTROPY_HASH_LEN);
399 return COMBINED_HASH_FAILED;
400 }
401 /* Reject empty final entropy. */
402 if (is_buffer_empty(entropy_hashed, ENTROPY_HASH_LEN)) {
403 log_printf("entropy_personalisation_hash: final entropy is all zeros (catastrophic)\r\n");
404 secure_zero(entropy_hashed, ENTROPY_HASH_LEN);
405 return ENTROPY_HASH_ERR_COMBINE_FAILED;
406 }
407 log_printf("entropy_personalisation_hash: combined entropy ready\r\n");
408 /* UID personalisation. */
409 log_printf("entropy_personalisation_hash: computing device UID hash (personalisation)\r\n");
410 rv = DEVICE_UID_HASH_ERR_HASH_FAILED;
411 for (attempt = 0; attempt < ENTROPY_CREATION_RETRY_COUNT; ++attempt) {
412 rv = compute_device_uid_hashed_global();
413 if (rv == DEVICE_UID_HASH_SUCCESS) break;
414 log_printf("entropy_personalisation_hash: attempt %d device UID hashing failed (rv=%d)\r\n",
415 attempt + 1, rv);
416 }
417 if (rv != DEVICE_UID_HASH_SUCCESS) {
418 log_printf("entropy_personalisation_hash: device UID hashing ultimately failed.\r\n");
419 secure_zero(entropy_hashed, ENTROPY_HASH_LEN);
420 return DEVICE_UID_HASH_FAILED;
421 }
422 log_printf("entropy_personalisation_hash: device UID hashed successfully\r\n");
423 /* Entropy path complete. */
424 log_printf("entropy_personalisation_hash: success\r\n");
425 return ENTROPY_CREATION_SUCCESS;
426}

428/* Final combine */
429
437int combine_all_hashed_entropy(uint8_t *out_hash, size_t out_len)
438{
439 cmox_hash_retval_t retval;
440 size_t out_len_local;
441 uint8_t combined[ENTROPY_HASH_LEN];
442 /* Check input buffers. */
443 if (out_hash == NULL) return ENTROPY_HASH_ERR_NULL_INPUT;
444 if (out_len != ENTROPY_HASH_LEN) return ENTROPY_HASH_ERR_BAD_LENGTH;
445 /* XOR the five source hashes. */
446 memset(combined, 0, ENTROPY_HASH_LEN);
447 for (size_t i = 0; i < ENTROPY_HASH_LEN; i++) {
448 combined[i] ^= buf_pc4_hashed[i];
449 combined[i] ^= buf_pc5_hashed[i];
450 combined[i] ^= buf_pb0_hashed[i];
451 combined[i] ^= jitter_accum_hashed[i];
452 combined[i] ^= sram_entropy_buffer_hashed[i];
453 }
454 /* Reject empty mixed value. */
455 if (is_buffer_empty(combined, ENTROPY_HASH_LEN)) {
456 memset(out_hash, 0, out_len);
457 return ENTROPY_HASH_ERR_COMBINE_FAILED;
458 }
459 /* Final SHA-256 pass. */
460 out_len_local = out_len;
461 retval = cmox_hash_compute(CMOX_SHA256_ALGO,
462 combined, ENTROPY_HASH_LEN,
463 out_hash, out_len,
464 &out_len_local);
465 /* Clear temporary mix. */
466 memset(combined, 0, ENTROPY_HASH_LEN);
467 if (retval != CMOX_HASH_SUCCESS || out_len_local != out_len) {
468 memset(out_hash, 0, out_len);
469 return ENTROPY_HASH_ERR_HASH_FAILED;
470 }
471 return ENTROPY_HASH_SUCCESS;
472}

601/* Nonce */
602
608int nonce_generation(void)
609{
610 cmox_hash_retval_t retval;
611 size_t out_len_local;
612 int attempt;
613 /* Full nonce hash before truncation. */
614 uint8_t hash_buf[32];
615 for (attempt = 0; attempt < NONCE_GEN_RETRY_COUNT; ++attempt) {
616 /* Advance message counter first. */
617 nonce_increment_message_counter();
618 /* Generate raw nonce. */
619 uint8_t *nonce_ = nonce_simple_generate();
620 if (nonce_ == NULL) {
621 log_printf("NONCE_GENERATION_INIT: attempt %d - nonce_simple_generate returned NULL\r\n",
622 attempt + 1);
623 continue;
624 }
625 /* Hash raw nonce. */
