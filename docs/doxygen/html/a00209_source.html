78/* Wait for I2C DMA completion or timeout. */
79static mpu9250_status_t mpu9250_wait_for_dma_completion(volatile i2c_dma_transfer_state_t *p_flag,
80 SemaphoreHandle_t sem,
81 uint32_t timeout_iter,
82 const char *context,
83 uint8_t is_tx)
84{
85 uint32_t t = 0U;
86 if (p_flag == NULL) {
87 MPU9250_LOG_PRINTF("[ERROR] %s: NULL DMA completion flag pointer\r\n", context);
88 app_fault_delay_ms(20U);
89 return MPU9250_STATUS_NULL_PTR;
90 }
91 /* Check completion flag first. */
92 if ((*p_flag) == I2C_DMA_TRANSFER_COMPLETE) {
93 *p_flag = I2C_DMA_TRANSFER_IDLE;
94 return MPU9250_STATUS_OK;
95 }
96 if ((*p_flag) == I2C_DMA_TRANSFER_ERROR) {
97 *p_flag = I2C_DMA_TRANSFER_IDLE;
98 MPU9250_LOG_PRINTF("[ERROR] %s: DMA transfer error\r\n", context);
99 app_fault_delay_ms(20U);
100 return (is_tx != 0U) ? MPU9250_STATUS_TX_TRANSFER_ERROR
101 : MPU9250_STATUS_RX_TRANSFER_ERROR;
102 }
103 /* RTOS path waits on DMA semaphore. */
104 if ((xTaskGetSchedulerState() == taskSCHEDULER_RUNNING) && (sem != NULL)) {
105 if (xSemaphoreTake(sem, pdMS_TO_TICKS(MPU9250_DMA_RTOS_TIMEOUT_MS)) == pdTRUE) {
106 if ((*p_flag) == I2C_DMA_TRANSFER_COMPLETE) {
107 *p_flag = I2C_DMA_TRANSFER_IDLE;
108 return MPU9250_STATUS_OK;
109 }
110 if ((*p_flag) == I2C_DMA_TRANSFER_ERROR) {
111 *p_flag = I2C_DMA_TRANSFER_IDLE;
112 MPU9250_LOG_PRINTF("[ERROR] %s: DMA transfer error\r\n", context);
113 app_fault_delay_ms(20U);
114 return (is_tx != 0U) ? MPU9250_STATUS_TX_TRANSFER_ERROR
115 : MPU9250_STATUS_RX_TRANSFER_ERROR;
116 }
117 }
118 } else {
119 /* Startup fallback before scheduler runs. */
120 while (((*p_flag) == I2C_DMA_TRANSFER_IDLE) && (t++ < timeout_iter)) {
121 cpu_sleep_until_irq();
122 }
123 if ((*p_flag) == I2C_DMA_TRANSFER_COMPLETE) {
124 *p_flag = I2C_DMA_TRANSFER_IDLE;
125 return MPU9250_STATUS_OK;
126 }
127 if ((*p_flag) == I2C_DMA_TRANSFER_ERROR) {
128 *p_flag = I2C_DMA_TRANSFER_IDLE;
129 MPU9250_LOG_PRINTF("[ERROR] %s: DMA transfer error\r\n", context);
130 app_fault_delay_ms(20U);
131 return (is_tx != 0U) ? MPU9250_STATUS_TX_TRANSFER_ERROR
132 : MPU9250_STATUS_RX_TRANSFER_ERROR;
133 }
134 }
135 *p_flag = I2C_DMA_TRANSFER_IDLE;
136 MPU9250_LOG_PRINTF("[ERROR] %s: DMA completion timeout\r\n", context);
137 app_fault_delay_ms(20U);
138 return (is_tx != 0U) ? MPU9250_STATUS_TX_TIMEOUT
139 : MPU9250_STATUS_RX_TIMEOUT;
140}

142/* Write one MPU9250 register and wait for DMA completion. */
143static mpu9250_status_t mpu9250_write_single_reg_blocking(uint8_t reg_addr,
144 uint8_t reg_val,
145 const char *context)
146{
147 i2c_dma_status_t dma_status;
148 i2c_data_buff[0] = reg_val;
149 mpu9250g_tx_cmplt = I2C_DMA_TRANSFER_IDLE;
150 /* Clear stale TX DMA token. */
151 mpu9250_clear_dma_semaphore_if_rtos(mpu9250_tx_sem);
152 dma_status = i2c_dma_write_reg(MPU9250_I2C_INSTANCE,
153 mpu9250_addr,
154 reg_addr,
155 (const uint8_t *)i2c_data_buff,
156 I2C_DATA_BUFF_LEN);
157 if (dma_status != I2C_DMA_OK) {
158 MPU9250_LOG_PRINTF("[ERROR] %s: i2c_dma_write_reg failed, status=%d\r\n",
159 context, (int)dma_status);
160 app_fault_delay_ms(20U);
161 return MPU9250_STATUS_I2C_WRITE_START_FAILED;
162 }
163 return mpu9250_wait_for_dma_completion(&mpu9250g_tx_cmplt,
164 mpu9250_tx_sem,
165 MPU9250_TX_TIMEOUT_ITER,
166 context,
167 1U);
168}

172 uint8_t *p_dst,

175{
176 i2c_dma_status_t dma_status;

742app_mpu9250_runtime_status_t app_mpu9250_runtime_process(void)
743{
744 enum { MAX_ATTEMPTS = 3 };
745 for (int attempt = 1; attempt <= MAX_ATTEMPTS; ++attempt) {
746 mpu9250_status_t mpu_rc;
747 mpu_rc = MPU_readProcessedData(&MPU9250);
748 if (mpu_rc == MPU9250_STATUS_OK) {
749 heartbeat_signal_mpu9250main();
750 system_fault_clear_active(SYSTEM_FAULT_OWNER_MPU9250);
751 return APP_MPU9250_RUNTIME_OK;
752 }
753 app_mpu9250_log_runtime_error("MPU_readProcessedData",
754 (int32_t)mpu_rc,
755 attempt,
756 MAX_ATTEMPTS);
757 if (attempt < MAX_ATTEMPTS) {
758 app_mpu9250_runtime_status_t recovery_status;
759 recovery_status = app_mpu9250_recover_and_reinit(attempt, MAX_ATTEMPTS);
760 if (recovery_status == APP_MPU9250_RUNTIME_OK) {
761 continue;
762 }
763 MPU9250_LOG_PRINTF("ACT: MPU9250 short backoff before retry\r\n");
764 app_fault_delay_ms(20U);
765 continue;
766 }
767 MPU9250_LOG_PRINTF("[ERROR] MPU_readProcessedData failed after %d attempts\r\n", attempt);
768 app_fault_delay_ms(20U);
769 return APP_MPU9250_RUNTIME_READ_FAIL;
770 }
771 MPU9250_LOG_PRINTF("[ERROR] app_mpu9250_runtime_process: unexpected fallthrough\r\n");
772 app_fault_delay_ms(20U);
773 return APP_MPU9250_RUNTIME_READ_FAIL;
774}

781mpu9250_status_t MPU_calcAttitude(MPU9250_t *pMPU9250)
782{
783 float accelPitch;
784 float accelRoll;
785 mpu9250_status_t status;
786 if (pMPU9250 == NULL) {
787 MPU9250_LOG_PRINTF("[ERROR] MPU_calcAttitude: NULL pointer\r\n");
788 app_fault_delay_ms(20U);
789 return MPU9250_STATUS_NULL_PTR;
790 }
791 status = MPU_readProcessedData(pMPU9250);
792 if (status != MPU9250_STATUS_OK) {
793 MPU9250_LOG_PRINTF("[ERROR] MPU_calcAttitude: failed to read processed sensor data, status=%d\r\n",
794 (int)status);
795 app_fault_delay_ms(20U);
796 return status;
797 }
798 accelPitch = atan2f(pMPU9250->sensorData.ay, pMPU9250->sensorData.az) * RAD2DEG;
799 accelRoll = atan2f(pMPU9250->sensorData.ax, pMPU9250->sensorData.az) * RAD2DEG;
800 pMPU9250->attitude.r =
801 pMPU9250->attitude.tau *
802 (pMPU9250->attitude.r - pMPU9250->sensorData.gy * pMPU9250->attitude.dt) +
803 (1.0f - pMPU9250->attitude.tau) * accelRoll;
804 pMPU9250->attitude.p =
805 pMPU9250->attitude.tau *
806 (pMPU9250->attitude.p - pMPU9250->sensorData.gx * pMPU9250->attitude.dt) +
807 (1.0f - pMPU9250->attitude.tau) * accelPitch;
808 pMPU9250->attitude.y += (pMPU9250->sensorData.gz * pMPU9250->attitude.dt);
809 return MPU9250_STATUS_OK;
810}
