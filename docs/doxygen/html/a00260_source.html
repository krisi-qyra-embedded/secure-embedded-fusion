20{
21 /* Build a 10 kHz timer tick, then wait for 30 counts. */
22 RCC->APB1ENR |= RCC_APB1ENR_TIM2EN;
23 (void)RCC->APB1ENR;
24 TIM2->PSC = (uint32_t)((TIM2_TIMCLK_HZ / TIM2_3MS_TICK_HZ) - 1UL);

26 TIM2->CNT = 0U;
27 TIM2->SR &= ~TIM_SR_UIF;
28 TIM2->CR1 |= TIM_CR1_CEN;

31 __asm__ volatile ("nop");

40void tim2_delay_ms(uint32_t ms)
41{
42 uint64_t base_presc;
43 uint32_t presc_reg;
44 uint64_t presc_plus1;
45 uint64_t tick_freq;
46 uint64_t ticks_needed;
47 const uint64_t ARR_MAX = 0xFFFFFFFFULL;
48 uint64_t remaining_ticks;
49 if (ms == 0U) {
50 return;
51 }
52 RCC->APB1ENR |= RCC_APB1ENR_TIM2EN;
53 (void)RCC->APB1ENR;
54 /* Start from a 1 ms timer base and keep PSC inside 16 bits. */
55 base_presc = (uint64_t)TIM2_TIMCLK_HZ / 1000ULL;
56 if (base_presc < 1ULL) {
57 base_presc = 1ULL;
58 }
59 if (base_presc <= 0xFFFFULL) {
60 presc_plus1 = base_presc;
61 presc_reg = (uint32_t)(presc_plus1 - 1ULL);
62 } else {
63 uint64_t factor = (base_presc + 0xFFFFULL - 1ULL) / 0xFFFFULL;
64 presc_plus1 = (base_presc + factor - 1ULL) / factor;
65 if (presc_plus1 < 1ULL) {
66 presc_plus1 = 1ULL;
67 }
68 if (presc_plus1 > 0xFFFFULL) {
69 presc_plus1 = 0xFFFFULL;
70 }
71 presc_reg = (uint32_t)(presc_plus1 - 1ULL);
72 }
73 /* Convert the requested delay to timer ticks. */
74 tick_freq = (uint64_t)TIM2_TIMCLK_HZ / presc_plus1;
75 ticks_needed = (tick_freq * (uint64_t)ms + 999ULL) / 1000ULL;
76 remaining_ticks = ticks_needed;
77 /* Split very long delays across the 32-bit ARR range. */
78 while (remaining_ticks > 0ULL) {
79 uint64_t this_chunk = (remaining_ticks <= ARR_MAX) ? remaining_ticks : ARR_MAX;
80 TIM2->PSC = presc_reg & 0xFFFFU;
81 TIM2->ARR = (uint32_t)(this_chunk - 1ULL);
82 TIM2->CNT = 0U;
83 TIM2->SR &= ~TIM_SR_UIF;
84 __DSB();
85 __ISB();
86 TIM2->CR1 |= TIM_CR1_CEN;
87 while ((TIM2->SR & TIM_SR_UIF) == 0U) {
88 __asm__ volatile ("nop");
89 }
90 TIM2->SR &= ~TIM_SR_UIF;
91 TIM2->CR1 &= ~TIM_CR1_CEN;
92 remaining_ticks -= this_chunk;
93 }
94}

162tim_status_t MX_TIM2_Init(void)
163{

166 __HAL_RCC_TIM2_CLK_ENABLE();
167 htim2.Instance = TIM2;
168 pclk1_hz = HAL_RCC_GetPCLK1Freq();
169 /* APB timers run at 2x PCLK when the APB prescaler is not 1. */
170 if ((RCC->CFGR & RCC_CFGR_PPRE1) == RCC_HCLK_DIV1)
171 {
172 tim_clk_hz = pclk1_hz;
173 }
174 else
175 {
176 tim_clk_hz = pclk1_hz * 2U;
177 }
178 /* 1 MHz timer base gives one count per microsecond. */
179 htim2.Init.Prescaler = (uint32_t)((tim_clk_hz / 1000000U) - 1U);
180 htim2.Init.CounterMode = TIM_COUNTERMODE_UP;
181 htim2.Init.Period = 100U - 1U;
182 htim2.Init.ClockDivision = TIM_CLOCKDIVISION_DIV1;
183 htim2.Init.AutoReloadPreload = TIM_AUTORELOAD_PRELOAD_DISABLE;
184 if (HAL_TIM_Base_Init(&htim2) != HAL_OK)
185 {
186 return TIM_STATUS_INIT_FAIL;
187 }
188 HAL_NVIC_SetPriority(TIM2_IRQn, 1U, 0U);
189 HAL_NVIC_EnableIRQ(TIM2_IRQn);
