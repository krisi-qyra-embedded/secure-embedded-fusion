154int collect_adc_samples_until_full(void)
155{
156 for (;;) {
157 if (!g_transfer_complete) {
158 continue;
159 }
160 /* Consume the DMA-complete flag atomically against the IRQ handler. */
161 __disable_irq();
162 g_transfer_complete = 0;
163 __enable_irq();
164 adc_get_latest_triplet(adc_buffer, SAMPLE_COUNT,
165 &pc4_val, &pc5_val, &pb0_val);
166 all_full = store_adc_byte_values(pc4_val, pc5_val, pb0_val);
167 if (all_full) {
168 return 0;
169 }
170 }
171 return -1;
172}
