31/* Calculate the 24-bit reload value for a 1 ms tick. */
32static int compute_reload(uint32_t sysclk_hz, uint32_t *out_reload)
33{
34 /* Reject invalid input before calculating the reload value. */
35 if (sysclk_hz == 0U || out_reload == NULL) return -1;
36 uint32_t reload = sysclk_hz / 1000U;
37 if (reload == 0U) return -1;
38 /* SysTick LOAD stores ticks-per-period minus one. */
39 reload -= 1U;
40 /* LOAD is limited to 24 bits on Cortex-M. */
41 if (reload > 0x00FFFFFFU) return -1;
42 *out_reload = reload;
43 return 0;
44}

51systick_status_t systick_init(uint32_t sysclk_hz, uint32_t priority)
52{
53 /* Do not silently reconfigure an active system timebase. */
54 if (systick_inited) {
55 systick_error_hook("systick_init: already initialized", SYSTICK_ERR_ALREADY_INITIALIZED);
56 return SYSTICK_ERR_ALREADY_INITIALIZED;
57 }
58 uint32_t reload;
59 if (compute_reload(sysclk_hz, &reload) != 0) {
60 systick_error_hook("systick_init: invalid sysclk or reload overflow", SYSTICK_ERR_RELOAD_OVERFLOW);
61 return SYSTICK_ERR_RELOAD_OVERFLOW;
62 }
63 /* Stop SysTick before touching LOAD/VAL. */
64 SysTick->CTRL = 0U;
65 /* Program one interrupt every millisecond. */
66 SysTick->LOAD = reload;
67 SysTick->VAL = 0U;
68 /* Complete register writes before enabling the timebase. */
69 __DSB(); __ISB();
70#if defined(NVIC_SetPriority) && (defined(SysTick_IRQn))
71 /* Use requested priority when valid, otherwise clamp it. */
72 if (priority <= (uint32_t)((1U << __NVIC_PRIO_BITS) - 1U)) {
73 NVIC_SetPriority(SysTick_IRQn, (uint32_t)priority);
74 } else {
75 NVIC_SetPriority(SysTick_IRQn, (uint32_t)((1U << __NVIC_PRIO_BITS) - 1U));
76 }
77#endif
78 /* Enable processor clock source, interrupt and counter. */
79 SysTick->CTRL = SysTick_CTRL_CLKSOURCE_Msk
80 | SysTick_CTRL_TICKINT_Msk
81 | SysTick_CTRL_ENABLE_Msk;
82 /* Read back LOAD as a lightweight configuration check. */
83 if ((SysTick->LOAD & 0x00FFFFFFU) != (reload & 0x00FFFFFFU)) {
84 systick_error_hook("systick_init: LOAD readback mismatch", SYSTICK_ERR_RELOAD_OVERFLOW);
85 }
86 /* Publish initialized state only after setup is complete. */
87 systick_inited = 1;
88 g_ms_ticks = 0U;
89 return SYSTICK_OK;
90}

114void systick_delay_ms(uint32_t ms)
115{
116 /* Zero delay is a fast no-op. */
117 if (ms == 0U) return;
118 if (!systick_inited) {
119 systick_error_hook("systick_delay_ms: not initialized", SYSTICK_ERR_NOT_INITIALIZED);
120 return;
121 }
122 /* Unsigned subtraction keeps the delay safe across tick wrap. */
123 uint32_t start = ms_get();
124 while ((ms_get() - start) < ms) {
125 /* Sleep between tick interrupts instead of busy-spinning. */
126 __DSB();
127 __ISB();
128 __WFI();
129 }
130}

136systick_status_t systick_selftest(uint32_t timeout_ms)
137{
138 /* Self-test requires an active SysTick timebase. */
139 if (!systick_inited) return SYSTICK_ERR_NOT_INITIALIZED;
140 /* One tick advance is enough to confirm the timebase is running. */
141 uint32_t t0 = ms_get();
142 uint32_t deadline = t0 + timeout_ms;

147 }
