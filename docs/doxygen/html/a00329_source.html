124static uint32_t esp32_gcm_crc32(
125 const uint8_t *data,
126 size_t length)
127{
128 uint32_t crc = UINT32_C(0xFFFFFFFF);
129
130 if ((data == NULL) && (length != 0U))
131 {
132 return 0U;
133 }
134
135 for (size_t i = 0U; i < length; ++i)
136 {
137 crc ^= data[i];
138
139 for (unsigned int bit = 0U; bit < 8U; ++bit)
140 {
141 const uint32_t mask =
142 (uint32_t)(0U - (crc & 1U));
143
144 crc =
145 (crc >> 1U) ^
146 (UINT32_C(0xEDB88320) & mask);
147 }
148 }
149
150 return ~crc;
151}

242static bool esp32_gcm_nvs_load_or_provision_rx_seed_once(void)
243{
244 esp32_gcm_rx_seed_record_t seed;
245 esp_err_t err;
246 size_t blob_size;
247
248 if (!s_gcm_nvs_open)
249 {
250 return false;
251 }
252
253 mbedtls_platform_zeroize(
254 &seed,
255 sizeof(seed));
256
257 blob_size =
258 sizeof(seed);
259
260 err =
261 nvs_get_blob(
262 s_gcm_nvs_handle,
263 ESP32_GCM_NVS_RX_SEED_KEY,
264 &seed,
265 &blob_size);
266
267 if (err == ESP_ERR_NVS_NOT_FOUND)
268 {
269 seed.magic =
270 ESP32_GCM_NVS_RX_SEED_MAGIC;
271
272 seed.version =
273 ESP32_GCM_NVS_RX_SEED_VERSION;
274
275 seed.session_id = 0U;
276 seed.counter_hi = 0U;
277 seed.counter_lo = 0U;
278
279 seed.crc32 =
280 esp32_gcm_crc32(
281 (const uint8_t *)&seed,
282 offsetof(
283 esp32_gcm_rx_seed_record_t,
284 crc32));
285
286 err =
287 nvs_set_blob(
288 s_gcm_nvs_handle,
289 ESP32_GCM_NVS_RX_SEED_KEY,
290 &seed,
291 sizeof(seed));
292
293 if (err == ESP_OK)
294 {
295 err =
296 nvs_commit(
297 s_gcm_nvs_handle);
298 }
299
300 if (err != ESP_OK)
301 {
302 ESP_LOGE(
303 TAG,
304 "One-time RX seed provisioning failed: %s",
305 esp_err_to_name(err));
306
307 mbedtls_platform_zeroize(
308 &seed,
309 sizeof(seed));
310
311 return false;
312 }
313
314 ESP_LOGI(
315 TAG,
316 "One-time NVS RX seed programmed: "
317 "session=00000000 counter=0 key=%s",
318 ESP32_GCM_NVS_RX_SEED_KEY);
319 }
320 else
321 {
322 if (err != ESP_OK)
323 {
324 ESP_LOGE(
325 TAG,
326 "nvs_get_blob(%s) failed: %s",
327 ESP32_GCM_NVS_RX_SEED_KEY,
328 esp_err_to_name(err));
329
330 mbedtls_platform_zeroize(
331 &seed,
332 sizeof(seed));
333
334 return false;
335 }
336
337 if ((blob_size != sizeof(seed)) ||
338 (!esp32_gcm_rx_seed_record_valid(
339 &seed)) ||
340 (seed.session_id != 0U) ||
341 (esp32_gcm_rx_seed_counter(
342 &seed) != UINT64_C(0)))
343 {
344 ESP_LOGE(
345 TAG,
346 "One-time RX seed is invalid or not {session=0,counter=0}; "
347 "failing closed");
348
349 mbedtls_platform_zeroize(
350 &seed,
351 sizeof(seed));
352
353 return false;
354 }
355
356 ESP_LOGI(
357 TAG,
358 "One-time NVS RX seed already provisioned: "
359 "session=00000000 counter=0 (no NVS rewrite)");
360 }
361
362 s_gcm_rx_session_id_ram = 0U;
363 s_gcm_rx_counter_ram = UINT64_C(0);
364 s_gcm_rx_ram_initialized = true;
365
366 mbedtls_platform_zeroize(
367 &seed,
368 sizeof(seed));
369
370 return true;
371}

603{

610 }

639 {
640 return false;
641 }
642
643 *session_id_out =
644 next.tx_session_id;
645
646 return true;
647}

770 session_id;
771
772 s_gcm_rx_counter_ram =
773 counter;

978 /* Persist the next TX session before resetting its packet counter. */
979 if (!esp32_gcm_nvs_reserve_tx_session(
980 &session_id))
981 {
982 return ESP32_AES_GCM_ERR_NVS;
983 }
984
985 mbedtls_platform_zeroize(
986 &ctx->nonce_state,
987 sizeof(ctx->nonce_state));
988
989 ctx->nonce_state.session_id =
990 session_id;
991
992 ctx->nonce_state.next_counter =
993 ESP32_GCM_COUNTER_START;
994
995 ctx->nonce_state.current_counter =
996 UINT64_C(0);
997
998 ctx->nonce_state.initialized =
999 true;
1000
1001 mbedtls_platform_zeroize(
1002 ctx->iv_gcm,
1003 sizeof(ctx->iv_gcm));
1004
1005 ESP_LOGI(
1006 TAG,
1007 "Reset-safe ESP32->STM32 GCM nonce session reserved: "
1008 "persistent_session=%08lX first_counter=%llu",
1009 (unsigned long)ctx->nonce_state.session_id,
1010 (unsigned long long)ctx->nonce_state.next_counter);
1011
1012 return ESP32_AES_GCM_SUCCESS;
1013}

1021esp32_aes_gcm_status_t esp32_gcm_nonce_allocate(
1022 esp32_aes_gcm_ctx_t *ctx)
1023{
1024 uint64_t counter;
1025
1026 if (ctx == NULL)
1027 {
1028 return ESP32_AES_GCM_ERR_BAD_PARAM;
1029 }
1030
1031 if (!ctx->initialized)
1032 {
1033 return ESP32_AES_GCM_ERR_NOT_INITIALIZED;
1034 }
1035
1036 if (!ctx->nonce_state.initialized)
1037 {
1038 return ESP32_AES_GCM_ERR_NO_SESSION;
1039 }
1040
1041 /* Reject the exhausted counter before it can wrap. */
1042 if (ctx->nonce_state.next_counter == UINT64_MAX)
1043 {
1044 return ESP32_AES_GCM_ERR_COUNTER_EXHAUSTED;
1045 }
1046
1047 /* Consume the counter before encryption so a failed send cannot roll it back. */
1048 counter =
1049 ctx->nonce_state.next_counter;
1050
1051 ctx->nonce_state.next_counter++;
1052 ctx->nonce_state.current_counter =
1053 counter;
1054
1055 store_be32(
1056 &ctx->iv_gcm[0],
1057 ctx->nonce_state.session_id);
1058
1059 store_be64(
1060 &ctx->iv_gcm[4],
1061 counter);
1062
1063 return ESP32_AES_GCM_SUCCESS;
1064}

1126 int rc;
1127
1128 if (output_len != NULL)
1129 {

1181 ciphertext_and_tag,
1182 required_size);

1185 tag,
1186 sizeof(tag));

1202 tag,

1204

1206 }

1210 MBEDTLS_GCM_ENCRYPT,

1212 ctx->iv_gcm,
1213 sizeof(ctx->iv_gcm),
1214 aad,
1215 aad_len,
1216 plaintext,
1217 ciphertext_and_tag,
1218 ESP32_AES_GCM_TAG_SIZE,
1219 tag);
1220
1221 if (rc != 0)
1222 {
1223 log_mbedtls_error(
1224 "mbedtls_gcm_crypt_and_tag(gcm_tx)",
1225 rc);
1226
1227 mbedtls_platform_zeroize(
1228 ciphertext_and_tag,
1229 required_size);
1230
1231 mbedtls_platform_zeroize(
1232 tag,
1233 sizeof(tag));
1234
1235 return ESP32_AES_GCM_ERR_ENCRYPT;
1236 }
1237
1238 memcpy(
1239 &ciphertext_and_tag[plaintext_len],
1240 tag,
1241 ESP32_AES_GCM_TAG_SIZE);
1242
1243 mbedtls_platform_zeroize(
1244 tag,
1245 sizeof(tag));
1246
1247 *output_len =
1248 required_size;
1249
1250 return ESP32_AES_GCM_SUCCESS;
1251}
