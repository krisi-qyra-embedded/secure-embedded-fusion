350esp32_ecdh_status_t esp32_ecdh_init(
351 esp32_ecdh_ctx_t *ctx)
352{
353 int rc;
354
355 if (ctx == NULL)
356 {
357 return ESP32_ECDH_ERR_BAD_PARAM;
358 }
359
360 /* The context must be zeroed before its first initialization. */
361 if (ctx->objects_initialized)
362 {
363 return ESP32_ECDH_ERR_ALREADY_INIT;
364 }
365
366 memset(
367 ctx,
368 0,
369 sizeof(*ctx));
370
371 mbedtls_entropy_init(
372 &ctx->entropy);
373
374 mbedtls_ctr_drbg_init(
375 &ctx->ctr_drbg);
376
377 mbedtls_ecp_group_init(
378 &ctx->group);
379
380 ecdh_party_init(
381 &ctx->sender);
382
383 ecdh_party_init(
384 &ctx->receiver);
385
386 ctx->objects_initialized = true;
387
388 if (mbedtls_ecdh_can_do(
389 MBEDTLS_ECP_DP_SECP256R1) != 1)
390 {
391 ESP_LOGE(
392 TAG,
393 "This Mbed TLS build does not support "
394 "ECDH with secp256r1");
395
396 esp32_ecdh_free(ctx);
397
398 return ESP32_ECDH_ERR_CURVE_UNSUPPORTED;
399 }
400
401 rc = mbedtls_ecp_group_load(
402 &ctx->group,
403 MBEDTLS_ECP_DP_SECP256R1);
404
405 if (rc != 0)
406 {
407 log_mbedtls_error(
408 "mbedtls_ecp_group_load",
409 rc);
410
411 esp32_ecdh_free(ctx);
412
413 return ESP32_ECDH_ERR_CURVE_LOAD;
414 }
415
416 /* Seed CTR-DRBG through the Mbed TLS entropy callback. */
417 rc = mbedtls_ctr_drbg_seed(
418 &ctx->ctr_drbg,
419 mbedtls_entropy_func,
420 &ctx->entropy,
421 s_drbg_personalization,
422 sizeof(s_drbg_personalization) - 1U);
423
424 if (rc != 0)
425 {
426 log_mbedtls_error(
427 "mbedtls_ctr_drbg_seed",
428 rc);
429
430 esp32_ecdh_free(ctx);
431
432 return ESP32_ECDH_ERR_DRBG_SEED;
433 }
434
435 ctx->drbg_seeded = true;
436
437 ESP_LOGI(
438 TAG,
439 "Mbed TLS entropy context initialized");
440
441 ESP_LOGI(
442 TAG,
443 "CTR-DRBG seeded from ESP32 platform entropy");
444
445 ESP_LOGI(
446 TAG,
447 "secp256r1 curve initialized");
448
449 return ESP32_ECDH_SUCCESS;
450}

459static esp32_ecdh_status_t export_and_validate_keypair(
460 esp32_ecdh_ctx_t *ctx,
461 esp32_ecdh_party_t *party)
462{
463 uint8_t public_key_uncompressed[
464 ESP32_ECDH_PUBLIC_KEY_UNCOMPRESSED_SIZE];
465
466 size_t uncompressed_len = 0U;
467 int rc;
468
469 if ((ctx == NULL) ||
470 (party == NULL))
471 {
472 return ESP32_ECDH_ERR_BAD_PARAM;
473 }
474
475 mbedtls_platform_zeroize(
476 public_key_uncompressed,
477 sizeof(public_key_uncompressed));
478
479 rc = mbedtls_ecp_check_privkey(
480 &ctx->group,
481 &party->private_key);
482
483 if (rc != 0)
484 {
485 log_mbedtls_error(
486 "mbedtls_ecp_check_privkey",
487 rc);
488
489 return ESP32_ECDH_ERR_PRIVATE_KEY_CHECK;
490 }
491
492 rc = mbedtls_ecp_check_pubkey(
493 &ctx->group,
494 &party->public_key);
495
496 if (rc != 0)
497 {
498 log_mbedtls_error(
499 "mbedtls_ecp_check_pubkey",
500 rc);
501
502 return ESP32_ECDH_ERR_PUBLIC_KEY_CHECK;
503 }
504
505 /* Export the private scalar as 32 big-endian bytes. */
506 rc = mbedtls_mpi_write_binary(
507 &party->private_key,
508 party->private_key_bytes,
509 sizeof(party->private_key_bytes));
510
511 if (rc != 0)
512 {
513 log_mbedtls_error(
514 "mbedtls_mpi_write_binary(private)",
515 rc);
516
517 ecdh_party_wipe_exported(
518 party);
519
520 return ESP32_ECDH_ERR_PRIVATE_EXPORT;
521 }
522
523 /* Export the public point as 0x04 || X[32] || Y[32]. */
524 rc = mbedtls_ecp_point_write_binary(
525 &ctx->group,
526 &party->public_key,
527 MBEDTLS_ECP_PF_UNCOMPRESSED,
528 &uncompressed_len,
529 public_key_uncompressed,
530 sizeof(public_key_uncompressed));
531
532 if (rc != 0)
533 {
534 log_mbedtls_error(
535 "mbedtls_ecp_point_write_binary",
536 rc);
537
538 mbedtls_platform_zeroize(
539 public_key_uncompressed,
540 sizeof(public_key_uncompressed));
541
542 ecdh_party_wipe_exported(
543 party);
544
545 return ESP32_ECDH_ERR_PUBLIC_EXPORT;
546 }
547
548 if ((uncompressed_len !=
549 ESP32_ECDH_PUBLIC_KEY_UNCOMPRESSED_SIZE) ||
550 (public_key_uncompressed[0] != 0x04U))
551 {
552 ESP_LOGE(
553 TAG,
554 "Unexpected public-key format: "
555 "length=%u prefix=0x%02X",
556 (unsigned int)uncompressed_len,
557 public_key_uncompressed[0]);
558
559 mbedtls_platform_zeroize(
560 public_key_uncompressed,
561 sizeof(public_key_uncompressed));
562
563 ecdh_party_wipe_exported(
564 party);
565
566 return ESP32_ECDH_ERR_PUBLIC_LENGTH;
567 }
568
569 /* Store the public point as raw X[32] || Y[32]. */
570 memcpy(
571 party->public_key_bytes,
572 &public_key_uncompressed[1],
573 ESP32_ECDH_PUBLIC_KEY_SIZE);
574
575 party->public_key_len =
576 ESP32_ECDH_PUBLIC_KEY_SIZE;
577
578 mbedtls_platform_zeroize(
579 public_key_uncompressed,
580 sizeof(public_key_uncompressed));
581
582 return ESP32_ECDH_SUCCESS;
583}

706 }

713 public_key,
714 ESP32_ECDH_PUBLIC_KEY_SIZE);
715

718 &party->public_key,

730 }
731

733 &ctx->group,
734 &party->private_key);
735
736 if (rc != 0)

738 log_mbedtls_error(
739 "mbedtls_ecp_check_privkey(provisioned)",
740 rc);
741
742 status = ESP32_ECDH_ERR_PRIVATE_KEY_CHECK;
743 goto cleanup;
744 }

748 &party->public_key);

754 rc);

756 status = ESP32_ECDH_ERR_PUBLIC_KEY_CHECK;
757 goto cleanup;
758 }
759
760 memcpy(
761 party->private_key_bytes,
762 private_key,
763 ESP32_ECDH_PRIVATE_KEY_SIZE);
764
765 memcpy(
766 party->public_key_bytes,
767 public_key,
768 ESP32_ECDH_PUBLIC_KEY_SIZE);
769
770 party->public_key_len =
771 ESP32_ECDH_PUBLIC_KEY_SIZE;
772
773cleanup:
774
775 mbedtls_platform_zeroize(
776 public_key_uncompressed,
777 sizeof(public_key_uncompressed));
778
779 if (status != ESP32_ECDH_SUCCESS)
780 {
781 ecdh_party_reset(party);
782 }
783
784 return status;
785}

1301static esp32_ecdh_status_t compute_shared_internal(
1302 esp32_ecdh_ctx_t *ctx,
1303 const mbedtls_mpi *own_private_key,
1304 const mbedtls_ecp_point *peer_public_key,
1305 uint8_t output_secret[
1306 ESP32_ECDH_SHARED_SECRET_SIZE],
1307 size_t *output_secret_len,
1308 esp32_ecdh_status_t compute_error,
1309 const char *side_name)
1310{
1311 mbedtls_mpi shared_secret_mpi;
1312 esp32_ecdh_status_t status = ESP32_ECDH_SUCCESS;
1313 size_t mpi_size;
1314 int rc;
1315
1316 if ((ctx == NULL) ||
1317 (own_private_key == NULL) ||
1318 (peer_public_key == NULL) ||
1319 (output_secret == NULL) ||
1320 (output_secret_len == NULL) ||
1321 (side_name == NULL))
1322 {
1323 return ESP32_ECDH_ERR_BAD_PARAM;
1324 }
1325
1326 mbedtls_mpi_init(&shared_secret_mpi);
1327
1328 /* Clear the fixed-width shared-secret output before computation. */
1329 mbedtls_platform_zeroize(
1330 output_secret,
1331 ESP32_ECDH_SHARED_SECRET_SIZE);
1332
1333 *output_secret_len = 0U;
1334
1335 rc = mbedtls_ecp_check_privkey(
1336 &ctx->group,
1337 own_private_key);
1338
1339 if (rc != 0)
1340 {
1341 log_mbedtls_error(
1342 "mbedtls_ecp_check_privkey",
1343 rc);
1344
1345 status = ESP32_ECDH_ERR_PRIVATE_KEY_CHECK;
1346 goto cleanup;
1347 }
1348
1349 rc = mbedtls_ecp_check_pubkey(
1350 &ctx->group,
1351 peer_public_key);
1352
1353 if (rc != 0)
1354 {
1355 log_mbedtls_error(
1356 "mbedtls_ecp_check_pubkey",
1357 rc);
1358
1359 status = ESP32_ECDH_ERR_PUBLIC_KEY_CHECK;
1360 goto cleanup;
1361 }
1362
1363 rc = mbedtls_ecdh_compute_shared(
1364 &ctx->group,
1365 &shared_secret_mpi,
1366 peer_public_key,
1367 own_private_key,
1368 mbedtls_ctr_drbg_random,
1369 &ctx->ctr_drbg);
1370
1371 if (rc != 0)
1372 {
1373 log_mbedtls_error(
1374 "mbedtls_ecdh_compute_shared",
1375 rc);
1376
1377 status = compute_error;
1378 goto cleanup;
1379 }
1380
1381 mpi_size = mbedtls_mpi_size(
1382 &shared_secret_mpi);
1383
1384 if ((mpi_size == 0U) ||
1385 (mpi_size > ESP32_ECDH_SHARED_SECRET_SIZE))
1386 {
1387 ESP_LOGE(
1388 TAG,
1389 "%s ECDH shared-secret size invalid: "
1390 "got=%u maximum=%u",
1391 side_name,
1392 (unsigned int)mpi_size,
1393 (unsigned int)ESP32_ECDH_SHARED_SECRET_SIZE);
1394
1395 status = ESP32_ECDH_ERR_SHARED_LENGTH;
1396 goto cleanup;
1397 }
1398
1399 /* Export the ECDH X coordinate as 32 big-endian bytes. */
1400 rc = mbedtls_mpi_write_binary(
1401 &shared_secret_mpi,
1402 output_secret,
1403 ESP32_ECDH_SHARED_SECRET_SIZE);
1404
1405 if (rc != 0)
1406 {
1407 log_mbedtls_error(
1408 "mbedtls_mpi_write_binary(shared)",
1409 rc);
1410
1411 status = ESP32_ECDH_ERR_SHARED_EXPORT;
1412 goto cleanup;
1413 }
1414
1415 if (buffer_is_all_zero(
1416 output_secret,
1417 ESP32_ECDH_SHARED_SECRET_SIZE))
1418 {
1419 ESP_LOGE(
1420 TAG,
1421 "%s ECDH shared secret is all zero",
1422 side_name);
1423
1424 status = ESP32_ECDH_ERR_SHARED_LENGTH;
1425 goto cleanup;
1426 }
1427
1428 *output_secret_len =
1429 ESP32_ECDH_SHARED_SECRET_SIZE;
1430
1431cleanup:
1432
1433 if (status != ESP32_ECDH_SUCCESS)
1434 {
1435 mbedtls_platform_zeroize(
1436 output_secret,
1437 ESP32_ECDH_SHARED_SECRET_SIZE);
1438
1439 *output_secret_len = 0U;
1440 }
1441
1442 mbedtls_mpi_free(
1443 &shared_secret_mpi);
1444
1445 return status;
1446}
