50static void sntp_time_sync_obtain_time(void)
51{
52 time_t now = 0;
53 struct tm time_info = {0};
54
55 time(&now);
56 localtime_r(&now, &time_info);
57
58 /* Treat a year before 2016 as an unset clock. */

61 sntp_time_sync_init_sntp();
62 /* Apply Central European time and daylight-saving rules. */
