118max31865_status_t Max31865_readRegisterN(Max31865_t *max31865,
119 uint8_t addr,
120 uint8_t *buffer,
121 uint8_t n)
122{
123 if ((max31865 == NULL) || (buffer == NULL) || (n == 0U)) {
124 return MAX31865_STATUS_INVALID_ARG;
125 }
126 const uint16_t total = (uint16_t)(1U + n);
127 if (total > MAX_STACK_TOTAL) {
128 return MAX31865_STATUS_SIZE_ERROR;
129 }
130 uint8_t txbuf[MAX_STACK_TOTAL];
131 uint8_t rxbuf[MAX_STACK_TOTAL];
132 addr &= 0x7FU;
133 txbuf[0] = addr;
134 for (uint16_t i = 1U; i < total; ++i) {
135 txbuf[i] = 0xFFU;
136 }
137 memset(rxbuf, 0, total);
138 /*
139 * Clear stale DMA semaphore tokens before starting the new full-duplex SPI transfer.
140 * RX -> DMA2_Stream0 -> max31865_spi_rx_sem
141 * TX -> DMA2_Stream1 -> max31865_spi_tx_sem
142 */
143 max31865_clear_dma_semaphore_if_rtos(max31865_spi_rx_sem);
144 max31865_clear_dma_semaphore_if_rtos(max31865_spi_tx_sem);
145 MPU_CS(CS_SELECT);
146 spi4_flush_rx();
147 spi4_check_and_clear_ovr();
148 if (spi4_dma_transmit_receive(txbuf, rxbuf, total) != SPI_DMA_STATUS_OK) {
149 MPU_CS(CS_DESELECT);
150 return MAX31865_STATUS_DMA_ERROR;
151 }
152 /*
153 * These wait functions take the binary semaphores internally:
154 * spi_dma_stream1_wait_complete() waits for max31865_spi_tx_sem
155 * spi_dma_stream0_wait_complete() waits for max31865_spi_rx_sem
156 */
157 if (spi_dma_stream1_wait_complete(TRANSFER_TIMEOUT_MS) != SPI_DMA_STATUS_OK) {
158 MPU_CS(CS_DESELECT);
159 return MAX31865_STATUS_TIMEOUT;
160 }
161 if (spi_dma_stream0_wait_complete(TRANSFER_TIMEOUT_MS) != SPI_DMA_STATUS_OK) {
162 MPU_CS(CS_DESELECT);
163 return MAX31865_STATUS_TIMEOUT;
164 }
165 if (spi4_wait_bsy_clear_ms(TRANSFER_TIMEOUT_MS) != SPI_DMA_STATUS_OK) {
166 MPU_CS(CS_DESELECT);
167 return MAX31865_STATUS_TIMEOUT;
168 }
169 for (uint16_t i = 0U; i < n; ++i) {
170 buffer[i] = rxbuf[1U + i];
171 }
172 MPU_CS(CS_DESELECT);
173 return MAX31865_STATUS_OK;
174}

384max31865_status_t Max31865_readRTD(Max31865_t *max31865, uint16_t *rtd_value)
385{
386 if ((max31865 == NULL) || (rtd_value == NULL)) {
387 return MAX31865_STATUS_INVALID_ARG;
388 }
389 max31865_status_t status = Max31865_clearFault(max31865);
390 if (status != MAX31865_STATUS_OK) {
391 MAX31865_LOG_PRINTF("WARN: readRTD clearFault r=%d\r\n", status);
392 app_fault_delay_ms(20U);
393 }
394 status = Max31865_enableBias(max31865, 1U);
395 if (status != MAX31865_STATUS_OK) {
396 MAX31865_LOG_PRINTF("WARN: readRTD enableBias r=%d\r\n", status);
397 app_fault_delay_ms(20U);
398 return status;
399 }
400 /*
401 * MAX31865 needs bias settling time before a one-shot conversion.
402 * Without this, transient raw=0x0000 / 0x7FFF samples can appear.
403 */
404 app_fault_delay_ms(MAX31865_BIAS_SETTLE_MS);
405 uint8_t cfg = 0U;
406 status = Max31865_readRegister8(max31865, MAX31856_CONFIG_REG, &cfg);
407 if (status != MAX31865_STATUS_OK) {
408 return status;
409 }
410 cfg |= MAX31856_CONFIG_1SHOT;
411 status = Max31865_writeRegister8(max31865, MAX31856_CONFIG_REG, cfg);
412 if (status != MAX31865_STATUS_OK) {
413 MAX31865_LOG_PRINTF("ERR: readRTD start 1shot failed r=%d\r\n", status);
414 app_fault_delay_ms(20U);
415 return status;
416 }
417 /*
418 * Critical fix:
419 * Do not read the RTD register immediately after starting one-shot.
420 * Wait for conversion to complete first.
421 */
422 app_fault_delay_ms(MAX31865_ONESHOT_CONVERSION_MS);
423 status = Max31865_readRegister16(max31865, MAX31856_RTDMSB_REG, rtd_value);
424 if (status != MAX31865_STATUS_OK) {
425 return status;
426 }
427 *rtd_value >>= 1U;
428 return MAX31865_STATUS_OK;
429}

492/* Print MAX31865 fault diagnosis. */
493static void Max31865_log_fault_diagnosis(uint16_t raw, uint8_t fault)
494{
495 const float rt_ohms = (((float)raw) / 32768.0f) * _MAX31865_RREF;
496 MAX31865_LOG_PRINTF("DBG: MAX31865 context: raw=0x%04X (%lu), approx_Rt=%.3f ohm, fault=0x%02X\r\n",
497 raw, (unsigned long)raw, rt_ohms, (unsigned)fault);
498 app_fault_delay_ms(20U);
499 /* Raw RTD code checks. */
500 if (raw == 0x7FFFU) {
501 MAX31865_LOG_PRINTF("DIAG: RTD code is FULL SCALE (32767). Datasheet meaning: saturated-high measurement.\r\n");
502 app_fault_delay_ms(20U);
503 } else if (raw == 0x0000U) {
504 MAX31865_LOG_PRINTF("DIAG: RTD code is NEAR ZERO (0). Datasheet meaning: saturated-low measurement.\r\n");
505 app_fault_delay_ms(20U);
506 }
507 /* No MAX31865 fault bits set. */
508 if (fault == 0U) {
509 if ((raw == 0x7FFFU) || (raw == 0x0000U)) {
510 MAX31865_LOG_PRINTF("DIAG: No MAX31865 fault bits are set. This makes a pure RTD threshold fault less likely.\r\n");
511 MAX31865_LOG_PRINTF("DIAG: Suspect SPI/CS/DMA/timing, bias/conversion timing, stale register read, transient power/noise, or intermittent wiring.\r\n");
512 app_fault_delay_ms(20U);
513 } else {
514 MAX31865_LOG_PRINTF("DIAG: No MAX31865 fault bits are set.\r\n");
515 app_fault_delay_ms(20U);
516 }
517 return;
518 }
519 /* D7: high threshold. */
520 if ((fault & MAX31865_FAULT_HIGHTHRESH) != 0U) {
521 MAX31865_LOG_PRINTF("DIAG: D7 HIGH-THRESHOLD fault: measured RTD resistance is above the programmed high threshold.\r\n");
522 MAX31865_LOG_PRINTF("DIAG: Possible causes: open RTD element, RTDIN+ shorted high, FORCE+ shorted high, or FORCE+ not connected to RTD.\r\n");
523 MAX31865_LOG_PRINTF("DIAG: Resulting data trend: FULL SCALE.\r\n");
524 app_fault_delay_ms(20U);
525 }
526 /* D6: low threshold. */
527 if ((fault & MAX31865_FAULT_LOWTHRESH) != 0U) {
528 MAX31865_LOG_PRINTF("DIAG: D6 LOW-THRESHOLD fault: measured RTD resistance is below the programmed low threshold.\r\n");
529 MAX31865_LOG_PRINTF("DIAG: Possible causes: RTDIN+ shorted to RTDIN-, RTDIN+ shorted low and not connected to RTD, or FORCE+ shorted low.\r\n");
530 MAX31865_LOG_PRINTF("DIAG: Resulting data trend: NEAR ZERO.\r\n");
531 app_fault_delay_ms(20U);
532 }
533 /* D5: REFIN low fault. */
534 if ((fault & MAX31865_FAULT_REFINLOW) != 0U) {
535 MAX31865_LOG_PRINTF("DIAG: D5 fault: VREFIN- > 0.85 x VBIAS.\r\n");
536 MAX31865_LOG_PRINTF("DIAG: Possible causes: open RTD element, FORCE+ shorted high and connected to RTD, FORCE+ unconnected, or FORCE+ shorted high and not connected to RTD.\r\n");
537 MAX31865_LOG_PRINTF("DIAG: Resulting data trend: FULL SCALE or INDETERMINATE.\r\n");
538 app_fault_delay_ms(20U);
539 }
540 /* D4: REFIN high fault. */
541 if ((fault & MAX31865_FAULT_REFINHIGH) != 0U) {
542 MAX31865_LOG_PRINTF("DIAG: D4 fault: VREFIN- < 0.85 x VBIAS.\r\n");
543 MAX31865_LOG_PRINTF("DIAG: Typical cause: FORCE- open / broken return path / low-side wiring issue.\r\n");
544 MAX31865_LOG_PRINTF("DIAG: Resulting data trend: may APPEAR VALID even though the measurement path is wrong.\r\n");
545 app_fault_delay_ms(20U);
546 }
547 /* D3: RTDIN low fault. */
548 if ((fault & MAX31865_FAULT_RTDINLOW) != 0U) {
549 MAX31865_LOG_PRINTF("DIAG: D3 fault: VRTDIN- < 0.85 x VBIAS.\r\n");
550 MAX31865_LOG_PRINTF("DIAG: Typical cause: FORCE- open or RTDIN- low-side wiring issue.\r\n");
551 MAX31865_LOG_PRINTF("DIAG: Resulting data trend: NEAR ZERO or deceptively VALID-looking.\r\n");
552 app_fault_delay_ms(20U);
553 }
554 /* D2: over/under-voltage. */
555 if ((fault & MAX31865_FAULT_OVUV) != 0U) {
556 MAX31865_LOG_PRINTF("DIAG: D2 over/under-voltage fault: a protected input went above VDD or below GND.\r\n");
557 MAX31865_LOG_PRINTF("DIAG: Resulting data trend: INDETERMINATE. Check supply, grounding, noise, and transients.\r\n");
558 app_fault_delay_ms(20U);
559 }
560 /* Final fault summary. */
561 if (raw == 0x7FFFU) {
562 if ((fault & (MAX31865_FAULT_HIGHTHRESH | MAX31865_FAULT_REFINLOW)) != 0U) {
563 MAX31865_LOG_PRINTF("DIAG: Holistic summary: this sample looks like an OPEN-CIRCUIT / VERY-HIGH-RESISTANCE style fault.\r\n");
564 app_fault_delay_ms(20U);
565 } else {
566 MAX31865_LOG_PRINTF("DIAG: Holistic summary: full-scale data observed; combine the fault bits above with wiring and power checks.\r\n");
567 app_fault_delay_ms(20U);
568 }
569 } else if (raw == 0x0000U) {
570 if ((fault & (MAX31865_FAULT_LOWTHRESH | MAX31865_FAULT_RTDINLOW)) != 0U) {
571 MAX31865_LOG_PRINTF("DIAG: Holistic summary: this sample looks like a SHORT / VERY-LOW-RESISTANCE style fault.\r\n");
572 app_fault_delay_ms(20U);
573 } else {
574 MAX31865_LOG_PRINTF("DIAG: Holistic summary: near-zero data observed; combine the fault bits above with wiring and power checks.\r\n");
575 app_fault_delay_ms(20U);
576 }
577 } else if ((fault & (MAX31865_FAULT_REFINHIGH | MAX31865_FAULT_RTDINLOW)) != 0U) {
578 MAX31865_LOG_PRINTF("DIAG: Holistic summary: data can look numerically plausible, but the analog return path is suspect. Do not trust the sample.\r\n");
579 app_fault_delay_ms(20U);
580 }
581}

646 attempt,
647 MAX_ATTEMPTS);
648 MAX31865_LOG_PRINTF("[WARN] Max31865_readTempC: readFault failed r=%d while diagnosing raw=0x%04X\r\n",
649 (int)fault_status,
650 raw);
651 app_fault_delay_ms(20U);
652 }
653 }
654 if ((status == MAX31865_STATUS_OK) && (raw != 0U) && (raw < 0x7FFFU)) {
655 float Z1;
656 float Z2;
657 float Z3;
658 float Z4;
659 float Rt;
660 float temp;
661 Rt = ((float)raw) / 32768.0f;
662 Rt *= _MAX31865_RREF;
663 Z1 = -RTD_A;
664 Z2 = RTD_A * RTD_A - (4.0f * RTD_B);
665 Z3 = (4.0f * RTD_B) / _MAX31865_RNOMINAL;
666 Z4 = 2.0f * RTD_B;
667 temp = Z2 + (Z3 * Rt);
668 if (temp >= 0.0f) {
669 temp = (sqrtf(temp) + Z1) / Z4;
670 } else {
671 Rt /= _MAX31865_RNOMINAL;
672 Rt *= 100.0f;
673 {
674 float rpoly = Rt;
675 temp = -242.02f;
676 temp += 2.2228f * rpoly;
677 rpoly *= Rt;
678 temp += 2.5859e-3f * rpoly;
679 rpoly *= Rt;
680 temp -= 4.8260e-6f * rpoly;
681 rpoly *= Rt;
682 temp -= 2.8183e-8f * rpoly;
683 rpoly *= Rt;
684 temp += 1.5243e-10f * rpoly;
685 }
686 }
687 if (fault_valid && (fault != 0U)) {
688 final_status = MAX31865_STATUS_FAULT;
689 app_max31865_log_runtime_error("MAX31865 fault register",
690 (int32_t)MAX31865_STATUS_FAULT,
691 attempt,
692 MAX_ATTEMPTS);
693 MAX31865_LOG_PRINTF("[ERROR] Max31865_readTempC: sample looked numerically usable, but MAX31865 reports fault=0x%02X (attempt %d/%d)\r\n",
694 (unsigned)fault,
695 attempt,
696 MAX_ATTEMPTS);
697 app_fault_delay_ms(20U);
698 Max31865_log_fault_diagnosis(raw, fault);
699 } else {
700 sample_failed = false;
701 *readTemp = temp;
702 /*
703 * Valid MAX31865 sample.
704 * Only successful conversion reaches here.
705 */
706 (void)telemetry_publish_max31865(temp);
707 if (system_fault_logging_active == 0U) {
708 MAX31865_LOG_PRINTF("[MAX31865] OK: TempC = %.4f C (raw=0x%04X, dec=%lu, approx_Rt=%.3f ohm)\r\n",
709 temp,
710 raw,
711 (unsigned long)raw,
712 (((float)raw) / 32768.0f) * _MAX31865_RREF);
713 }
714 heartbeat_signal_main();
715 system_fault_clear_active(SYSTEM_FAULT_OWNER_MAX31865);
716 return MAX31865_STATUS_OK;
717 }
718 }
719 if ((status == MAX31865_STATUS_OK) &&
720 (sample_failed != false) &&
721 ((raw == 0U) || (raw >= 0x7FFFU))) {
722 final_status = MAX31865_STATUS_INVALID_DATA;
723 app_max31865_log_local_runtime_error("MAX31865 invalid RTD raw sample",
724 (int32_t)MAX31865_STATUS_INVALID_DATA,
725 attempt,
726 MAX_ATTEMPTS);
727 MAX31865_LOG_PRINTF("[ERROR] Max31865_readTempC: invalid RTD value raw=0x%04X (%lu) (attempt %d/%d)\r\n",
728 raw,
729 (unsigned long)raw,
730 attempt,
731 MAX_ATTEMPTS);
732 app_fault_delay_ms(20U);
733 if (raw == 0x7FFFU) {
734 MAX31865_LOG_PRINTF("DIAG: Raw code hit FULL SCALE (32767). In the MAX31865 tables this corresponds to a saturated-high result.\r\n");
735 app_fault_delay_ms(20U);
736 } else if (raw == 0x0000U) {
737 MAX31865_LOG_PRINTF("DIAG: Raw code hit NEAR ZERO (0). In the MAX31865 tables this corresponds to a saturated-low result.\r\n");
738 app_fault_delay_ms(20U);
739 } else {
740 MAX31865_LOG_PRINTF("DIAG: Raw code is outside the accepted window for this routine, so the sample is treated as invalid.\r\n");
741 app_fault_delay_ms(20U);
742 }
743 if (fault_valid) {
744 Max31865_log_fault_diagnosis(raw, fault);
745 } else {
746 MAX31865_LOG_PRINTF("[WARN] Max31865_readTempC: fault register unavailable for this invalid sample; diagnosis limited to raw-code behavior.\r\n");
747 app_fault_delay_ms(20U);
748 }
749 }
750 if (attempt >= MAX_ATTEMPTS) {
751 system_fault_mark_in_progress(SYSTEM_FAULT_OWNER_MAX31865);
752 MAX31865_LOG_PRINTF("ACT: MAX31865 sample attempt %d/%d failed -> recovery budget exhausted, no further retry\r\n",
753 attempt,
754 MAX_ATTEMPTS);
755 app_fault_delay_ms(20U);
756 MAX31865_LOG_PRINTF("[FATAL] MAX31865 failed after retry %d/%d -> reporting fatal to task (last_status=%d)\r\n",
757 attempt,
758 MAX_ATTEMPTS,
759 (int)final_status);
760 app_fault_delay_ms(20U);
761 return final_status;
762 }
763 Max31865_log_recovery_action(attempt, MAX_ATTEMPTS);
764 if (spi4_recover() == SPI_DMA_STATUS_OK) {
765 heartbeat_signal_main();
766 MAX31865_LOG_PRINTF("INFO: Max31865_readTempC: spi4_recover() succeeded, doing full restore/init\r\n");
767 if (spi4_restore_and_init() == SPI_DMA_STATUS_OK) {
768 heartbeat_signal_main();
769 MAX31865_LOG_PRINTF("INFO: Max31865_readTempC: spi4_restore_and_init() succeeded, reinitializing device\r\n");
770 memset(max31865, 0, sizeof(*max31865));
771 if (Max31865_init(max31865, 3U, 50U) != MAX31865_STATUS_OK) {
772 system_fault_mark_in_progress(SYSTEM_FAULT_OWNER_MAX31865);
773 MAX31865_LOG_PRINTF("[FATAL] MAX31865 init failed after SPI recovery -> reporting fatal to task\r\n");
774 app_fault_delay_ms(20U);
775 return MAX31865_STATUS_INIT_FAILED;
776 }
777 heartbeat_signal_main();
778 MAX31865_LOG_PRINTF("ACT: MAX31865 recovery complete after attempt %d/%d, retrying temperature conversion\r\n",
779 attempt,
780 MAX_ATTEMPTS);
781 spi4_short_delay_ms(5U);
782 continue;
783 }
784 app_max31865_log_runtime_error("spi4_restore_and_init",
785 (int32_t)SPI_DMA_STATUS_RECOVERY_FAILED,
786 attempt,
787 MAX_ATTEMPTS);
788 MAX31865_LOG_PRINTF("[WARN] Max31865_readTempC: spi4_restore_and_init() failed (attempt %d/%d)\r\n",
789 attempt,
790 MAX_ATTEMPTS);
791 app_fault_delay_ms(20U);
792 } else {
793 app_max31865_log_runtime_error("spi4_recover",
794 (int32_t)SPI_DMA_STATUS_RECOVERY_FAILED,
795 attempt,
796 MAX_ATTEMPTS);
797 MAX31865_LOG_PRINTF("[WARN] Max31865_readTempC: spi4_recover() failed (attempt %d/%d)\r\n",
798 attempt,
799 MAX_ATTEMPTS);
800 app_fault_delay_ms(20U);
801 }
802 MAX31865_LOG_PRINTF("ACT: MAX31865 short backoff before retry %d/%d\r\n",
803 attempt + 1,
804 MAX_ATTEMPTS);
805 spi4_short_delay_ms(10U);
806 }
807 system_fault_mark_in_progress(SYSTEM_FAULT_OWNER_MAX31865);
808 MAX31865_LOG_PRINTF("[ERROR] Max31865_readTempC: unexpected fallthrough; returning INVALID_DATA\r\n");
809 app_fault_delay_ms(20U);
810 return MAX31865_STATUS_INVALID_DATA;
811}
