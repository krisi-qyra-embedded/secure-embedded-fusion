202uint32_t device_generate_fingerprint(uint8_t *out_buffer, uint32_t out_len)
203{
204 if (out_buffer == NULL || out_len < 16) {
205 return 0;
206 }
207 if (!uid_read) {
208 device_uid_read();
209 }
210 /* Fingerprint layout starts with the 12-byte UID. */
211 uint32_t offset = 0;
212 for (uint32_t i = 0; i < UID_SIZE_BYTES && offset < out_len; i++) {
213 out_buffer[offset++] = device_uid_buffer[i];
214 }
215 if (offset + 2 <= out_len) {
216 /* Append the factory Flash-size register when space is available. */
217 uint16_t flash_size = *(volatile uint16_t *)0x1FFF7A22;
218 out_buffer[offset++] = (flash_size >> 0) & 0xFF;
219 out_buffer[offset++] = (flash_size >> 8) & 0xFF;
220 }
221 if (offset + 4 <= out_len) {
222 /* Finish with device and revision identification from DBGMCU. */
223 uint32_t dbgmcu_idcode = DBGMCU->IDCODE;
224 out_buffer[offset++] = (dbgmcu_idcode >> 0) & 0xFF;
225 out_buffer[offset++] = (dbgmcu_idcode >> 8) & 0xFF;
226 out_buffer[offset++] = (dbgmcu_idcode >> 16) & 0xFF;
227 out_buffer[offset++] = (dbgmcu_idcode >> 24) & 0xFF;
228 }
229 return offset;
230}
