55jitter_status_t jitter_init(void)
56{
57 /* Stop TIM1 while the capture path is configured. */
58 TIM1->CR1 &= ~TIM_CR1_CEN;
59 /* Enable GPIOA, TIM1 and DMA2 clocks. */
60 RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;
61 RCC->APB2ENR |= RCC_APB2ENR_TIM1EN;
62 RCC->AHB1ENR |= RCC_AHB1ENR_DMA2EN;
63 jitter_barrier();
64 if (!(RCC->AHB1ENR & RCC_AHB1ENR_GPIOAEN)) return JITTER_ERR_GPIO_CLK;
65 if (!(RCC->APB2ENR & RCC_APB2ENR_TIM1EN)) return JITTER_ERR_TIM_CLK;
66 if (!(RCC->AHB1ENR & RCC_AHB1ENR_DMA2EN)) return JITTER_ERR_DMA_CLK;
67 /* PA10 is routed to TIM1_CH3 input capture. */
68 GPIOA->MODER &= ~(3U << (10 * 2));
69 GPIOA->MODER |= (2U << (10 * 2));
70 GPIOA->AFR[1] &= ~(0xFU << ((10 - 8) * 4));
71 GPIOA->AFR[1] |= (0x1U << ((10 - 8) * 4));
72 GPIOA->OSPEEDR |= (3U << (10 * 2));
73 GPIOA->PUPDR &= ~(3U << (10 * 2));
74 jitter_barrier();
75 /* Free-running 16-bit timer keeps the raw timing resolution. */
76 TIM1->PSC = 0U;
77 TIM1->ARR = 0xFFFFU;
78 TIM1->CNT = 0U;
79 TIM1->EGR = TIM_EGR_UG;
80 /* Map TI3 onto capture channel 3. */
81 TIM1->CCMR2 &= ~((3U << 0) | (3U << 2) | (0xFU << 4));
82 TIM1->CCMR2 |= (1U << 0);
83 TIM1->CCER &= ~(TIM_CCER_CC3P | TIM_CCER_CC3NP);
84 TIM1->CCER |= TIM_CCER_CC3E;
85 /* Each CC3 event generates a DMA request. */
86 TIM1->DIER |= TIM_DIER_CC3DE;
87 TIM1->DIER |= TIM_DIER_UIE;
88 jitter_wait_dma_disable();
89 jitter_clear_dma_flags();
90 /* DMA2 Stream6 reads CCR3 into the capture buffer. */
91 uint32_t cr = 0;
92 cr |= (6U << DMA_SxCR_CHSEL_Pos);
93 cr |= DMA_SxCR_MINC;
94 cr |= (2U << DMA_SxCR_PSIZE_Pos);
95 cr |= (2U << DMA_SxCR_MSIZE_Pos);
96 cr |= (2U << DMA_SxCR_PL_Pos);
97 cr |= DMA_SxCR_TCIE;
98 DMA2_Stream6->CR = cr;
99 DMA2_Stream6->PAR = (uint32_t)&TIM1->CCR3;
100 /* Return the stream to the base buffer with no active transfer. */
101 DMA2_Stream6->M0AR = (uint32_t)capture_buffer;
102 DMA2_Stream6->NDTR = 0U;
103 /* DMA completion IRQ runs above the TIM1 update IRQ. */
104 NVIC_SetPriority(DMA2_Stream6_IRQn, 2);
105 NVIC_ClearPendingIRQ(DMA2_Stream6_IRQn);
106 NVIC_EnableIRQ(DMA2_Stream6_IRQn);
107 NVIC_SetPriority(TIM1_UP_TIM10_IRQn, 3);
108 NVIC_ClearPendingIRQ(TIM1_UP_TIM10_IRQn);
109 NVIC_EnableIRQ(TIM1_UP_TIM10_IRQn);
110 /* Start the free-running capture timer after setup. */
111 TIM1->CR1 |= TIM_CR1_CEN;
112 TIM1->EGR = TIM_EGR_UG;
113 return JITTER_OK;
114}

120jitter_status_t jitter_prepare_dma_sample(uint32_t index)
121{
122 if (index >= JITTER_CAPTURE_SIZE)
123 {
124 return JITTER_ERR_INVALID_PARAM;
125 }
126 jitter_status_t st;
127 /* Re-arm each sample from a known disabled DMA state. */
128 st = jitter_wait_dma_disable();
129 if (st != JITTER_OK) return st;
130 jitter_clear_dma_flags();
131 DMA2_Stream6->M0AR = (uint32_t)&capture_buffer[index];
132 DMA2_Stream6->NDTR = 1U;
133 jitter_barrier();

148jitter_status_t jitter_collect_samples(void)
149{
150 uint32_t i;
151 uint32_t timeout;
152 /* Start a fresh capture batch. */
153 idx_jitter = 0;
154 memset((void*)jitter_differences, 0, sizeof(jitter_differences));
155 for (i = 0; i < JITTER_CAPTURE_SIZE; i++)
156 {
157 g_jitter_capture_complete = 0;
158 jitter_status_t st = jitter_prepare_dma_sample(i);
159 if (st != JITTER_OK) return st;
160 /* Vary the software delay from the live timer value and sample index. */
161 uint32_t seed = TIM1->CNT ^ (i * 0x45D9F3BU);
162 seed ^= (seed << 13);
163 seed ^= (seed >> 17);
164 seed ^= (seed << 5);
165 uint32_t delay = (seed & 0xFFU) + 1U;
166 for (volatile uint32_t d = 0; d < delay * 50U; d++) { __NOP(); }
167 /* Capture TIM1 after the variable delay. */
168 jitter_trigger_capture_event();
169 /* Wait for DMA transfer-complete from this sample only. */
170 timeout = JITTER_SAMPLE_TIMEOUT;
171 while (!g_jitter_capture_complete && timeout--)
172 {
173 __NOP();
174 }
175 if (timeout == 0U)
176 {
177 jitter_wait_dma_disable();
178 return JITTER_ERR_DMA_TRANSFER;
179 }
180 /* Leave the stream disabled before the next sample is armed. */
181 jitter_wait_dma_disable();
182 }
183 return jitter_compute_differences();
184}

189jitter_status_t jitter_compute_differences(void)
190{
191 if (JITTER_CAPTURE_SIZE < 2) return JITTER_ERR_NO_SAMPLES;
192 for (uint32_t i = 0; i < (JITTER_CAPTURE_SIZE - 1); i++)
193 {
194 /* Use adjacent captures so timer wrap can be handled explicitly. */
195 uint32_t t1 = capture_buffer[i] & 0xFFFFU;
196 uint32_t t2 = capture_buffer[i + 1] & 0xFFFFU;
197 uint32_t diff;
198 /* Account for the 16-bit timer wrapping between captures. */
199 if (t2 >= t1) diff = t2 - t1;
200 else diff = (0xFFFFU - t1) + t2 + 1U;
201 jitter_differences[i] = diff;
202 }
203 jitter_differences[JITTER_CAPTURE_SIZE - 1] = 0;
204 idx_jitter = JITTER_CAPTURE_SIZE - 1;
205 return JITTER_OK;
206}

225uint32_t jitter_assess_entropy(void)
226{
227 uint32_t zero = 0, one = 0, small = 0, large = 0;
228 for (uint32_t i = 0; i < JITTER_CAPTURE_SIZE - 1; i++)
229 {
230 uint32_t d = jitter_differences[i];
231 /* Ignore empty differences in the quick variability check. */
232 if (d == 0) continue;
233 if (d & 1U) one++; else zero++;
234 if (d < 10U) small++;
235 if (d > 1000U) large++;
236 }
237 uint32_t samples = zero + one;
238 if (samples == 0) return 0;
239 int32_t diff = (int32_t)zero - (int32_t)one;
240 if (diff < 0) diff = -diff;
241 /* Start from bit-balance, then penalize suspicious distributions. */
242 uint32_t score = 100U - (100U * (uint32_t)diff) / samples;
243 if (small > (samples / 2U)) score = (score > 20U) ? score - 20U : 0U;
244 if (large > (samples / 10U)) score = (score > 10U) ? score - 10U : 0U;
245 return score;
246}
