221 gpio_config_spi4_pins();
222#ifdef RCC_APB2ENR_SPI4EN

224#else

226#endif
227 __DSB();

230 SPI4->CR1 = 0U;
231 SPI4->CR2 = 0U;
232 SPI4->SR = 0U;
233 {
234 uint32_t cr1 = 0U;
235#ifdef SPI_CR1_BIDIMODE
236 cr1 = SPI4->CR1 & ~(SPI_CR1_BIDIMODE | SPI_CR1_BIDIOE);
237#else

239#endif
240#ifdef SPI_CR1_CRCEN

242#endif
243#ifdef SPI_CR1_DFF
244 cr1 &= ~SPI_CR1_DFF;
245#endif
246 cr1 &= ~SPI_CR1_LSBFIRST;
247 /* Master mode with software NSS and MAX31865 CPOL/CPHA settings. */
248 cr1 |= SPI_CR1_MSTR;
249 cr1 |= SPI_CR1_SSM;
250 cr1 |= SPI_CR1_SSI;
251#ifdef SPI_CR1_BIDIMODE
252 cr1 &= ~SPI_CR1_BIDIMODE;
253#endif
254 cr1 &= ~SPI_CR1_RXONLY;
255 cr1 &= ~SPI_CR1_CPOL;
256 cr1 |= SPI_CR1_CPHA;
257 cr1 &= ~SPI_CR1_BR;
258 /* BR code is selected from PCLK2 and the target SCK. */
259 cr1 |= (SPI4_BR_CODE << SPI_CR1_BR_Pos);
260 SPI4->CR1 = cr1;
261#ifdef SPI_CR2_FRF
262 SPI4->CR2 &= ~SPI_CR2_FRF;
263#endif
264#ifdef SPI_CR2_DS
265 SPI4->CR2 &= ~SPI_CR2_DS;
266 SPI4->CR2 |= (0x7U << SPI_CR2_DS_Pos);
267#endif
268 SPI4->CR2 &= ~(SPI_CR2_RXDMAEN | SPI_CR2_TXDMAEN);
269 /* Both SPI directions are serviced through DMA. */
270 SPI4->CR2 |= (SPI_CR2_RXDMAEN | SPI_CR2_TXDMAEN);
271 }
272 SPI4->CR1 |= SPI_CR1_SPE;
273#ifdef RCC_AHB1ENR_DMA2EN
274 RCC->AHB1ENR |= RCC_AHB1ENR_DMA2EN;
275#else
276 RCC->AHB1ENR |= (1U << 22);
277#endif
278 __DSB();
279 __ISB();
280 /* Reset RX/TX stream control before applying channel configuration. */
281 DMA2_Stream0->CR = 0U;
282 DMA2_Stream1->CR = 0U;

290 }
291 if (wait_dma_stream_disabled(&DMA2_Stream1->CR, &DMA2->LIFCR, DMA_SxCR_EN, DMA_LIFCR_ALL_FLAGS_STREAM1) != SPI_DMA_STATUS_OK) {
292 spi_dma_mark_fault_owner();
293 SPI_DMA_LOG_PRINTF("[ERROR][SPI_DMA][SPI4][MAX31865][INIT] DMA TX stream disable timeout | stream=DMA2_Stream1 channel=%u\r\n",
294 (unsigned)SPI4_DMA_CHANNEL);
295 spi4_log_diag_snapshot("init_stream1_disable_timeout");
296 app_fault_delay_ms(20U);
297 return SPI_DMA_STATUS_STREAM_DISABLE_TIMEOUT;
298 }
299 {
300 /* Stream0 is peripheral-to-memory for SPI4 RX. */
301 uint32_t s0cr = 0U;
302 s0cr |= (SPI4_DMA_CHANNEL << 25);
303 s0cr &= ~((3U << 6) | (1U << 7));
304 s0cr |= (1U << 10);
305 s0cr &= ~(1U << 9);
306 s0cr &= ~((3U << 11) | (3U << 13));
307 s0cr |= (2U << 16);
308 s0cr |= DMA_SxCR_TCIE;
309 DMA2_Stream0->CR = s0cr;
310 DMA2->LIFCR = DMA_LIFCR_ALL_FLAGS_STREAM0;
311 }
312 {
313 /* Stream1 is memory-to-peripheral for SPI4 TX. */
314 uint32_t s1cr = 0U;
315 s1cr |= (SPI4_DMA_CHANNEL << 25);
316 s1cr |= (1U << 6);
317 s1cr |= (1U << 10);
318 s1cr &= ~(1U << 9);
319 s1cr &= ~((3U << 11) | (3U << 13));
320 s1cr |= (2U << 16);
321 s1cr |= DMA_SxCR_TCIE;

323 DMA2->LIFCR = DMA_LIFCR_ALL_FLAGS_STREAM1;

327 const uint32_t s0cr_read = DMA2_Stream0->CR;
328 const uint32_t s1cr_read = DMA2_Stream1->CR;
329 const uint32_t dir1 = (s1cr_read >> 6) & 0x3U;

336 (unsigned long)ch0,

342 }
343 }

345 /* Priority 5 allows the DMA IRQs to use FreeRTOS semaphore APIs. */
346 NVIC_SetPriority(DMA2_Stream0_IRQn, 5);
347 NVIC_EnableIRQ(DMA2_Stream0_IRQn);
348 NVIC_ClearPendingIRQ(DMA2_Stream1_IRQn);

361}
362/* Configure SPI4 SCK, MISO and MOSI pins. */
363static void gpio_config_spi4_pins(void)
364{
365 const uint32_t pb13 = 13U;
366 const uint32_t pa11 = 11U;
367 const uint32_t pa1 = 1U;
368 GPIOB->MODER &= ~(3U << (pb13 * 2U));
369 GPIOB->MODER |= (2U << (pb13 * 2U));
370 GPIOB->AFR[1] = (GPIOB->AFR[1] & ~(0xFU << ((pb13 - 8U) * 4U))) | (SPI4_AF_SCK_MISO << ((pb13 - 8U) * 4U));
371 GPIOB->OSPEEDR = (GPIOB->OSPEEDR & ~(3U << (pb13 * 2U))) | (2U << (pb13 * 2U));
372 GPIOB->PUPDR &= ~(3U << (pb13 * 2U));
373 GPIOA->MODER &= ~(3U << (pa11 * 2U));
374 GPIOA->MODER |= (2U << (pa11 * 2U));
375 GPIOA->AFR[1] = (GPIOA->AFR[1] & ~(0xFU << ((pa11 - 8U) * 4U))) | (SPI4_AF_SCK_MISO << ((pa11 - 8U) * 4U));
376 GPIOA->OSPEEDR = (GPIOA->OSPEEDR & ~(3U << (pa11 * 2U))) | (2U << (pa11 * 2U));
377 GPIOA->PUPDR &= ~(3U << (pa11 * 2U));
378 GPIOA->MODER &= ~(3U << (pa1 * 2U));
379 GPIOA->MODER |= (2U << (pa1 * 2U));
380 GPIOA->AFR[0] = (GPIOA->AFR[0] & ~(0xFU << (pa1 * 4U))) | (SPI4_AF_MOSI_PA1 << (pa1 * 4U));
381 GPIOA->OSPEEDR = (GPIOA->OSPEEDR & ~(3U << (pa1 * 2U))) | (2U << (pa1 * 2U));
382 GPIOA->PUPDR &= ~(3U << (pa1 * 2U));
383 __DSB();
384 __ISB();
385}

588/* Wait for a DMA stream to stop, then force-disable if needed. */
589static spi_dma_status_t wait_dma_stream_disabled(volatile __IO uint32_t *stream_cr,
590 volatile __IO uint32_t *flag_reg,
591 uint32_t cr_en_mask,
592 uint32_t clear_mask)
593{
594 /* Give hardware a short normal-disable window first. */
595 uint32_t short_wait = SPI_DMA_SHORT_WAIT_COUNT;
596 while (((*stream_cr & cr_en_mask) != 0U) && (short_wait-- != 0U)) {
597 cpu_sleep_until_irq();
598 }
599 if (((*stream_cr & cr_en_mask) == 0U)) {
600 return SPI_DMA_STATUS_OK;
601 }
602 /* Force-clear EN and stale flags if the normal wait did not finish. */
603 *(volatile uint32_t *)stream_cr = (*stream_cr & ~cr_en_mask);
604 *flag_reg = clear_mask;
605 for (volatile uint32_t i = 0U; i < SPI_DMA_FORCE_DISABLE_SETTLE_LOOPS; ++i) {
606 __asm__ volatile("nop");
607 }
608 return (((*stream_cr & cr_en_mask) == 0U) ? SPI_DMA_STATUS_OK : SPI_DMA_STATUS_STREAM_DISABLE_TIMEOUT);
609}
610/* Wait for DMA completion using RTOS semaphore or startup polling. */
611static spi_dma_status_t spi_dma_wait_complete(volatile spi_dma_transfer_state_t *state,
612 SemaphoreHandle_t sem,
613 uint32_t timeout_ms)
614{
615 if (state == NULL) {
616 return SPI_DMA_STATUS_INVALID_ARG;
617 }
618 /* Runtime path blocks on the matching DMA completion semaphore. */
619 if ((sem != NULL) && (xTaskGetSchedulerState() == taskSCHEDULER_RUNNING)) {
620 if (xSemaphoreTake(sem, pdMS_TO_TICKS(timeout_ms)) != pdTRUE) {
621 spi4_log_diag_snapshot("dma_wait_sem_timeout");
622 return SPI_DMA_STATUS_TIMEOUT;
623 }
624 if (*state == SPI_DMA_TRANSFER_SUCCESS) {
625 return SPI_DMA_STATUS_OK;
626 }
627 spi4_log_diag_snapshot("dma_wait_sem_transfer_error");
628 return SPI_DMA_STATUS_TRANSFER_ERROR;
629 }
630 {
631 /* Startup fallback waits on the state flag with a bounded timeout. */
632 const uint32_t start_ms = ms_get();
633 while (*state == SPI_DMA_TRANSFER_IDLE) {
634#ifdef USE_IWDG
635 IWDG_Refresh();
636#endif
637 cpu_sleep_until_irq();
638 if ((ms_get() - start_ms) >= timeout_ms) {
639 spi4_log_diag_snapshot("dma_wait_poll_timeout");
640 return SPI_DMA_STATUS_TIMEOUT;
641 }
642 }
643 }
644 if (*state == SPI_DMA_TRANSFER_SUCCESS) {
645 return SPI_DMA_STATUS_OK;
646 }
647 spi4_log_diag_snapshot("dma_wait_poll_transfer_error");
648 return SPI_DMA_STATUS_TRANSFER_ERROR;
649}

776spi_dma_status_t spi4_recover(void)
777{
778 /* First try to recover without resetting the SPI peripheral. */
779 SPI_DMA_LOG_PRINTF("[INFO][SPI_DMA][SPI4][MAX31865][RECOVER] start\r\n");
780 DMA2_Stream0->CR &= ~DMA_SxCR_EN;
781 DMA2_Stream1->CR &= ~DMA_SxCR_EN;
782 __DSB();
783 __ISB();
784 dma_clear_all_flags();
785 spi4_short_delay_ms(2U);
786 spi4_flush_rx();
787 spi4_check_and_clear_ovr();
788 SPI4->CR1 &= ~SPI_CR1_SPE;
789 __DSB();
790 __ISB();
791 {
792 uint32_t timeout = 1000U;
793 /* Give SPI4 a bounded window to leave BSY after DMA is stopped. */
794 while (((SPI4->SR & SPI_SR_BSY) != 0U) && (timeout-- != 0U)) {
795 __asm__ volatile("nop");
796 }
797 }
798 /* Use APB reset only when the graceful stop did not fully settle. */
799 if (((DMA2_Stream0->CR & DMA_SxCR_EN) == 0U) &&
800 ((DMA2_Stream1->CR & DMA_SxCR_EN) == 0U) &&
801 ((SPI4->SR & SPI_SR_BSY) == 0U)) {
802 SPI_DMA_LOG_PRINTF("[INFO][SPI_DMA][SPI4][MAX31865][RECOVER] graceful disable succeeded\r\n");
803 } else {
804 spi_dma_mark_fault_owner_task();
805 SPI_DMA_LOG_PRINTF("[WARN][SPI_DMA][SPI4][MAX31865][RECOVER] graceful disable failed -> toggling SPI APB reset\r\n");
806 spi4_log_diag_snapshot("recover_graceful_disable_failed");
807 app_fault_delay_ms(20U);
808#ifdef RCC_APB2RSTR_SPI4RST
809 RCC->APB2RSTR |= RCC_APB2RSTR_SPI4RST;
810 spi4_short_delay_ms(1U);
811 RCC->APB2RSTR &= ~RCC_APB2RSTR_SPI4RST;
812#else
813 RCC->APB2RSTR |= (1U << 13);
814 spi4_short_delay_ms(1U);
815 RCC->APB2RSTR &= ~(1U << 13);
816#endif
817 __DSB();
818 __ISB();
819 dma_clear_all_flags();
820 DMA2_Stream0->CR = 0U;
821 DMA2_Stream1->CR = 0U;
822 spi4_short_delay_ms(2U);
823 }
824 dma_clear_all_flags();
825 DMA2_Stream0->CR &= ~DMA_SxCR_EN;
826 DMA2_Stream1->CR &= ~DMA_SxCR_EN;
827 __DSB();
828 __ISB();
829 spi4_flush_rx();
830 spi4_check_and_clear_ovr();
831 SPI4->CR1 |= SPI_CR1_SPE;
832 __DSB();
833 __ISB();
834 /* Restore device-level MAX31865 state after the bus is healthy. */
835 if (spi4_device_reinit() != 0) {
836 spi_dma_mark_fault_owner_task();
837 SPI_DMA_LOG_PRINTF("[ERROR][SPI_DMA][SPI4][MAX31865][RECOVER] spi4_device_reinit failed\r\n");
838 spi4_log_diag_snapshot("recover_device_reinit_failed");
839 app_fault_delay_ms(20U);
840 return SPI_DMA_STATUS_DEVICE_REINIT_FAILED;
841 }
842 /* Final self-test confirms the recovered path before returning. */
843 if (spi4_selftest() != 0) {
844 spi_dma_mark_fault_owner_task();
845 SPI_DMA_LOG_PRINTF("[ERROR][SPI_DMA][SPI4][MAX31865][RECOVER] spi4_selftest failed\r\n");
846 spi4_log_diag_snapshot("recover_selftest_failed");
847 app_fault_delay_ms(20U);
848 return SPI_DMA_STATUS_SELFTEST_FAILED;
849 }
850 SPI_DMA_LOG_PRINTF("[INFO][SPI_DMA][SPI4][MAX31865][RECOVER] lightweight restore OK\r\n");
851 return SPI_DMA_STATUS_OK;
852}
