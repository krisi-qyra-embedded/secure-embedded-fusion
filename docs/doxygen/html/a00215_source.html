20{

73/* Write one 1-Wire bit using blocking timing. */
74static void OneWire_WriteBitBlocking(OneWire_t *OW, uint8_t bit)
75{
76 OneWire_EnterCritical();
77 /* Write-1 uses a short low pulse, write-0 keeps the bus low longer. */
78 if (bit != 0U)
79 {
80 OneWire_Pin_Level(OW, 0U);
81 OneWire_Pin_Mode(OW, Output);
82 delay_us(6U);
83 OneWire_Pin_Mode(OW, Input);
84 delay_us(64U);
85 }
86 else
87 {
88 OneWire_Pin_Level(OW, 0U);
89 OneWire_Pin_Mode(OW, Output);
90 delay_us(60U);
91 OneWire_Pin_Mode(OW, Input);
92 delay_us(10U);
93 }
94 OneWire_ExitCritical();
95}

176OneWire_Status_t OneWire_Reset(OneWire_t *OW)
177{
178 uint8_t presence_sample;
179 if (OneWire_IsConfigValid(OW) == 0U)
180 {
181 return ONEWIRE_STATUS_BAD_PARAM;
182 }
183 OneWire_EnterCritical();
184 /* Reset pulse holds the bus low long enough for presence detection. */
185 OneWire_Pin_Level(OW, 0U);
186 OneWire_Pin_Mode(OW, Output);
187 delay_us(480U);
188 OneWire_Pin_Mode(OW, Input);
189 delay_us(70U);
190 /* A low level here means at least one device answered. */
191 presence_sample = OneWire_Pin_Read(OW);
192 delay_us(410U);
193 OneWire_ExitCritical();
194 return (presence_sample == 0U) ? ONEWIRE_STATUS_OK : ONEWIRE_STATUS_NO_DEVICE;
195}

202OneWire_Status_t OneWire_Search(OneWire_t *OW, uint8_t Cmd)
203{
204 uint8_t id_bit_number = 1U;
205 uint8_t last_zero = 0U;
206 uint8_t rom_byte_number = 0U;
207 uint8_t search_result = 0U;
208 uint8_t rom_byte_mask = 1U;
209 uint8_t id_bit;
210 uint8_t cmp_id_bit;
211 uint8_t search_direction;
212 OneWire_Status_t st;
213 if (OneWire_IsConfigValid(OW) == 0U)
214 {
215 return ONEWIRE_STATUS_BAD_PARAM;
216 }
217 if (OW->LastDeviceFlag != 0U)
218 {
219 return ONEWIRE_STATUS_NOT_FOUND;
220 }
221 /* Search starts from a fresh reset/presence cycle. */
222 st = OneWire_Reset(OW);
223 if (st != ONEWIRE_STATUS_OK)
224 {
225 OW->LastDiscrepancy = 0U;
226 OW->LastDeviceFlag = 0U;
227 OW->LastFamilyDiscrepancy = 0U;
228 return ONEWIRE_STATUS_NO_DEVICE;
229 }
230 st = OneWire_WriteByte(OW, Cmd);
231 if (st != ONEWIRE_STATUS_OK)
232 {
233 return st;
234 }
235 do
236 {
237 /* Read ROM bit and its complement to resolve each search branch. */
238 st = OneWire_ReadBit(OW, &id_bit);
239 if (st != ONEWIRE_STATUS_OK)
240 {
241 return st;
242 }
243 st = OneWire_ReadBit(OW, &cmp_id_bit);
244 if (st != ONEWIRE_STATUS_OK)
245 {
246 return st;
247 }
248 /* 1/1 means no device is participating in this search position. */
249 if ((id_bit == 1U) && (cmp_id_bit == 1U))
250 {
251 break;
252 }
253 else
254 {
255 if (id_bit != cmp_id_bit)
256 {
257 search_direction = id_bit;
258 }
259 else
260 {
261 if (id_bit_number < OW->LastDiscrepancy)
262 {
263 search_direction = ((OW->RomByte[rom_byte_number] & rom_byte_mask) != 0U) ? 1U : 0U;
264 }
265 else
266 {
267 search_direction = (id_bit_number == OW->LastDiscrepancy) ? 1U : 0U;
268 }
269 /* Remember the last zero branch so the next search can continue there. */
270 if (search_direction == 0U)
271 {
272 last_zero = id_bit_number;
273 if (last_zero < 9U)
274 {
275 OW->LastFamilyDiscrepancy = last_zero;
276 }
277 }
278 }
279 if (search_direction == 1U)
280 {
281 OW->RomByte[rom_byte_number] |= rom_byte_mask;
282 }
283 else
284 {
285 OW->RomByte[rom_byte_number] &= (uint8_t)(~rom_byte_mask);
286 }
287 /* Tell all devices which ROM branch remains active. */
288 OneWire_WriteBitBlocking(OW, search_direction);
289 id_bit_number++;
290 rom_byte_mask <<= 1U;
291 if (rom_byte_mask == 0U)
292 {
293 rom_byte_number++;
294 rom_byte_mask = 1U;
295 }
296 }
297 } while (rom_byte_number < 8U);
298 /* All 64 ROM bits were resolved. */
299 if (id_bit_number >= 65U)
300 {
301 OW->LastDiscrepancy = last_zero;
302 if (OW->LastDiscrepancy == 0U)
303 {
304 OW->LastDeviceFlag = 1U;
305 }
306 search_result = 1U;
307 }
308 if ((search_result == 0U) || (OW->RomByte[0] == 0U))
309 {
310 OW->LastDiscrepancy = 0U;
311 OW->LastDeviceFlag = 0U;
312 OW->LastFamilyDiscrepancy = 0U;
313 return ONEWIRE_STATUS_NOT_FOUND;
314 }
315 return ONEWIRE_STATUS_OK;
316}

411uint8_t OneWire_CRC8(const uint8_t *Addr, uint8_t Len)
412{
413 uint8_t crc = 0U;
414 uint8_t inbyte;
415 uint8_t i;
416 uint8_t mix;
417 while (Len-- != 0U)
418 {
419 inbyte = *Addr++;
420 for (i = 8U; i != 0U; i--)
421 {
422 /* Dallas/Maxim CRC uses polynomial 0x8C in reflected form. */
423 mix = (uint8_t)((crc ^ inbyte) & 0x01U);
424 crc >>= 1U;
425 crc ^= (mix != 0U) ? 0x8CU : 0x00U;
426 inbyte >>= 1U;
427 }
428 }
429 return crc;
