21DEV_StatusTypeDef flash_ex_erase(FLASH_EraseInitTypeDef *pt_erase_init, uint32_t *sect_err)
22{
23 DEV_StatusTypeDef status = DEV_ERROR;
24 uint32_t index = 0U;
25 /* Never start a new erase/program cycle while Flash is busy. */
26 status = flash_wait_for_last_operation(FLASH_TIMEOUT_VALUE);
27 if( status == DEV_OK)
28 {
29 /* 0xFFFFFFFF means no sector-specific erase error has been reported yet. */
30 *sect_err = 0xFFFFFFFFU;
31 if(pt_erase_init->TypeErase == FLASH_TYPEERASE_MASSERASE)
32 {
33 flash_mass_erase((uint8_t)pt_erase_init->VoltageRange);
34 status = flash_wait_for_last_operation(FLASH_TIMEOUT_VALUE);
35 FLASH->CR &=~(1U<<2);
36 }
37 else
38 {
39 for(index = pt_erase_init->Sector; index < (pt_erase_init->NbSectors + pt_erase_init->Sector); index++)
40 {
41 flash_sector_erase(index, (uint8_t)pt_erase_init->VoltageRange);
42 status = flash_wait_for_last_operation(FLASH_TIMEOUT_VALUE);
43 /* Clear the sector-erase selection after each operation. */
44 CLEAR_BIT(FLASH->CR, (FLASH_CR_SER | FLASH_CR_SNB));
45 if(status !=DEV_OK)
46 {
47 *sect_err = index;
48 break;
49 }
50 }
51 }
52 flush_caches();
53 }
54 return status;
55}

100DEV_StatusTypeDef flash_wait_for_last_operation(uint32_t timeout)
101{
102 uint32_t tickstart = 0U;
103 pFlash.ErrorCode = 0;
104 /* Bound the busy wait with the project timebase. */
105 tickstart = get_tick();
106 while((FLASH->SR & FLASH_SR_BSY) != RESET)
107 {
108 if(timeout != MAX_DELAY)
109 {
110 if((timeout == 0) || ((get_tick() - tickstart) > timeout))
111 {
112 return DEV_TIMEOUT;
113 }
114 }
115 }
116 if((FLASH->SR & (1U<<0)) != RESET)
117 {
118 FLASH->SR = (1U<<0);
119 }
120 return DEV_OK;
121}

147 }
148 else

303uint32_t flash_write_data(uint32_t start_sect_addr,uint32_t *data, uint16_t numberofwords)
304{
305 FLASH_EraseInitTypeDef EraseInitStruct;
306 uint32_t sect_err;
307 uint16_t write_count = 0;
308 flash_unlock();
309 /* Erase every sector touched by the requested word range. */
310 uint32_t start_sector = get_sector(start_sect_addr);
311 uint32_t end_sect_addr = start_sect_addr + numberofwords * 4;
312 uint32_t end_sector = get_sector(end_sect_addr);
313 EraseInitStruct.TypeErase = FLASH_TYPEERASE_SECTORS;
314 EraseInitStruct.VoltageRange = FLASH_VOLTAGE_RANGE_3;
315 EraseInitStruct.Sector = start_sector;
316 EraseInitStruct.NbSectors = (end_sector - start_sector) + 1;
317 if(flash_ex_erase(&EraseInitStruct, &sect_err) != DEV_OK )
318 {
319 return flash_get_error();
320 }
321 while(write_count < numberofwords )
322 {
323 if(flash_program(FLASH_TYPEPROGRAM_WORD, start_sect_addr,data[write_count]) == DEV_OK)
324 {
325 start_sect_addr += 4;
326 write_count++;
327 }
328 else
329 {
330 return flash_get_error();
331 }
332 }
333 return 0;
334}

352int flash_read_data(uint32_t start_sect_addr, uint32_t *rx_buff, uint16_t numberofwords)
353{
354 if (rx_buff == NULL) {
355 return FLASH_READ_ERR_NULL_PTR;
356 }
357 if (numberofwords == 0) {
358 return FLASH_READ_OK;
359 }
360 /* 32-bit reads require a word-aligned Flash address. */
361 if (start_sect_addr & 0x3U) {
362 return FLASH_READ_ERR_ALIGNMENT;
363 }
364 for (uint16_t i = 0; i < numberofwords; ++i) {
365 rx_buff[i] = *(__IO uint32_t *)(start_sect_addr + ((uint32_t)i * 4U));
366 }
367 return FLASH_READ_OK;
368}
