38int device_uid_read(void)
39{
40 /* Defensive: ensure buffer exists */
41 if (device_uid_buffer == NULL) {
42 log_printf("device_uid_read_safe: device_uid_buffer is NULL\n");
43 return DEVICE_UID_READ_NULL_BUFFER;
44 }
45
46 /* Read three 32-bit words from UID registers */
47 uint32_t uid0 = *(volatile uint32_t *)UID_WORD0_ADDR;
48 uint32_t uid1 = *(volatile uint32_t *)UID_WORD1_ADDR;
49 uint32_t uid2 = *(volatile uint32_t *)UID_WORD2_ADDR;
50
51 /* Defensive: reject all-zero UID (should never happen on real silicon) */
52 if ((uid0 == 0U) && (uid1 == 0U) && (uid2 == 0U)) {
53 log_printf("device_uid_read_safe: UID registers returned all zeros\n");
54 return DEVICE_UID_READ_INVALID_UID;
55 }
56
57 /* Store as bytes (little-endian) */
58 device_uid_buffer[0] = (uint8_t)((uid0 >> 0) & 0xFF);
59 device_uid_buffer[1] = (uint8_t)((uid0 >> 8) & 0xFF);
60 device_uid_buffer[2] = (uint8_t)((uid0 >> 16) & 0xFF);
61 device_uid_buffer[3] = (uint8_t)((uid0 >> 24) & 0xFF);
62
63 device_uid_buffer[4] = (uint8_t)((uid1 >> 0) & 0xFF);
64 device_uid_buffer[5] = (uint8_t)((uid1 >> 8) & 0xFF);
65 device_uid_buffer[6] = (uint8_t)((uid1 >> 16) & 0xFF);
66 device_uid_buffer[7] = (uint8_t)((uid1 >> 24) & 0xFF);
67
68 device_uid_buffer[8] = (uint8_t)((uid2 >> 0) & 0xFF);
69 device_uid_buffer[9] = (uint8_t)((uid2 >> 8) & 0xFF);
70 device_uid_buffer[10] = (uint8_t)((uid2 >> 16) & 0xFF);
71 device_uid_buffer[11] = (uint8_t)((uid2 >> 24) & 0xFF);
72
73 uid_read = 1;
74
75 return DEVICE_UID_READ_SUCCESS;
76}

325uint32_t device_generate_fingerprint(uint8_t *out_buffer,
326 uint32_t out_len)
327{
328 if (out_buffer == NULL || out_len < 16) {
329 return 0;
330 }
331
332 if (!uid_read) {
333 device_uid_read();
334 }
335
336 /* Start with UID (12 bytes) */
337 uint32_t offset = 0;
338
339 for (uint32_t i = 0;
340 i < UID_SIZE_BYTES && offset < out_len;
341 i++) {
342
343 out_buffer[offset++] = device_uid_buffer[i];
344 }
345
346 /* Add Flash size (F_SIZE register at 0x1FFF7A22) */
347 if (offset + 2 <= out_len) {
348
349 uint16_t flash_size =
350 *(volatile uint16_t *)0x1FFF7A22;
351
352 out_buffer[offset++] =
353 (flash_size >> 0) & 0xFF;
354
355 out_buffer[offset++] =
356 (flash_size >> 8) & 0xFF;
357 }
358
359 /* Add Device ID and Revision ID from DBGMCU */
360 if (offset + 4 <= out_len) {
361
362 uint32_t dbgmcu_idcode = DBGMCU->IDCODE;
363
364 out_buffer[offset++] =
365 (dbgmcu_idcode >> 0) & 0xFF; /* DEV_ID low */
366
367 out_buffer[offset++] =
368 (dbgmcu_idcode >> 8) & 0xFF; /* DEV_ID high */
369
370 out_buffer[offset++] =
371 (dbgmcu_idcode >> 16) & 0xFF; /* REV_ID low */
372
373 out_buffer[offset++] =
374 (dbgmcu_idcode >> 24) & 0xFF; /* REV_ID high */
375 }
376
377 return offset;
378}
