95int8_t bme_GetGasReference(void)
96{
97 const uint8_t TARGET_VALID_READS = 3U;
98 const uint8_t MAX_ATTEMPTS_TOTAL = 10U;
99 const uint8_t MAX_READ_RETRIES = 2U;
100 const uint32_t MAX_RUNTIME_MS = 1500U;
101 int8_t rslt;
102 int8_t final_status = 0;
103 struct bme680_field_data data;
104 uint64_t acc = 0ULL;
105 uint8_t valid_count = 0U;
106 uint16_t attempts = 0U;
107 uint32_t start_ms = ms_get();
108 uint32_t last_valid_gas = 0U;
109 /* Mark calibration active so supervision can tolerate the longer path. */
110 bme680_gas_reference_active = 1U;
111 heartbeat_signal_bme680main();
112 rslt = null_ptr_check(&bme.gas_sensor);
113 if (rslt != BME680_OK) {
114 BME680_LOG_PRINTF("[ERROR] bme_GetGasReference: null_ptr_check failed err=%d\r\n",
115 (int)rslt);
116 tim2_delay_ms(20U);
117 final_status = -1;
118 goto gas_ref_exit;
119 }
120 /* Collect a few valid gas samples within a bounded attempt/time budget. */
121 while ((valid_count < TARGET_VALID_READS) &&
122 (attempts < MAX_ATTEMPTS_TOTAL)) {
123 uint32_t elapsed_ms = ms_get() - start_ms;
124 if (elapsed_ms > MAX_RUNTIME_MS) {
125 break;
126 }
127 attempts++;
128 heartbeat_signal_bme680main();
129 /* Start a fresh forced-mode measurement before each reference read. */
130 if (bme.gas_sensor.power_mode == BME680_FORCED_MODE) {
131 rslt = bme680_set_sensor_mode(&bme.gas_sensor);
132 if (rslt != BME680_OK) {
133 BME680_LOG_PRINTF("[WARN] bme_GetGasReference: set_sensor_mode failed err=%d attempt=%u/%u\r\n",
134 (int)rslt,
135 (unsigned)attempts,
136 (unsigned)MAX_ATTEMPTS_TOTAL);
137 tim2_delay_ms(20U);
138 continue;
139 }
140 {
141 /* Wait for the measurement profile, capped to keep calibration bounded. */
142 uint32_t wait_ms = (bme.meas_period > 0U) ? (uint32_t)bme.meas_period : 100U;
143 if (wait_ms > 250U) {
144 wait_ms = 250U;
145 }
146 heartbeat_signal_bme680main();
147 if (xTaskGetSchedulerState() == taskSCHEDULER_RUNNING) {
148 vTaskDelay(pdMS_TO_TICKS(wait_ms));
149 } else {
150 bme.gas_sensor.delay_ms(wait_ms);
151 }
152 }
153 }
154 memset(&data, 0, sizeof(data));
155 /* Retry only the sensor-data read on transient failures. */
156 for (uint8_t retry = 0U; retry < MAX_READ_RETRIES; ++retry) {
157 heartbeat_signal_bme680main();
158 rslt = bme680_get_sensor_data(&data, &bme.gas_sensor);
159 if (rslt == BME680_OK) {
160 heartbeat_signal_bme680();
161 break;
162 }
163 BME680_LOG_PRINTF("[WARN] bme_GetGasReference: transient read err=%d retry=%u/%u\r\n",
164 (int)rslt,
165 (unsigned)(retry + 1U),
166 (unsigned)MAX_READ_RETRIES);
167 tim2_delay_ms(20U);
168 if (xTaskGetSchedulerState() == taskSCHEDULER_RUNNING) {
169 vTaskDelay(pdMS_TO_TICKS(2U + ((uint32_t)retry * 5U)));
170 } else {
171 bme.gas_sensor.delay_ms(2U + ((uint32_t)retry * 5U));
172 }
173 }
174 if (rslt != BME680_OK) {
175 BME680_LOG_PRINTF("[ERROR] bme_GetGasReference: read failed permanently err=%d\r\n",
176 (int)rslt);
177 tim2_delay_ms(20U);
178 final_status = -3;
179 goto gas_ref_exit;
180 }
181 /* Only gas-valid samples contribute to the reference. */
182 if ((data.status & BME680_GASM_VALID_MSK) != 0U) {
183 last_valid_gas = (uint32_t)data.gas_resistance;
184 bme.gas_resistance = last_valid_gas;
185 acc += (uint64_t)last_valid_gas;
186 valid_count++;
187 }
188 heartbeat_signal_bme680main();
189 }
190 heartbeat_signal_bme680main();
191 if ((valid_count == 0U) || (last_valid_gas == 0U)) {
192 BME680_LOG_PRINTF("[ERROR] bme_GetGasReference: no valid gas readings attempts=%u\r\n",
193 (unsigned)attempts);
194 tim2_delay_ms(20U);
195 final_status = -2;
196 goto gas_ref_exit;
197 }
198 {
199 /* Keep the average for diagnostics, but use the latest valid sample as reference. */
200 uint32_t avg = (uint32_t)(acc / (uint64_t)valid_count);
201 /* Latest valid gas value gives runtime IAQ a closer starting point. */
202 bme.gas_reference = last_valid_gas;
203 BME680_LOG_PRINTF("[INFO] bme_GetGasReference: valid=%u attempts=%u gas_ref=%lu avg=%lu\r\n",
204 (unsigned)valid_count,
205 (unsigned)attempts,
206 (unsigned long)bme.gas_reference,
207 (unsigned long)avg);
208 tim2_delay_ms(20U);
209 }
210 /* Allow a short startup window to pull an overly-high reference down. */
211 bme680_gas_ref_startup_regulation_left =
212 BME680_STARTUP_GAS_REF_REGULATION_SAMPLES;
213 heartbeat_signal_bme680main();
214 heartbeat_signal_bme680();
215 final_status = 0;
216gas_ref_exit:
217 bme680_gas_reference_active = 0U;
218 return final_status;
219}

224int16_t bme_GetGasScore(void)
225{
226 uint32_t gas_ref;
227 uint32_t gas_res;
228 uint32_t gas_score;
229 /* Work with local copies of the current gas state. */
230 gas_ref = (uint32_t)bme.gas_reference;
231 gas_res = (uint32_t)bme.gas_resistance;
232 if (gas_res == 0U) {
233 BME680_LOG_PRINTF("[ERROR] bme_GetGasScore: gas_resistance is zero\r\n");
234 tim2_delay_ms(20U);
235 return -1;
236 }
237 /* First valid runtime sample can seed an empty reference. */
238 if (gas_ref == 0U) {
239 bme.gas_reference = gas_res;
240 gas_ref = gas_res;
241 }
242 /* During startup only, follow lower valid gas values. */
243 if (bme680_gas_ref_startup_regulation_left > 0U) {
244 if (gas_res < gas_ref) {
245 bme.gas_reference = gas_res;
246 gas_ref = gas_res;
247 }
248 bme680_gas_ref_startup_regulation_left--;
249 }
250 if (gas_ref == 0U) {
251 BME680_LOG_PRINTF("[ERROR] bme_GetGasScore: gas_reference is zero after regulation\r\n");
252 tim2_delay_ms(20U);
253 return -1;
254 }
255 /* Equal or better gas resistance gets the full gas score. */
256 if (gas_res >= gas_ref) {
257 gas_score = 75U;
258 } else {
259 gas_score = (uint32_t)(((uint64_t)gas_res * 75ULL) / (uint64_t)gas_ref);
260 }
261 if (gas_score > 75U) {
262 gas_score = 75U;
263 }
264 return (int16_t)gas_score;
265}

270int16_t bme_GetHumidityScore(void)
271{
272 const float OPT_LOW = 38.0f;
273 const float OPT_HIGH = 42.0f;
274 const float MAX_SCORE_F = 25.0f;
275 const float EPS = 1e-6f;
276 /* Use current humidity against the configured reference. */
277 float hum = (float) bme.humidity;
278 float hum_ref = (float) bme.hum_reference;
279 /* Reject a reference that would break the score calculation. */
280 if (!isfinite(hum_ref) || hum_ref <= 0.0f || hum_ref >= 100.0f) {
281 BME680_LOG_PRINTF("[ERROR] bme_GetHumidityScore: invalid hum_reference=%.3f\r\n", hum_ref);
282 tim2_delay_ms(20);
283 return -1;
284 }
285 if (!isfinite(hum) || hum < 0.0f) {
286 BME680_LOG_PRINTF("[ERROR] bme_GetHumidityScore: invalid humidity value=%.3f\r\n", hum);
287 tim2_delay_ms(20);
288 return -1;
289 }
290 float score_f = 0.0f;
291 /* The 38-42% zone receives the full humidity score. */
292 if (hum >= OPT_LOW && hum <= OPT_HIGH) {
293 score_f = MAX_SCORE_F;
294 } else if (hum < OPT_LOW) {
295 score_f = (MAX_SCORE_F * hum) / hum_ref;
296 } else {
297 /* Above the optimal zone, scale against the remaining humidity range. */
298 float denom = 100.0f - hum_ref;
299 if (denom <= EPS) {
300 BME680_LOG_PRINTF("[ERROR] bme_GetHumidityScore: denom too small (100 - hum_ref = %.6f)\r\n", denom);
301 tim2_delay_ms(20);
302 return -1;
303 }
304 score_f = ((-0.25f / denom) * hum + 0.416666f) * 100.0f;
305 }
306 /* Reject numeric failure before clamping and conversion. */
307 if (!isfinite(score_f)) {
308 BME680_LOG_PRINTF("[ERROR] bme_GetHumidityScore: numeric failure (score=%f)\r\n", score_f);
309 tim2_delay_ms(20);
310 return -1;
311 }
312 /* Keep the humidity contribution inside 0..25. */
313 if (score_f < 0.0f) score_f = 0.0f;
314 if (score_f > MAX_SCORE_F) score_f = MAX_SCORE_F;
315 int16_t score = (int16_t)(score_f + 0.5f);
316 return score;
317}

345 }

348 iaq_label =

350 }
351 else if (iaq_score >= 51)

353 iaq_label = "Moderate";
354 }
