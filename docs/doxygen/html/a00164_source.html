328/* Waits for the I2C_DMA transfer to complete*/
329static int bme680_wait_dma_complete_or_timeout(volatile i2c_dma_transfer_state_t *state,
330 SemaphoreHandle_t sem)
331{
332 uint32_t t = 0U;
333
334 if (state == NULL) {
335 return 0;
336 }
337
338 /* Blocking/non-DMA paths may complete before an IRQ semaphore is involved. */
339 if (*state == I2C_DMA_TRANSFER_COMPLETE) {
340 return 1;
341 }
342
343 if (*state == I2C_DMA_TRANSFER_ERROR) {
344 return 0;
345 }
346
347 /* FreeRTOS path: blocks the current task until DMA IRQ gives the semaphore. */
348 if ((xTaskGetSchedulerState() == taskSCHEDULER_RUNNING) && (sem != NULL)) {
349 if (xSemaphoreTake(sem, pdMS_TO_TICKS(BME680_USER_I2C_RTOS_TIMEOUT_MS)) == pdTRUE) {
350 return (*state == I2C_DMA_TRANSFER_COMPLETE) ? 1 : 0;
351 }
352
353 return 0;
354 }
355
356 /* Bare-metal/startup fallback: used before scheduler start during sensor init. */
357 while ((*state == I2C_DMA_TRANSFER_IDLE) &&
358 (t++ < BME680_USER_I2C_TIMEOUT_ITER)) {
359 cpu_sleep_until_irq();
360 }
361
362 return (*state == I2C_DMA_TRANSFER_COMPLETE) ? 1 : 0;
363}

367 * BME680_USER_I2C_E_INVALID_ARG (-1)
368 * BME680_USER_I2C_E_START_FAIL (-2)

371int8_t user_i2c_read(void *handle,
372 uint8_t dev_id,
373 uint8_t reg_addr,
374 uint8_t *reg_data,
375 uint16_t len)
376{
377 I2C_TypeDef *I2Cx;
378 const uint8_t slave7 = to7bit((uint32_t)dev_id);
379 i2c_dma_status_t st;
380
381 if ((handle == NULL) || (reg_data == NULL) || (len == 0U)) {
382 return (int8_t)BME680_USER_I2C_E_INVALID_ARG;
383 }
384
385 I2Cx = (I2C_TypeDef *)handle;
386 if (I2Cx != I2C1) {
387 return (int8_t)BME680_USER_I2C_E_INVALID_ARG;
388 }
389
390 bme680g_rx_cmplt = I2C_DMA_TRANSFER_IDLE;
391
392 /* Clear any stale DMA IRQ semaphore token before starting a new RX transfer. */
393 bme680_clear_dma_semaphore_if_rtos(bme680_rx_sem);
394
395 st = i2c_dma_read(I2Cx, slave7, reg_addr, reg_data, len);
396 if (st != I2C_DMA_OK) {
397 return (int8_t)BME680_USER_I2C_E_START_FAIL;
398 }
399
400 if (bme680_wait_dma_complete_or_timeout(&bme680g_rx_cmplt, bme680_rx_sem)) {
401 bme680g_rx_cmplt = I2C_DMA_TRANSFER_IDLE;
402 return (int8_t)BME680_USER_I2C_OK;
403 }
404
405 bme680g_rx_cmplt = I2C_DMA_TRANSFER_IDLE;
406 I2Cx->CR1 |= I2C_CR1_STOP;
407
408 return (int8_t)BME680_USER_I2C_E_TIMEOUT;
409}

567int8_t bme680_set_regs(const uint8_t *reg_addr, const uint8_t *reg_data, uint8_t len, struct bme680_dev *dev)
568{
569 int8_t rslt = BME680_E_COM_FAIL;
570 uint8_t tmp_buff[BME680_TMP_BUFFER_LENGTH] = { 0 };
571 uint16_t index = 0;
572
573 /* Null pointer check */
574 rslt = null_ptr_check(dev);
575 if (rslt != BME680_OK) {
576 BME680_LOG_PRINTF("[ERROR] bme680_set_regs: null_ptr_check failed (err=%d)\r\n", (int)rslt);
577 app_fault_delay_ms(20);
578 return BME680_E_NULL_PTR;
579 }
580
581 /* This implementation is I2C-only — SPI support intentionally removed */
582 if (dev->intf != BME680_I2C_INTF) {
583 BME680_LOG_PRINTF("[ERROR] bme680_set_regs: device interface is not I2C (intf=%d). SPI removed.\r\n", dev->intf);
584 app_fault_delay_ms(20);
585 return BME680_E_COM_FAIL;
586 }
587
588 /* Validate length: now we only buffer reg_data bytes, so len must be < tmp buffer length */
589 if ((len == 0) || (len > (int)BME680_TMP_BUFFER_LENGTH)) {
590 BME680_LOG_PRINTF("[ERROR] bme680_set_regs: invalid length (%u)\r\n", (unsigned)len);
591 app_fault_delay_ms(20);
592 return BME680_E_INVALID_LENGTH;
593 }
594
595 /* Iterate through reg_addr/reg_data pairs and batch consecutive registers */
596 index = 0;
597 while (index < len) {
598 uint8_t start_reg = reg_addr[index];
599 uint8_t run_len = 1;
600 tmp_buff[0] = reg_data[index];
601
602 /* Build longest run of consecutive registers starting at index */
603 while ((index + run_len) < len) {
604 uint8_t prev = reg_addr[index + run_len - 1];
605 uint8_t cur = reg_addr[index + run_len];
606 if ((uint8_t)(cur) == (uint8_t)(prev + 1) && run_len < (uint8_t)BME680_TMP_BUFFER_LENGTH) {
607 tmp_buff[run_len] = reg_data[index + run_len];
608 run_len++;
609 } else {
610 break;
611 }
612 }
613
614 /* Perform I2C write: start_reg, tmp_buff[0..run_len-1] */
615 dev->com_rslt = dev->write(dev->handle, dev->dev_id, start_reg, tmp_buff, run_len);
616 if (dev->com_rslt != 0) {
617 BME680_LOG_PRINTF("[ERROR] bme680_set_regs: I2C write failed for start_reg=0x%02X (com_rslt=%d)\r\n",
618 (unsigned)start_reg, (int)dev->com_rslt);
619 app_fault_delay_ms(20);
620 rslt = BME680_E_COM_FAIL;
621 break;
622 }
623
624 /* advance index by run_len */
625 index += run_len;
626 }
627
628 if (index == len && dev->com_rslt == 0) {
629 rslt = BME680_OK;
630 }
631
632 return rslt;
633}

821 }

826 data_array[count] = data;
827 count++;
828 }

831 * RUN_GAS / NBCONV selection
832 * ------------------------ */
833 if (desired_settings & (BME680_RUN_GAS_SEL | BME680_NBCONV_SEL)) {

852 if (desired_settings & BME680_NBCONV_SEL)
853 data = BME680_SET_BITS_POS_0(data, BME680_NBCONV, dev->gas_sett.nb_conv);

855 reg_array[count] = reg_addr;
856 data_array[count] = data;
857 count++;
858 }

865 app_fault_delay_ms(20);
866 }
867 }
868
869 } while (0);
870
871 /* Restore previous intended power mode (preserve original behavior) */
872 dev->power_mode = intended_power_mode;
873
874 return rslt;

968int8_t bme680_set_sensor_mode(struct bme680_dev *dev)
969{
970 int8_t rslt = BME680_E_COM_FAIL;
971 uint8_t tmp_pow_mode = 0;
972 uint8_t pow_mode = 0;
973 const uint8_t reg_addr = BME680_CONF_T_P_MODE_ADDR;
974
975 /* Step 1 — Null pointer check (must remain) */
976 rslt = null_ptr_check(dev);
977 if (rslt != BME680_OK) {
978 BME680_LOG_PRINTF("[ERROR] bme680_set_sensor_mode: null_ptr_check failed (err=%d)\r\n", (int)rslt);
979 app_fault_delay_ms(20);
980 return BME680_E_NULL_PTR;
981 }
982
983 /* Step 2 — Poll until device enters sleep mode */
984 do {
985 rslt = bme680_get_regs(BME680_CONF_T_P_MODE_ADDR, &tmp_pow_mode, 1, dev);
986 if (rslt != BME680_OK) {
987 BME680_LOG_PRINTF("[ERROR] bme680_set_sensor_mode: failed to read power-mode reg (err=%d)\r\n", (int)rslt);
988 app_fault_delay_ms(20);
989 return rslt;
990 }
991
992 pow_mode = (uint8_t)(tmp_pow_mode & BME680_MODE_MSK);
993
994 if (pow_mode != BME680_SLEEP_MODE) {
995 /* Put sensor to sleep (clear mode bits) */
996 uint8_t sleep_val = (uint8_t)(tmp_pow_mode & (~BME680_MODE_MSK));
997 rslt = bme680_set_regs(&reg_addr, &sleep_val, 1, dev);
998 if (rslt != BME680_OK) {
999 BME680_LOG_PRINTF("[ERROR] bme680_set_sensor_mode: failed to write sleep mode (err=%d)\r\n", (int)rslt);
1000 app_fault_delay_ms(20);
1001 return rslt;
1002 }
1003 /* Wait a bit before polling again */
1004 dev->delay_ms(BME680_POLL_PERIOD_MS);
1005 }
1006 /* loop until pow_mode == SLEEP */
1007 } while (pow_mode != BME680_SLEEP_MODE);
1008
1009 /* Step 3 — If requested mode is not sleep, program it now */
1010 if (dev->power_mode != BME680_SLEEP_MODE) {
1011 uint8_t new_val = (uint8_t)((tmp_pow_mode & ~BME680_MODE_MSK) | (dev->power_mode & BME680_MODE_MSK));
1012 if (rslt == BME680_OK) {
1013 rslt = bme680_set_regs(&reg_addr, &new_val, 1, dev);
1014 if (rslt != BME680_OK) {
1015 BME680_LOG_PRINTF("[ERROR] bme680_set_sensor_mode: failed to set requested power_mode=%u (err=%d)\r\n",
1016 (unsigned)dev->power_mode, (int)rslt);
1017 app_fault_delay_ms(20);
1018 return rslt;
1019 }
1020 }
1021 }
1022
1023 return rslt;
1024}

1076void bme680_set_profile_dur(uint16_t duration, struct bme680_dev *dev)
1077{
1078 /* Calculate durations in microseconds then convert to ms as in original logic */
1079 uint32_t tph_dur_us = 0U; /* TPH-related duration in microseconds */
1080 uint32_t meas_cycles = 0U; /* effective number of measurement cycles */
1081
1082 /* map oversampling setting -> measurement cycles (same mapping you used) */
1083 const uint8_t os_to_meas_cycles[6] = { 0U, 1U, 2U, 4U, 8U, 16U };
1084
1085 /* sum cycles for temperature, pressure and humidity oversampling */
1086 meas_cycles = (uint32_t)os_to_meas_cycles[dev->tph_sett.os_temp];
1087 meas_cycles += (uint32_t)os_to_meas_cycles[dev->tph_sett.os_pres];
1088 meas_cycles += (uint32_t)os_to_meas_cycles[dev->tph_sett.os_hum];
1089
1090 /* TPH measurement duration (microseconds) */
1091 tph_dur_us = meas_cycles * UINT32_C(1963); /* base per-cycle time */
1092 tph_dur_us += UINT32_C(477 * 4); /* TPH switching time */
1093 tph_dur_us += UINT32_C(477 * 5); /* gas measurement time */
1094 tph_dur_us += UINT32_C(500); /* rounding pad */
1095
1096 /* Convert to milliseconds (integer division) */
1097 tph_dur_us /= UINT32_C(1000);
1098
1099 /* add wake-up time (1 ms) */
1100 tph_dur_us += UINT32_C(1);
1101
1102 /* final heater duration = requested duration - computed TPH duration (both in ms) */
1103 dev->gas_sett.heatr_dur = (uint16_t)((uint32_t)duration - (uint32_t)tph_dur_us);
1104}

1941static uint8_t calc_heater_dur(uint16_t dur)
1942{
1943 uint32_t remaining = (uint32_t)dur;
1944 uint8_t factor = 0;
1945 uint8_t durval = 0;
1946
1947 /* Max duration shortcut (preserve original behavior) */
1948 if (remaining >= 0xFC0U) {
1949 return 0xFFU;
1950 }
1951
1952 /* Keep dividing by 4 until the value fits in 6 bits (<= 0x3F) */
1953 while (remaining > 0x3FU) {
1954 remaining = remaining / 4U;
1955 /* factor increments correspond to how many times we divided by 4 */
1956 if (factor < 0xFFU) { /* defensive guard, though factor will stay small */
1957 ++factor;
1958 }
1959 }
1960
1961 /* Compose register value: low 6 bits = remaining, high 2 bits = factor */
1962 durval = (uint8_t)(remaining + ((uint32_t)factor << 6));
1963
1964 return durval;
