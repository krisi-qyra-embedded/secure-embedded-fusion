49static void DS18B20_AppSleepMs(uint32_t ms)
50{
51 if (ms == 0U) {
52 return;
53 }
54 /* Scheduler running -> use RTOS delay. */
55 if (xTaskGetSchedulerState() == taskSCHEDULER_RUNNING)
56 {
57 TickType_t ticks = pdMS_TO_TICKS(ms);
58 if (ticks == 0U) {
59 ticks = 1U;
60 }
61 vTaskDelay(ticks);
62 return;
63 }
64 /* Before scheduler start use blocking delay.
65 * No DS18B20 IRQ here to wake WFI.
66 */
67 delay_ms(ms);

130/* Check that all expected sensors can return a value. */
131static DS18B20_Status_t DS18B20_AppVerifyAllSensorsReady(DS18B20_App_t *App)
132{
133 uint8_t i;
134 DS18B20_Status_t st;
135 float temp;
136 if (App == NULL) {
137 return DS18B20_STATUS_BAD_PARAM;
138 }
139 if (App->OW.RomCnt == 0U) {
140 return DS18B20_STATUS_NO_DEVICE;
141 }
142 if ((App->ExpectedSensorCount != 0U) &&
143 (App->OW.RomCnt != App->ExpectedSensorCount))
144 {
145 return DS18B20_STATUS_NOT_FOUND;
146 }
147 /* Verify the complete sensor group. */
148 st = DS18B20_StartAll(&App->OW);
149 if (st != DS18B20_STATUS_OK) {

156 if (st != DS18B20_STATUS_OK) {
157 return st;
158 }

160 }
161 return DS18B20_STATUS_OK;

164/* Run one of the 3 recovery steps. */
165static DS18B20_Status_t DS18B20_AppApplyRecoveryPhase(DS18B20_App_t *App,
166 uint8_t phase)
167{
168 DS18B20_Status_t ds_st;
169 OneWire_Status_t ow_st;
170 if (App == NULL) {
171 return DS18B20_STATUS_BAD_PARAM;
172 }
173 switch (phase)
174 {
175 case 1U:
176 {
177 DS18B20_AppLogErrorDelay(
178 "ACT: DS18B20 recovery 1/3 -> soft OneWire re-arm + bus reset\r\n");
179 ow_st = OneWire_Init(&App->OW);
180 if (ow_st != ONEWIRE_STATUS_OK) {
181 return DS18B20_MapOwStatus(ow_st);
182 }
183 ow_st = OneWire_Reset(&App->OW);
184 if (ow_st != ONEWIRE_STATUS_OK) {
185 return DS18B20_MapOwStatus(ow_st);
186 }
187 /* First recovery try keeps the current ROM table. */
188 return DS18B20_AppVerifyAllSensorsReady(App);
189 }
190 case 2U:
191 {
192 DS18B20_AppLogErrorDelay(
193 "ACT: DS18B20 recovery 2/3 -> full rescan + DS18B20 re-init\r\n");
194 DS18B20_AppClearDiscoveryState(App);
195 ds_st = DS18B20_Init(&App->DS, &App->OW);
196 if (ds_st != DS18B20_STATUS_OK) {
197 return ds_st;
198 }
199 DS18B20_LOG_PRINTF("[DBG][DS18B20] Init result=%s RomCnt=%u Expected=%u\r\n",
200 DS18B20_StatusStr(ds_st),
201 (unsigned int)App->OW.RomCnt,
202 (unsigned int)App->ExpectedSensorCount);
203 return DS18B20_AppVerifyAllSensorsReady(App);
204 }
205 case 3U:
206 {
207 DS18B20_AppLogErrorDelay(
208 "ACT: DS18B20 recovery 3/3 -> cold GPIO re-bootstrap + full re-init\r\n");
209 DS18B20_AppGpioInit(App->OW.DataPort, App->OW.DataPin);
210 DS18B20_AppSleepMs(10U);
211 DS18B20_AppClearDiscoveryState(App);
212 ds_st = DS18B20_Init(&App->DS, &App->OW);
213 if (ds_st != DS18B20_STATUS_OK) {
214 return ds_st;
215 }
216 return DS18B20_AppVerifyAllSensorsReady(App);
217 }
218 default:
219 return DS18B20_STATUS_BAD_PARAM;
220 }

230static DS18B20_AppRuntimeStatus_t DS18B20_AppRunRecoverySequence(DS18B20_App_t *App,
231 const char *reason)
232{
233 DS18B20_AppMarkFaultInProgress();
234 uint8_t phase;
235 DS18B20_Status_t st;
236 if (App == NULL) {
237 return DS18B20_APP_RUNTIME_BAD_PARAM;
238 }
239 for (phase = 1U; phase <= 3U; phase++)
240 {
241 st = DS18B20_AppApplyRecoveryPhase(App, phase);
242 if (st == DS18B20_STATUS_OK)
243 {
244 DS18B20_AppLogErrorDelay(
245 "OK: DS18B20 recovery succeeded on attempt %u/3 (%s)\r\n",
246 (unsigned int)phase,
247 (reason != NULL) ? reason : "runtime");
248 App->RecoveryAttempt = 0U;
249 DS18B20_AppClearFaultActive();
250 DS18B20_AppResetRuntimeState(App);
251 return DS18B20_APP_RUNTIME_RECOVERED;
252 }
253 DS18B20_AppLogErrorDelay(
254 "[ERR] DS18B20 recovery attempt %u/3 failed (%s): %s\r\n",
255 (unsigned int)phase,
256 (reason != NULL) ? reason : "runtime",
257 DS18B20_StatusStr(st));
258 }
259 DS18B20_AppLogErrorDelay(
260 "[FATAL] DS18B20 recovery exhausted 3/3 attempts (%s)\r\n",
261 (reason != NULL) ? reason : "runtime");
262 DS18B20_AppMarkFaultCompleted();
263 return DS18B20_APP_RUNTIME_RECOVERY_FAIL;
264}

429/* Try to recover one failed sensor. */
430static DS18B20_AppRuntimeStatus_t DS18B20_AppRecoverFailedSensor(DS18B20_App_t *App,
431 uint8_t sensor_index)
432{
433 DS18B20_AppMarkFaultInProgress();
434 uint8_t attempt;
435 uint8_t sensor_rom[8];
436 float recovered_temp = 0.0f;
437 DS18B20_Status_t ds_st;
438 OneWire_Status_t ow_st;
439 if ((App == NULL) || (sensor_index >= App->OW.RomCnt)) {
440 return DS18B20_APP_RUNTIME_BAD_PARAM;
441 }
442 memcpy(sensor_rom, App->DS.DevAddr[sensor_index], sizeof(sensor_rom));
443 for (attempt = 1U; attempt <= 3U; attempt++)
444 {
445 switch (attempt)
446 {
447 case 1U:
448 {
449 DS18B20_AppLogErrorDelay(
450 "ACT: DS18B20 sensor %u recovery 1/3 -> OneWire re-arm + bus reset\r\n",
451 (unsigned int)(sensor_index + 1U));
452 ow_st = OneWire_Init(&App->OW);
453 if (ow_st != ONEWIRE_STATUS_OK) {
454 DS18B20_AppLogErrorDelay(
455 "[ERR] DS18B20 sensor %u recovery 1/3 OneWire_Init failed\r\n",
456 (unsigned int)(sensor_index + 1U));
457 continue;
458 }
459 ow_st = OneWire_Reset(&App->OW);
460 if (ow_st != ONEWIRE_STATUS_OK) {
461 DS18B20_AppLogErrorDelay(
462 "[ERR] DS18B20 sensor %u recovery 1/3 OneWire_Reset failed\r\n",
463 (unsigned int)(sensor_index + 1U));
464 continue;
465 }
466 break;
467 }
468 case 2U:
469 {
470 DS18B20_AppLogErrorDelay(
471 "ACT: DS18B20 sensor %u recovery 2/3 -> reconfigure only failed sensor\r\n",
472 (unsigned int)(sensor_index + 1U));
473 ow_st = OneWire_Reset(&App->OW);
474 if (ow_st != ONEWIRE_STATUS_OK) {
475 DS18B20_AppLogErrorDelay(
476 "[ERR] DS18B20 sensor %u recovery 2/3 OneWire_Reset failed\r\n",
477 (unsigned int)(sensor_index + 1U));
478 continue;
479 }
480 ds_st = DS18B20_SetResolution(&App->OW, sensor_rom, App->DS.Resolution);
481 if (ds_st != DS18B20_STATUS_OK) {
482 DS18B20_AppLogErrorDelay(

485 DS18B20_StatusStr(ds_st));
486 continue;
487 }
488 ds_st = DS18B20_SetTempAlarm(&App->OW, sensor_rom, 0, 31);
489 if (ds_st != DS18B20_STATUS_OK) {
490 DS18B20_AppLogErrorDelay(
491 "[ERR] DS18B20 sensor %u recovery 2/3 SetTempAlarm failed: %s\r\n",
492 (unsigned int)(sensor_index + 1U),
493 DS18B20_StatusStr(ds_st));
494 continue;
495 }
496 break;
497 }
498 case 3U:
499 default:
500 {
501 DS18B20_AppLogErrorDelay(
502 "ACT: DS18B20 sensor %u recovery 3/3 -> stop group + full DS18B20 re-bootstrap\r\n",
503 (unsigned int)(sensor_index + 1U));
504 DS18B20_AppGpioInit(App->OW.DataPort, App->OW.DataPin);
505 App->OW.RomCnt = 0U;
506 memset(App->DS.DevAddr, 0, sizeof(App->DS.DevAddr));
507 memset(App->DS.AlmAddr, 0, sizeof(App->DS.AlmAddr));
508 memset(App->DS.Temperature, 0, sizeof(App->DS.Temperature));
509 ds_st = DS18B20_Init(&App->DS, &App->OW);
510 if (ds_st != DS18B20_STATUS_OK) {
511 DS18B20_AppLogErrorDelay(
512 "[ERR] DS18B20 sensor %u recovery 3/3 DS18B20_Init failed: %s\r\n",
513 (unsigned int)(sensor_index + 1U),
514 DS18B20_StatusStr(ds_st));
515 continue;
516 }
517 if (DS18B20_AppSensorCountHealthy(App) == 0U) {
518 DS18B20_AppLogErrorDelay(
519 "[ERR] DS18B20 sensor %u recovery 3/3 sensor-count check failed\r\n",
520 (unsigned int)(sensor_index + 1U));
521 continue;
522 }
523 break;
524 }
525 }
526 ds_st = DS18B20_Read(&App->OW, sensor_rom, &recovered_temp);
527 if (ds_st == DS18B20_STATUS_OK)
528 {
529 App->DS.Temperature[sensor_index] = recovered_temp;
530 DS18B20_AppLogErrorDelay(
531 "OK: DS18B20 sensor %u recovered on attempt %u/3 -> resume normal sampling\r\n",
532 (unsigned int)(sensor_index + 1U),
533 (unsigned int)attempt);
534 App->RecoveryAttempt = 0U;
535 DS18B20_AppClearFaultActive();
536 DS18B20_AppResetRuntimeState(App);
537 return DS18B20_APP_RUNTIME_RECOVERED;
538 }
539 DS18B20_AppLogErrorDelay(
540 "[ERR] DS18B20 sensor %u retry %u/3 readback failed: %s\r\n",
541 (unsigned int)(sensor_index + 1U),
542 (unsigned int)attempt,
543 DS18B20_StatusStr(ds_st));
544 }
545 DS18B20_AppLogErrorDelay(
546 "[FATAL] DS18B20 sensor %u failed after 3 dedicated recovery attempts\r\n",
547 (unsigned int)(sensor_index + 1U));
548 DS18B20_AppMarkFaultCompleted();
549 return DS18B20_APP_RUNTIME_RECOVERY_FAIL;
550}

601 if (st != ONEWIRE_STATUS_OK) return DS18B20_MapOwStatus(st);
602 st = OneWire_ReadByte(OW, &conf);
603 if (st != ONEWIRE_STATUS_OK) return DS18B20_MapOwStatus(st);
604 st = OneWire_ReadByte(OW, &conf);
605 if (st != ONEWIRE_STATUS_OK) return DS18B20_MapOwStatus(st);
606 st = OneWire_ReadByte(OW, &conf);
607 if (st != ONEWIRE_STATUS_OK) return DS18B20_MapOwStatus(st);
608 st = OneWire_ReadByte(OW, &conf);
609 if (st != ONEWIRE_STATUS_OK) return DS18B20_MapOwStatus(st);
610 *resolution = (uint8_t)(((conf & 0x60U) >> 5U) + 9U);

788DS18B20_Status_t DS18B20_Read(OneWire_t *OW, const uint8_t *ROM, float *Destination)
789{
790 OneWire_Status_t st;
791 uint16_t temperature;
792 uint8_t resolution;
793 int16_t digit;
794 uint8_t minus = 0U;
795 float decimal;
796 uint8_t i;
797 uint8_t data[9];
798 uint8_t crc;
799 if ((OW == NULL) || (Destination == NULL)) {
800 return DS18B20_STATUS_BAD_PARAM;
801 }
802 if (DS18B20_IsValid(ROM) == 0U) {
803 return DS18B20_STATUS_INVALID_ROM;
804 }
805 st = OneWire_Reset(OW);
806 if (st != ONEWIRE_STATUS_OK) {
807 return DS18B20_MapOwStatus(st);
808 }
809 st = OneWire_SelectWithPointer(OW, ROM);
810 if (st != ONEWIRE_STATUS_OK) {
811 return DS18B20_MapOwStatus(st);
812 }
813 st = OneWire_WriteByte(OW, DS18B20_CMD_READSCRATCHPAD);
814 if (st != ONEWIRE_STATUS_OK) {
815 return DS18B20_MapOwStatus(st);
816 }
817 for (i = 0U; i < 9U; i++) {
818 st = OneWire_ReadByte(OW, &data[i]);
819 if (st != ONEWIRE_STATUS_OK) {
820 return DS18B20_MapOwStatus(st);
821 }
822 }
823 crc = OneWire_CRC8(data, 8U);
824 if (crc != data[8]) {
825 return DS18B20_STATUS_CRC_ERROR;
826 }
827 temperature = (uint16_t)data[0] | ((uint16_t)data[1] << 8U);
828 if ((temperature & 0x8000U) != 0U) {
829 temperature = (uint16_t)(~temperature + 1U);
830 minus = 1U;
831 }
832 resolution = (uint8_t)(((data[4] & 0x60U) >> 5U) + 9U);
833 digit = (int16_t)(temperature >> 4U);
834 digit |= (int16_t)(((temperature >> 8U) & 0x07U) << 4U);
835 switch (resolution)
836 {
837 case 9:
838 decimal = (float)((temperature >> 3U) & 0x01U) * DS18B20_DECIMAL_STEPS_9BIT;
839 break;
840 case 10:
841 decimal = (float)((temperature >> 2U) & 0x03U) * DS18B20_DECIMAL_STEPS_10BIT;
842 break;
843 case 11:
844 decimal = (float)((temperature >> 1U) & 0x07U) * DS18B20_DECIMAL_STEPS_11BIT;
845 break;
846 case 12:
847 decimal = (float)(temperature & 0x0FU) * DS18B20_DECIMAL_STEPS_12BIT;
848 break;
849 default:
850 return DS18B20_STATUS_BUS_ERROR;
851 }
852 *Destination = (float)digit + decimal;
853 if (minus != 0U) {
854 *Destination = -(*Destination);
855 }
856 return DS18B20_STATUS_OK;
857}

987{
988 OneWire_Status_t ow_st;
989 DS18B20_Status_t ds_st;
990 if ((DS == NULL) || (OW == NULL)) {
991 return DS18B20_STATUS_BAD_PARAM;
992 }
993 if (DS->ConversionTimeoutMs == 0U) {
994 DS->ConversionTimeoutMs = DS18B20_GetConversionTimeMs(DS->Resolution) + 50U;
995 }
996 memset(DS->DevAddr, 0, sizeof(DS->DevAddr));
997 memset(DS->AlmAddr, 0, sizeof(DS->AlmAddr));
998 memset(DS->Temperature, 0, sizeof(DS->Temperature));
999 ow_st = OneWire_Init(OW);
1000 if (ow_st != ONEWIRE_STATUS_OK) {
1001 return DS18B20_MapOwStatus(ow_st);
1002 }
1003 while (OW->RomCnt < DS18B20_MaxCnt)
1004 {
1005 ow_st = OneWire_Search(OW, ONEWIRE_CMD_SEARCHROM);
1006 if (ow_st == ONEWIRE_STATUS_NOT_FOUND) {
1007 break;
1008 }
1009 if (ow_st != ONEWIRE_STATUS_OK) {
1010 return DS18B20_MapOwStatus(ow_st);
1011 }
1012 ow_st = OneWire_GetDevRom(OW, DS->DevAddr[OW->RomCnt]);
1013 if (ow_st != ONEWIRE_STATUS_OK) {
1014 return DS18B20_MapOwStatus(ow_st);
1015 }
1016 if (DS18B20_IsValid(DS->DevAddr[OW->RomCnt]) == 0U) {
1017 continue;
1018 }
1019 ds_st = DS18B20_SetResolution(OW, DS->DevAddr[OW->RomCnt], DS->Resolution);
1020 if (ds_st != DS18B20_STATUS_OK) {
1021 return ds_st;
1022 }
1023 ds_st = DS18B20_SetTempAlarm(OW, DS->DevAddr[OW->RomCnt], 0, 0);
1024 if (ds_st != DS18B20_STATUS_OK) {
1025 return ds_st;
1026 }
1027 OW->RomCnt++;
1028 }
1029 return (OW->RomCnt != 0U) ? DS18B20_STATUS_OK : DS18B20_STATUS_NOT_FOUND;
1030}

1160 DS18B20_AppRuntimeStatus_t DS18B20_AppRun(DS18B20_App_t *App)
1161 {
1162 uint8_t i;
1163 uint8_t is_done = 0U;
1164 DS18B20_Status_t st;
1165 DS18B20_AppRuntimeStatus_t recovery_st;
1166 if (App == NULL) {
1167 return DS18B20_APP_RUNTIME_BAD_PARAM;
1168 }
1169 switch (App->State)
1170 {
1171 case DS18B20_APP_START_CONVERSION:
1172 {
1173 if (App->OW.RomCnt == 0U)
1174 {
1175 DS18B20_AppLogErrorDelay("[ERR] No DS18B20 devices available\r\n");
1176 recovery_st = DS18B20_AppRunRecoverySequence(App, "no-device at start-conversion");
1177 if (recovery_st == DS18B20_APP_RUNTIME_RECOVERED) {
1178 return DS18B20_APP_RUNTIME_RECOVERED;
1179 }
1180 return DS18B20_APP_RUNTIME_RECOVERY_FAIL;
1181 }
1182 if ((App->ExpectedSensorCount != 0U) &&
1183 (App->OW.RomCnt != App->ExpectedSensorCount))
1184 {
1185 DS18B20_AppLogErrorDelay("[ERR] Expected %u sensors but have %u\r\n",
1186 (unsigned int)App->ExpectedSensorCount,
1187 (unsigned int)App->OW.RomCnt);
1188 recovery_st = DS18B20_AppRunRecoverySequence(App, "wrong sensor count at start-conversion");
1189 if (recovery_st == DS18B20_APP_RUNTIME_RECOVERED) {
1190 return DS18B20_APP_RUNTIME_RECOVERED;
1191 }
1192 return DS18B20_APP_RUNTIME_RECOVERY_FAIL;
1193 }
1194 st = DS18B20_StartAll(&App->OW);
1195 if (st != DS18B20_STATUS_OK)
1196 {
1197 DS18B20_AppLogErrorDelay("[ERR] DS18B20_StartAll failed: %s\r\n",
1198 DS18B20_StatusStr(st));
1199 recovery_st = DS18B20_AppRunRecoverySequence(App, "StartAll");
1200 if (recovery_st == DS18B20_APP_RUNTIME_RECOVERED) {
1201 return DS18B20_APP_RUNTIME_RECOVERED;
1202 }
1203 return DS18B20_APP_RUNTIME_RECOVERY_FAIL;
1204 }
1205 App->DeadlineMs = HAL_GetTick() + App->DS.ConversionTimeoutMs;
1206 App->TimeoutReported = 0U;
1207 App->State = DS18B20_APP_WAIT_CONVERSION;
1208 return DS18B20_APP_RUNTIME_OK;
1209 }
1210 case DS18B20_APP_WAIT_CONVERSION:
1211 {
1212 st = DS18B20_IsConversionDone(&App->OW, &is_done);
1213 if (st != DS18B20_STATUS_OK)
1214 {
1215 DS18B20_AppLogErrorDelay("[ERR] DS18B20_IsConversionDone failed: %s\r\n",
1216 DS18B20_StatusStr(st));
1217 recovery_st = DS18B20_AppRunRecoverySequence(App, "IsConversionDone");
1218 if (recovery_st == DS18B20_APP_RUNTIME_RECOVERED) {
1219 return DS18B20_APP_RUNTIME_RECOVERED;
1220 }
1221 return DS18B20_APP_RUNTIME_RECOVERY_FAIL;
1222 }
1223 if (is_done != 0U)
1224 {
1225 App->State = DS18B20_APP_READ_AND_PRINT;
1226 return DS18B20_APP_RUNTIME_OK;
1227 }
1228 if ((int32_t)(HAL_GetTick() - App->DeadlineMs) >= 0)
1229 {
1230 if (App->TimeoutReported == 0U)
1231 {
1232 DS18B20_AppLogErrorDelay("[ERR] DS18B20 conversion timeout\r\n");
1233 App->TimeoutReported = 1U;
1234 }
1235 recovery_st = DS18B20_AppRunRecoverySequence(App, "conversion timeout");
1236 if (recovery_st == DS18B20_APP_RUNTIME_RECOVERED) {
1237 return DS18B20_APP_RUNTIME_RECOVERED;
1238 }
1239 return DS18B20_APP_RUNTIME_RECOVERY_FAIL;
1240 }
1241 DS18B20_AppSleepMs(1U);
1242 return DS18B20_APP_RUNTIME_OK;
1243 }
1244 case DS18B20_APP_READ_AND_PRINT:
1245 default:
1246 {
1247 /* One failed read rejects the whole sensor group. */
1248 for (i = 0U; i < App->OW.RomCnt; i++)
1249 {
1250 st = DS18B20_Read(&App->OW, App->DS.DevAddr[i], &App->DS.Temperature[i]);
1251 if (st != DS18B20_STATUS_OK)
1252 {
