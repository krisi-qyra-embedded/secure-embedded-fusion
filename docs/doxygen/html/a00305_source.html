238static bool aws_json_append(
239 char *payload,
240 size_t payload_size,
241 size_t *offset,
242 const char *format,
243 ...)
244{
245 if ((payload == NULL) ||
246 (payload_size == 0U) ||
247 (offset == NULL) ||
248 (format == NULL) ||
249 (*offset >= payload_size))
250 {
251 return false;
252 }
253 va_list args;
254 va_start(
255 args,
256 format);
257 int written =
258 vsnprintf(
259 payload + *offset,
260 payload_size - *offset,
261 format,
262 args);
263 va_end(args);
264 if (written < 0)
265 {
266 payload[*offset] = '\0';
267 return false;
268 }
269 if ((size_t)written >=
270 (payload_size - *offset))
271 {
272 payload[
273 payload_size - 1U] = '\0';
274 return false;
275 }
276 *offset +=
277 (size_t)written;
278 return true;
279}

286static void aws_json_sanitize_ascii(
287 const char *src,
288 char *dst,
289 size_t dst_size)
290{
291 size_t i = 0U;
292 if ((dst == NULL) ||
293 (dst_size == 0U))
294 {
295 return;
296 }
297 if (src == NULL)
298 {
299 dst[0] = '\0';
300 return;
301 }
302 while ((src[i] != '\0') &&
303 ((i + 1U) < dst_size))
304 {
305 unsigned char c =
306 (unsigned char)src[i];
307 dst[i] =
308 ((c < 0x20U) ||
309 (c == '"') ||
310 (c == '\\'))
311 ? '_'
312 : (char)c;
313 i++;
314 }
315 dst[i] = '\0';
316}

786

819 }

821 (NETWORK_RECONNECTED != rc))
822 {
823 ESP_LOGW(
824 TAG,
825 "AWS_YIELD | event=non_success | "

827 rc);

831 aws_iot_watchdog_kick();
832 continue;
833 }

836 "Stack remaining for task '%s' is %d",

838 uxTaskGetStackHighWaterMark(NULL));

840 pdMS_TO_TICKS(
841 AWS_IOT_PUBLISH_PERIOD_MS));
842 aws_iot_watchdog_kick();
843 /* Build one payload from local and STM32 readings. */
844 if (!build_sensor_json_payload(

846 sizeof(cPayload)))
