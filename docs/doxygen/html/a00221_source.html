114 }

117 while ((HAL_GetTick() - start) < ms)

184/* Read, validate and decode one PMS5003 frame. */
185static PMS25_Status_t PMS25_ReadInternal(PMS25_Data_t *out,
186 uint32_t timeout_ms,
187 uint8_t *frame_available_out)
188{
189 uint8_t frame[PMS25_FRAME_SIZE];
190 uint8_t first_byte;
191 uint16_t checksum_calc;
192 uint16_t checksum_rx;
193 uint32_t start_tick;
194 uint32_t elapsed;
195 uint32_t remaining;
196 UART_Status_t uart_st;
197 PMS25_Status_t last_nonfatal = PMS25_STATUS_TIMEOUT;
198 uint8_t i;
199 /* A frame read needs valid output storage and a bounded timeout. */
200 if ((out == NULL) || (timeout_ms == 0U))
201 {
202 return PMS25_STATUS_INVALID_PARAM;
203 }
204 if (frame_available_out != NULL)
205 {
206 *frame_available_out = 0U;
207 }
208 memset(frame, 0, sizeof(frame));
209 /* Keep one deadline for the complete 32-byte frame. */
210 start_tick = HAL_GetTick();
211 while (1)
212 {
213 elapsed = HAL_GetTick() - start_tick;
214 if (elapsed >= timeout_ms)
215 {
216 return last_nonfatal;
217 }
218 remaining = timeout_ms - elapsed;
219 /* Scan one byte at a time until the first header byte is found. */
220 uart_st = UART1_RX_DMA_Receive(&first_byte, 1U, remaining);
221 if (uart_st != UART_STATUS_OK)
222 {
223 return PMS25_MapUartStatus(uart_st);
224 }
225 /* Ignore bytes until frame sync starts with 0x42. */
226 if (first_byte != PMS25_HEADER_BYTE_0)
227 {
228 last_nonfatal = PMS25_STATUS_SYNC_ERROR;
229 continue;
230 }
231 frame[0] = first_byte;
232 elapsed = HAL_GetTick() - start_tick;
233 if (elapsed >= timeout_ms)
234 {
235 return last_nonfatal;
236 }
237 remaining = timeout_ms - elapsed;
238 /* Once synced, collect the rest of the frame in one DMA read. */
239 uart_st = UART1_RX_DMA_Receive(&frame[1], 31U, remaining);
240 if (uart_st != UART_STATUS_OK)
241 {
242 return PMS25_MapUartStatus(uart_st);
243 }
244 /* Second header byte must be 0x4D. */
245 if (frame[1] != PMS25_HEADER_BYTE_1)
246 {
247 last_nonfatal = PMS25_STATUS_SYNC_ERROR;
248 continue;
249 }
250 if (frame_available_out != NULL)
251 {
252 *frame_available_out = 1U;
253 }
254 /* Checksum covers bytes 0..29. */
255 checksum_calc = 0U;
256 for (i = 0U; i < 30U; i++)
257 {
258 checksum_calc = (uint16_t)(checksum_calc + frame[i]);
259 }
260 /* Last two bytes carry the sensor checksum. */
261 checksum_rx = PMS25_ReadBE16(&frame[30]);
262 if (checksum_calc != checksum_rx)
263 {
264 last_nonfatal = PMS25_STATUS_CHECKSUM_ERROR;
265 continue;
266 }
267 /* Validate the PMS5003 payload length before decoding fields. */
268 out->frame_len = PMS25_ReadBE16(&frame[2]);
269 if (out->frame_len != PMS25_EXPECTED_FRAME_LEN)
270 {
271 last_nonfatal = PMS25_STATUS_LENGTH_ERROR;
272 continue;
273 }
274 /* Decode concentration values. */
275 out->pm1_0_standard = PMS25_ReadBE16(&frame[4]);
276 out->pm2_5_standard = PMS25_ReadBE16(&frame[6]);
277 out->pm10_standard = PMS25_ReadBE16(&frame[8]);
278 out->pm1_0_atmospheric = PMS25_ReadBE16(&frame[10]);
279 out->pm2_5_atmospheric = PMS25_ReadBE16(&frame[12]);
280 out->pm10_atmospheric = PMS25_ReadBE16(&frame[14]);
281 /* Decode particle counters. */
282 out->particles_0_3um = PMS25_ReadBE16(&frame[16]);
283 out->particles_0_5um = PMS25_ReadBE16(&frame[18]);
284 out->particles_1_0um = PMS25_ReadBE16(&frame[20]);
285 out->particles_2_5um = PMS25_ReadBE16(&frame[22]);
286 out->particles_5_0um = PMS25_ReadBE16(&frame[24]);
287 out->particles_10um = PMS25_ReadBE16(&frame[26]);
288 out->reserved = PMS25_ReadBE16(&frame[28]);
289 out->checksum = checksum_rx;
290 return PMS25_STATUS_OK;
291 }
292}

413PMS25_AppRuntimeStatus_t PMS25_AppRun(PMS25_App_t *app)
414{
415 PMS25_Status_t pm_st;
416 UART_Status_t uart_st;
417 uint8_t frame_available = 0U;
418 uint32_t now;
419 uint32_t no_frame_elapsed;
420 if (app == NULL)
421 {
422 PMS25_MarkFaultInProgress();
423 PM25_LOG_PRINTF("[ERROR][PM25] PMS25_AppRun: NULL app pointer\r\n");
424 app_fault_delay_ms(20U);
425 return PMS25_APP_RUNTIME_BAD_PARAM;
426 }
427 /* Try to receive one complete validated frame. */
428 pm_st = PMS25_ReadInternal(&app->Data,
429 app->FramePollTimeoutMs,
430 &frame_available);
431 now = HAL_GetTick();
432 /* Transport/sync/checksum faults enter the shared fault path. */
433 if ((pm_st == PMS25_STATUS_TIMEOUT) ||
434 (pm_st == PMS25_STATUS_UART_ERROR) ||
435 (pm_st == PMS25_STATUS_SYNC_ERROR) ||
436 (pm_st == PMS25_STATUS_CHECKSUM_ERROR)) {
437 PMS25_MarkFaultInProgress();
438 }
439 /* Any complete frame keeps the no-frame timer alive. */
440 if (frame_available != 0U)
441 {
442 app->LastFrameAvailableTick = now;
443 }
444 /* Valid frame clears recovery state and can be published. */
445 if (pm_st == PMS25_STATUS_OK)
446 {
447 app->RecoveryAttempt = 0U;
448 PMS25_ClearFaultActive();
449 (void)telemetry_publish_pm25(
450 app->Data.pm1_0_standard,
451 app->Data.pm2_5_standard,
452 app->Data.pm10_standard,
453 app->Data.pm1_0_atmospheric,
454 app->Data.pm2_5_atmospheric,
455 app->Data.pm10_atmospheric,
456 app->Data.particles_0_3um,
457 app->Data.particles_0_5um,
458 app->Data.particles_1_0um,
459 app->Data.particles_2_5um,
460 app->Data.particles_5_0um,
461 app->Data.particles_10um);
462 if (system_fault_capture_or_critical_fault_active() == 0U)
463 {
464 PMS25_LogSample(&app->Data);
465 }
466 return PMS25_APP_RUNTIME_OK;
467 }
468 /* Do not recover on a single bad or missing frame. */
469 no_frame_elapsed = now - app->LastFrameAvailableTick;
470 /* Wait until the no-frame threshold is really exceeded. */
471 if (no_frame_elapsed < app->NoFrameFaultMs)
472 {
473 PMS25_AppSleepMs(PMS25_PREFAULT_BACKOFF_MS);
474 return PMS25_APP_RUNTIME_OK;
475 }
476 /* Count bounded UART recovery attempts. */
477 if (app->RecoveryAttempt < app->MaxRecoveryAttempts)
478 {
479 app->RecoveryAttempt++;
480 }
481 else
482 {
483 app->RecoveryAttempt = app->MaxRecoveryAttempts;
484 }
485 PMS25_LogRuntimeReadFailure(pm_st,
486 no_frame_elapsed,
487 app->NoFrameFaultMs,
488 app->RecoveryAttempt,
489 app->MaxRecoveryAttempts);
490 PMS25_ClearData(app);
491 /* Reinitialize the UART/PM2.5 receive path before retrying. */
492 uart_st = UART1_PM25_Recover(app);
493 if (uart_st == UART_STATUS_OK)
494 {
495 PM25_LOG_PRINTF("[INFO][PM25] UART1_PM25_Recover OK after attempt %u/%u\r\n",
496 (unsigned)app->RecoveryAttempt,
497 (unsigned)app->MaxRecoveryAttempts);
498 app_fault_delay_ms(20U);
499 }
500 else
501 {
502 PMS25_MarkFaultInProgress();
503 PM25_LOG_PRINTF("[WARN][PM25] UART1_PM25_Recover failed, st=%s, attempt=%u/%u\r\n",
504 PMS25_UartStatusStr(uart_st),
505 (unsigned)app->RecoveryAttempt,
506 (unsigned)app->MaxRecoveryAttempts);
507 app_fault_delay_ms(20U);
508 }
509 /* Keep the fault window expired until a real frame returns. */
510 app->LastFrameAvailableTick = HAL_GetTick() - app->NoFrameFaultMs;
511 /* Recovery budget exhausted: report fatal to the task. */
512 if (app->RecoveryAttempt >= app->MaxRecoveryAttempts)
513 {
514 PMS25_MarkFaultInProgress();
515 PM25_LOG_PRINTF("[FATAL][PM25] recovery attempts exhausted (%u/%u) -> reporting fatal to task\r\n",
516 (unsigned)app->RecoveryAttempt,
517 (unsigned)app->MaxRecoveryAttempts);
518 app_fault_delay_ms(PMS25_FATAL_LOG_FLUSH_MS);
519 return PMS25_APP_RUNTIME_FATAL;
520 }
521 if (app->RecoveryDelayMs != 0U)
522 {
523 PMS25_AppSleepMs(app->RecoveryDelayMs);
524 }
525 return PMS25_APP_RUNTIME_OK;
526}
