81void log_write_bytes(const uint8_t *data, uint32_t len)
82{
83 if ((data == NULL) || (len == 0U)) {
84 return;
85 }
86 /* Drop the whole message if the ring buffer cannot hold it. */
87 if (len > log_free()) {
88 return;
89 }
90 /* Copy once or wrap at the end of the ring buffer. */
91 {
92 uint32_t idx = log_head & LOG_MASK;
93 uint32_t first = LOG_BUF_SIZE - idx;
94 if (first >= len) {
95 memcpy(&log_buf[idx], data, len);
96 } else {
97 memcpy(&log_buf[idx], data, first);
98 memcpy(&log_buf[0], data + first, len - first);
99 }
100 }
101 /* Publish the new head atomically against the IRQ consumer. */
102 __disable_irq();
103 log_head = (log_head + len) & LOG_MASK;
104 __enable_irq();
105 /* If TX is idle, send the first byte and let TXE continue the rest. */
106 if ((USART2->CR1 & USART_CR1_TXEIE) == 0U) {
107 if (log_tail != log_head) {
108 uint8_t c = log_buf[log_tail & LOG_MASK];
109 log_tail = (log_tail + 1U) & LOG_MASK;
110 USART2->DR = c;
111 USART2->CR1 |= USART_CR1_TXEIE;
112 }
113 }
114}

157void uart_log_irq_handler(void)
158{
159 /* RXNE is handled first because reading DR clears the receive event. */
160 if ((USART2->SR & USART_SR_RXNE) != 0U) {
161 char received = (char)(USART2->DR & 0xFFU);
162 /* Echo directly when TX is idle, otherwise queue it. */
163 if ((USART2->CR1 & USART_CR1_TXEIE) == 0U) {
164 USART2->DR = (uint8_t)received;
165 } else {
166 log_putc(received);
167 }
168 }
169 /* TXE drains one queued byte per interrupt. */
170 if ((USART2->SR & USART_SR_TXE) != 0U) {
171 if (log_tail != log_head) {
172 uint8_t c = log_buf[log_tail & LOG_MASK];
173 log_tail = (log_tail + 1U) & LOG_MASK;
174 USART2->DR = c;
175 } else {
176 USART2->CR1 &= ~USART_CR1_TXEIE;
177 }
178 }
