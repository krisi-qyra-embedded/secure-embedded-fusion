68void log_write_bytes(const uint8_t *data, uint32_t len)
69{
70 if (len == 0) return;
71 /* Drop the complete message rather than block when the ring is full. */
72 if (len > log_free()) return;
73 /* Split the copy only when it wraps around the ring boundary. */
74 uint32_t idx = log_head & LOG_MASK;
75 uint32_t first = LOG_BUF_SIZE - idx;
76 if (first >= len) {
77 memcpy(&log_buf[idx], data, len);
78 } else {
79 memcpy(&log_buf[idx], data, first);
80 memcpy(&log_buf[0], data + first, len - first);
81 }
82 /* Publish the new producer index atomically with respect to the IRQ consumer. */
83 __disable_irq();
84 log_head = (log_head + len) & LOG_MASK;
85 __enable_irq();
86 /* Kick transmission directly when TXE interrupts are currently disabled. */
87 if (!(USART2->CR1 & (1U << 7))) {
88 if (log_tail != log_head) {
89 uint8_t c = log_buf[log_tail & LOG_MASK];
90 log_tail = (log_tail + 1) & LOG_MASK;
91 USART2->DR = c;
92 USART2->CR1 |= (1U << 7);
93 }
94 }
95}
