13/*
14 * Combined watchdog / supervisor module for:
15 * - MAX31865 SPI/DMA heartbeat
16 * - MAX31865 main/task heartbeat
17 * - BME680 I2C heartbeat
18 * - MPU9250 I2C heartbeat
19 * - BME680 main/task heartbeat
20 * - MPU9250 main/task heartbeat
21 * - PM2.5 fault-capture participation (no heartbeat supervision)
22 * - DS18B20 fault-capture participation (no heartbeat supervision)
23 * - ESP32 UART AES-GCM fault-capture participation
24 * (progress timestamp recorded, no normal heartbeat enforcement)
25 *
26 * Design rule:
27 * supervisor_check_and_kick() is the only place that refreshes the IWDG.
28 * If any monitored heartbeat becomes stale, the supervisor stops refreshing
29 * the watchdog and lets the IWDG reset the MCU.
30 */

34#endif

145void escalate_to_iwdg_reset(const char *reason)
146{
147 uint32_t start_ms;
148 backup_write_marker(INTENTIONAL_RESET_TAG);
149 log_printf("ESCALATE -> IWDG reset: %s\r\n", reason ? reason : "(no reason)");
150 app_fault_delay_ms(2U);
151 start_ms = ms_get();
152 __DSB();
153 __ISB();
154 /*
155 * Preferred path:
156 * Stop refreshing IWDG and let the hardware watchdog reset the MCU.
157 */
158 for (;;) {
159 if ((ms_get() - start_ms) >= IWDG_RESET_FALLBACK_MS) {
160 log_printf("[FATAL] IWDG reset fallback expired -> forcing NVIC_SystemReset()\r\n");
161 app_fault_delay_ms(2U);
162 __DSB();
163 __ISB();
164 NVIC_SystemReset();
165 }
166 __WFI();
167 }
168}

309void supervisor_check_and_kick(void)
310{
311 uint32_t now = ms_get();
312 /*
313 * Fatal reset was requested by one sensor task.
314 *
315 * no reset immediately. Keeping the IWDG alive for a short bounded grace
316 * window so the other critical sensor task can also finish and print its
317 * 3/3 fault chain. Reset immediately if all currently active critical
318 * sensors have completed their fatal reporting.
319 */
320 if (iwdg_reset_requested != 0U) {
321 uint32_t age = now - iwdg_reset_request_ms;
322 uint32_t active;
323 uint32_t completed;
324 uint32_t in_progress;
325 uint32_t capture_completed;
326 uint32_t capture_in_progress;
327 uint32_t capture_count;
328 taskENTER_CRITICAL();
329 active =
330 system_fault_active_mask & SYSTEM_FAULT_MASK_CRITICAL;
331 completed =
332 system_fault_completed_mask & SYSTEM_FAULT_MASK_CRITICAL;
333 in_progress =
334 system_fault_in_progress_mask & SYSTEM_FAULT_MASK_CRITICAL;
335 /*
336 * During enrollment, keep adding newly detected faulty sensors.
337 * This fixes the MPU-first/BME-second race.
338 */
339 if (fault_capture_locked == 0U) {
340 fault_capture_mask |= active;
341 if (age >= FAULT_CAPTURE_ENROLL_MS) {
342 fault_capture_locked = 1U;
343 }
344 } else {
345 /*
346 * Even after lock, still allow already-faulting sensors to be included.
347 * This protects against task scheduling latency.
348 */
349 fault_capture_mask |= active;
350 }
351 taskEXIT_CRITICAL();
352 capture_completed =
353 completed & fault_capture_mask;
354 capture_in_progress =
355 in_progress & fault_capture_mask;
356 capture_count = fault_popcount32(fault_capture_mask);
357 /*
358 * Normal completion path:
359 * single fault -> reset after that one completes
360 * multi fault -> reset after all captured sensors complete
361 */
362 if ((fault_capture_mask != 0U) &&
363 ((capture_completed & fault_capture_mask) == fault_capture_mask) &&
364 (capture_in_progress == 0U) &&
365 (age >= FAULT_CAPTURE_ENROLL_MS)) {
366 SYSTEM_LOG_PRINTF("[FAULT][COMPLETE] capture_mask=0x%08lX active=0x%08lX in_progress=0x%08lX completed=0x%08lX count=%lu age=%lu last_owner=%u\r\n",
367 (unsigned long)fault_capture_mask,
368 (unsigned long)active,
369 (unsigned long)in_progress,
370 (unsigned long)completed,
371 (unsigned long)capture_count,
372 (unsigned long)age,
373 (unsigned)system_fault_owner);
374 app_fault_delay_ms(20U);
375 if (capture_count > 1U) {
376 escalate_to_iwdg_reset("multi fault capture complete");
377 } else {
378 escalate_to_iwdg_reset("single fault capture complete");
379 }
380 }
381 /*
382 * Hard safety path:
383 * Do not let the system freeze forever if one task never clears in_progress.
384 */
385 if (age >= FAULT_CAPTURE_HARD_MAX_MS) {
386 SYSTEM_LOG_PRINTF("[FAULT][HARD_TIMEOUT] capture_mask=0x%08lX active=0x%08lX in_progress=0x%08lX completed=0x%08lX count=%lu age=%lu last_owner=%u\r\n",
387 (unsigned long)fault_capture_mask,
388 (unsigned long)active,
389 (unsigned long)in_progress,
390 (unsigned long)completed,
391 (unsigned long)capture_count,
392 (unsigned long)age,
393 (unsigned)system_fault_owner);
394 app_fault_delay_ms(20U);
395 /*
396 * Safety cleanup before reset: if one captured task got trapped in
397 * in_progress, do not keep waiting. We are already in hard-timeout.
398 */
399 taskENTER_CRITICAL();
400 system_fault_completed_mask |= fault_capture_mask;
401 system_fault_in_progress_mask &= ~fault_capture_mask;
402 taskEXIT_CRITICAL();
403 escalate_to_iwdg_reset("fault capture hard timeout");
404 }
405 if (iwdg_refresh() != IWDG_OK) {
406 IWDG_ErrorHandler("iwdg_refresh failed during fault capture", IWDG_ERR_HW);
407 escalate_to_iwdg_reset("iwdg_refresh failed during fault capture");
408 }
409 return;
410 }
411 /* Normal heartbeat checks */
412 uint32_t age_spi = now - hb_spi_ts;
413 uint32_t age_main = now - hb_main_ts;
414 uint32_t age_bme680 = now - hb_bme680_ts; /* wrap-safe */
415 uint32_t age_bme680main = now - hb_bme680main_ts;
416 uint32_t age_mpu9250 = now - hb_mpu9250_ts;
417 uint32_t age_mpu9250main = now - hb_mpu9250main_ts;
418 uint32_t age_esp32_uart_aes_gcm = now - hb_esp32_uart_aes_gcm_ts;
419 /* MAX31865 SPI/DMA */
420 if (age_spi > HEARTBEAT_WINDOW_MS) {
421 log_printf("SUPERVISOR: MAX31865 SPI/DMA heartbeat missed (age=%u ms)\r\n",
422 (unsigned)age_spi);
423 app_fault_delay_ms(2);
424 escalate_to_iwdg_reset("MAX31865 SPI/DMA heartbeat missed");
425 }
426 /* MAX31865 task */
427 if (age_main > HEARTBEAT_WINDOW_MS) {
428 log_printf("SUPERVISOR: MAX31865 main/task heartbeat missed (age=%u ms)\r\n",
429 (unsigned)age_main);
430 app_fault_delay_ms(2);
431 escalate_to_iwdg_reset("MAX31865 main/task heartbeat missed");
432 }
433 /* BME680 driver */
434 if (age_bme680 > HEARTBEAT_WINDOW_MS) {
435 log_printf("SUPERVISOR: BME680 I2C heartbeat missed (age=%u ms)\r\n",
436 (unsigned)age_bme680);
437 app_fault_delay_ms(2);
438 escalate_to_iwdg_reset("BME680 I2C heartbeat missed");
439 }
440 /* BME680 task */
441 if (age_bme680main > HEARTBEAT_WINDOW_MS) {
442 log_printf("SUPERVISOR: BME680 main/task heartbeat missed (age=%u ms)\r\n",
443 (unsigned)age_bme680main);
444 app_fault_delay_ms(2);
445 escalate_to_iwdg_reset("BME680 main/task heartbeat missed");
446 }
447 /* MPU9250 driver */
448 if (age_mpu9250 > HEARTBEAT_WINDOW_MS) {
449 log_printf("SUPERVISOR: MPU9250 I2C heartbeat missed (age=%u ms)\r\n",
450 (unsigned)age_mpu9250);
451 app_fault_delay_ms(2);
452 escalate_to_iwdg_reset("MPU9250 I2C heartbeat missed");
453 }
454 /* MPU9250 task */
455 if (age_mpu9250main > HEARTBEAT_WINDOW_MS) {
456 log_printf("SUPERVISOR: MPU9250 main/task heartbeat missed (age=%u ms)\r\n",
457 (unsigned)age_mpu9250main);
458 app_fault_delay_ms(2);
459 escalate_to_iwdg_reset("MPU9250 main/task heartbeat missed");
460 }
461 /* ESP32 encrypted link */
462if (age_esp32_uart_aes_gcm > HEARTBEAT_WINDOW_MS) {
463 log_printf(
464 "SUPERVISOR: ESP32 UART AES-GCM heartbeat missed "
465 "(age=%u ms)\r\n",
466 (unsigned)age_esp32_uart_aes_gcm);
467 app_fault_delay_ms(2U);
468 escalate_to_iwdg_reset(
469 "ESP32 UART AES-GCM heartbeat missed");
470 }
471 /* All checks passed, refresh watchdog. */
472 if (iwdg_refresh() != IWDG_OK) {
473 IWDG_ErrorHandler("iwdg_refresh failed", IWDG_ERR_HW);
474 escalate_to_iwdg_reset("iwdg_refresh failed");
475 }

564/* Wait until IWDG update bits clear. */
565static iwdg_status_t wait_iwdg_ready(uint32_t budget)
566{
567 const uint32_t timeout_ms = (budget == 0U) ? iwdg_poll_budget : budget;
568 const uint32_t start = ms_get();
569 while (READ_BIT(IWDG->SR, IWDG_SR_PVU | IWDG_SR_RVU) != 0U) {
570 if ((ms_get() - start) >= timeout_ms) {
571 return IWDG_ERR_TIMEOUT;
572 }
573 __DSB();
574 __ISB();
575 __WFI();
576 }
577 return IWDG_OK;
578}

588iwdg_status_t iwdg_init(uint32_t prescaler, uint32_t reload, uint32_t poll_budget)
589{
590 if (reload > IWDG_RELOAD_MAX) {
591 IWDG_ErrorHandler("IWDG: invalid reload", IWDG_ERR_INVALID_PARAM);
592 return IWDG_ERR_INVALID_PARAM;
593 }
594 if (iwdg_inited) {
595 return IWDG_ERR_ALREADY_INIT;
596 }
597 if (poll_budget != 0U) {
598 iwdg_poll_budget = poll_budget;
599 }
600 /* Enable IWDG. */
601 IWDG->KR = IWDG_KEY_ENABLE;
602 __asm volatile ("nop");
603 /* Enable PR/RLR write access. */
604 IWDG->KR = IWDG_KEY_WR_ACCESS_ENABLE;
605 IWDG->PR = prescaler;
606 IWDG->RLR = reload;
607 if (wait_iwdg_ready(0U) != IWDG_OK) {
608 IWDG_ErrorHandler("IWDG: timeout waiting for PR/RLR update", IWDG_ERR_TIMEOUT);
609 return IWDG_ERR_TIMEOUT;
610 }
611 /* Load new reload value. */
612 IWDG->KR = IWDG_KEY_RELOAD;
613#if defined(IWDG_RLR)
614 if ((IWDG->RLR & IWDG_RELOAD_MAX) != (reload & IWDG_RELOAD_MAX)) {
615 IWDG_ErrorHandler("IWDG: RLR readback mismatch", IWDG_ERR_HW);
616 }
617#endif
618 iwdg_inited = true;
619 return IWDG_OK;
620}

627iwdg_status_t iwdg_refresh(void)
628{
629 if (!iwdg_inited) {
630 IWDG_ErrorHandler("IWDG: refresh called before init", IWDG_ERR_INVALID_PARAM);
631 return IWDG_ERR_INVALID_PARAM;
632 }
633 IWDG->KR = IWDG_KEY_RELOAD;
634 if (READ_BIT(IWDG->SR, IWDG_SR_PVU | IWDG_SR_RVU) != 0U) {
635 if (wait_iwdg_ready(iwdg_poll_budget / 10U) != IWDG_OK) {
636 IWDG_ErrorHandler("IWDG: refresh caused SR pending", IWDG_ERR_TIMEOUT);
637 return IWDG_ERR_TIMEOUT;
638 }
639 }
640 return IWDG_OK;
641}
