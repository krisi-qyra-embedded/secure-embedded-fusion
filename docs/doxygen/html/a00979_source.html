108/* Timer-derived nonce component. */
113uint32_t read_tim1_hashed_loop(void)
114{
115 /* TIM1 contributes the dynamic timer half of the nonce. */
116 uint16_t cnt = read_tim1_counter();
117 uint8_t bytes[4];
118 bytes[0] = (uint8_t)(cnt & 0xFF);
119 bytes[1] = (uint8_t)((cnt >> 8) & 0xFF);
120 bytes[2] = (uint8_t)(((cnt ^ (cnt >> 3)) & 0xFF));
121 bytes[3] = (uint8_t)(((cnt * 7u) ^ (cnt >> 5)) & 0xFF);
122 uint32_t hash = 0x12345678U;
123 for (size_t i = 0; i < sizeof(bytes); ++i) {
124 hash = (hash << 5) | (hash >> 27);
125 hash ^= (uint32_t)bytes[i];
126 hash += (uint32_t)(i + 1);
127 uint32_t rotated_cnt = (((uint32_t)cnt << (i)) | ((uint32_t)cnt >> (16 - (i)))) & 0xFFFFU;
128 hash ^= rotated_cnt;
129 }
130 hash ^= (uint32_t)cnt;
131 return hash;
132}
